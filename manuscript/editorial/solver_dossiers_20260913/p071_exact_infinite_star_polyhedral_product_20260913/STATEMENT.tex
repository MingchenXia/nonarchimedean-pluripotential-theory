% Book-local P07.1 candidate, 13 September 2026.
% This theorem combines the exact infinite CLD star with one convex
% polyhedral factor.  It makes no novelty or priority claim.

\begin{theorem}
\label{thm:p071-infinite-star-polyhedral-collapse}
Let \(B\ne\varnothing\), let \(X_B\) be the exact
Chambert--Loir--Ducros star of
\Cref{thm:p071-infinite-star-collapse-candidate}, and let \(P\) be a
nonempty convex polyhedron in a finite-dimensional real affine space
\(L_P\).  Put
\[
                              X=X_B\times P.
\]

We use the following exact product generator sheaf.  On the product of an
ordinary interval in a branch with a relatively open convex subset of
\(P\), a real-valued generator is locally the restriction of an ambient
smooth function on \(\mathbf R\times L_P\) whose full Hessian in the branch
and polyhedral variables is positive semidefinite.  At a full endpoint
\(-\infty\), the generator is additionally constant in the branch variable
on a terminal collar, locally uniformly in the polyhedral variable.

At a node point, take a standard star neighbourhood \(U\), with finite
exceptional set \(A\subset B\) and exceptional collars
\((-R_a,0]\), and a relatively open convex set \(V\subset P\).  Write
\[
 g_b(t,y)=g(p_b(t),y),\qquad c_g(y)=g(\eta,y).
\]
A generator on \(U\times V\) is required to have jointly ambient-smooth
convex branch functions, one finite set \(K\subset B\setminus A\) such that
\begin{equation}
 g_b(t,y)=c_g(y)
 \quad
 \bigl(b\in B\setminus(A\cup K),\
       t\in[-\infty,0],\ y\in V\bigr),
 \label{eq:p071-infinite-polyhedral-smooth-finite-activity}
\end{equation}
and the exact balance law
\begin{equation}
 \sum_{a\in A}\partial_tg_a(0,y)
   +\sum_{b\in K}\partial_tg_b(0,y)=0
 \qquad(y\in V).
 \label{eq:p071-infinite-polyhedral-smooth-balance}
\end{equation}
On every full active branch, the terminal-collar condition is also imposed.
Let \(\mathcal G_{X,0}\) be the sheaf defined by these local conditions,
let \(\mathcal P_{X,1}\) be its sheaf of local pointwise decreasing
sequential limits, and let \(\mathcal P_X\) be the smallest subsheaf of
\(\mathbf R\cup\{-\infty\}\)-valued functions which contains
\(\mathcal G_{X,0}\) and is closed under such limits.

Define a comparison sheaf \(\mathcal J_X\) as follows.  Its functions are
upper semicontinuous.  Every branch function \(u_b(t,y)\), with the node
value inserted at \(t=0\), is extended-real jointly convex on every ordinary
convex subset of its branch--polyhedral domain.  On a full branch,
\begin{equation}
 u_b(-\infty,y)=\lim_{t\to-\infty}u_b(t,y)
 \quad\text{in }\mathbf R\cup\{-\infty\}.
 \label{eq:p071-infinite-polyhedral-endpoint-trace}
\end{equation}
Near a node, require one standard product \(U\times V\) and one countable
set \(K\subset B\setminus A\) such that
\begin{equation}
 u_b(t,y)=c(y):=u(\eta,y)
 \quad
 \bigl(b\in B\setminus(A\cup K),\
       t\in[-\infty,0],\ y\in V\bigr).
 \label{eq:p071-infinite-polyhedral-countable-activity}
\end{equation}
Finally, let \(F\subset B\setminus A\) be finite, choose \(r_b>0\) for
\(b\in F\), choose \(0<\rho_a<R_a\) for \(a\in A\), and put
\[
 \Lambda=\sum_{b\in F}\frac1{r_b}
              +\sum_{a\in A}\frac1{\rho_a}.
\]
If \(\Lambda>0\), \(c(y_0)\in\mathbf R\), and
\begin{equation}
 \Lambda y_0=
   \sum_{b\in F}\frac{y_b}{r_b}
      +\sum_{a\in A}\frac{y_a}{\rho_a},
 \label{eq:p071-infinite-polyhedral-barycentre}
\end{equation}
for points \(y_0,y_b,y_a\in V\), require the exact product junction
inequality
\begin{equation}
 \Lambda c(y_0)\leq
   \sum_{b\in F}\frac{u_b(-r_b,y_b)}{r_b}
      +\sum_{a\in A}\frac{u_a(-\rho_a,y_a)}{\rho_a}.
 \label{eq:p071-infinite-polyhedral-junction}
\end{equation}
The case \(F=A=\varnothing\) is vacuous.  Membership in
\(\mathcal J_X\) means that every point has a neighbourhood on which these
conditions hold.

Then
\begin{equation}
                 \mathcal P_{X,1}=\mathcal P_X=\mathcal J_X.
 \label{eq:p071-infinite-polyhedral-collapse}
\end{equation}
More precisely, every germ in these sheaves has a pointwise decreasing
approximation by members of \(\mathcal G_{X,0}\) on one fixed smaller
neighbourhood.  At a node, this neighbourhood may be chosen as
\(U'\times V'\), with \(U'\) standard and \(V'\) relatively open and
convex in \(P\).  At a full endpoint it may be chosen as one fixed terminal
product neighbourhood.

The theorem permits an arbitrary branch set, all faces and corners of one
convex polyhedron, mixed branch--polyhedral Hessian terms, and the exact CLD
terminal collars.  It does not cover a separately convex generator sheaf,
an additional nonconstant Stone-affine direction, a non-CLD infinite star,
a nonconvex or branched chamber gluing, or a nonambient branch structure.
Consequently it narrows but does not retire P07.1.
\end{theorem}
