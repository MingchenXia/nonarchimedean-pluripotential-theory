% Book-local P07.1 theorem, 13 September 2026.
% This theorem combines a locally constant Stone parameter with the exact
% infinite CLD star of Section 6.7.3.  It makes no novelty or priority claim.

\begin{theorem}
\label{thm:p071-stone-infinite-star-collapse}
Let \(Z\) be a locally compact Hausdorff space with a basis of clopen
subsets, let \(B\ne\varnothing\) be an arbitrary set, and let \(X_B\) be
the exact Chambert--Loir--Ducros star of
\Cref{thm:p071-infinite-star-collapse-candidate}.  Thus
\[
 X_B=(B\times[-\infty,0])/{\sim},
 \qquad p_b(0)=\eta\quad(b\in B),
\]
with the cofinite-branch topology, the CLD smooth structure, and the
comparison sheaf \(\mathscr J_B\) of that theorem.  Put
\[
                              X=Z\times X_B .
\]

For an ordinary open \(W\subset X\), let \(\mathcal G_{X,0}(W)\) be the
real-valued functions which locally, after shrinking the \(Z\)-variable
to a clopen neighbourhood, are independent of that variable and are
smooth psh functions on \(X_B\).  Let \(\mathcal P_{X,1}\) be the sheaf
of local pointwise decreasing limits of sequences in
\(\mathcal G_{X,0}\), and let \(\mathcal P_X\) be the smallest subsheaf
of \(\mathbf R\cup\{-\infty\}\)-valued functions which contains
\(\mathcal G_{X,0}\) and is closed under such limits.

Define a comparison sheaf \(\mathcal C_X\) as follows.  Away from
\(Z\times\{\eta\}\), a section \(u\) belongs to \(\mathcal C_X\) if
locally there are a compact clopen \(C\subset Z\), a compact metrizable
zero-dimensional space \(Y\), a continuous surjection
\(q:C\twoheadrightarrow Y\), and an upper semicontinuous function
\begin{equation}
 \bar u:Y\times V\longrightarrow\mathbf R\cup\{-\infty\},
 \qquad
 u(z,x)=\bar u(q(z),x),
 \label{eq:p071-stone-infinite-factor-away}
\end{equation}
such that every fibre \(\bar u_\zeta\) belongs to
\(\mathscr J_B(V)\).

Near a node point \((z_0,\eta)\), require in addition that the
factorization hold on \(C\times U\), where \(U\) is a standard
neighbourhood with finite exceptional set \(A\subset B\), and that there
be one countable set \(K\subset B\setminus A\), independent of
\(\zeta\), for which
\begin{equation}
 \bar u(\zeta,p_b(t))=\bar u(\zeta,\eta)
 \quad
 \bigl(\zeta\in Y,\ b\in B\setminus(A\cup K),\
       t\in[-\infty,0]\bigr).
 \label{eq:p071-stone-infinite-common-activity}
\end{equation}
The empty set and finite sets are allowed as countable sets.

Then \(\mathcal C_X\) is a sheaf and
\begin{equation}
                 \mathcal P_{X,1}=\mathcal P_X=\mathcal C_X .
 \label{eq:p071-stone-infinite-collapse}
\end{equation}
More precisely, every germ in these sheaves has a pointwise decreasing
approximation by members of \(\mathcal G_{X,0}\) on one fixed smaller
product neighbourhood.  At a node point, the neighbourhood may be
chosen as \(C\times U'\), with \(C\) compact clopen and \(U'\) standard.

The common countable set in
\eqref{eq:p071-stone-infinite-common-activity} is intrinsic to sequential
generation: one sequence of CLD-smooth functions can activate only a
countable union of finite branch sets.  The theorem allows both a
possibly nonmetrizable Stone parameter and an arbitrarily large exact
CLD branch set.  It does not cover nonconstant affine directions on the
Stone factor, a non-CLD infinite star, an additional polyhedral factor,
or nonambient branch structures.  Consequently it narrows but does not
retire P07.1.
\end{theorem}
