\begin{lemma}
\label{lem:p21p3-type-one-zero-atom}
Let $K$ be a complete nontrivially valued non-Archimedean field, let
$Y$ be a good separated strict rig-smooth $K$-analytic curve without
boundary, and let $u$ be a continuous CLD-plurisubharmonic function on
$Y$. Then the positive curvature measure
\[
                 \mu_u:=\mathrm d'\mathrm d''[u]
\]
has no atom at any type-$1$ point of $Y$.
\end{lemma}

\begin{proposition}
\label{prop:p21p3-curve-rigidity}
Let $K$ be an algebraically closed complete nontrivially valued
non-Archimedean field and let $X$ be a good separated strict reduced
$K$-analytic space, pure of dimension one and without boundary. Assume
that the normalization of each irreducible component of $X$ is
rig-smooth. For every irreducible component $X_i$, write
\[
        \nu_i:\widetilde X_i\longrightarrow X_i\hookrightarrow X,
        \qquad
        \delta_{X_i}:=(\nu_i)_*\delta_{\widetilde X_i}.
\]
Then the following assertions hold.

\begin{enumerate}
\item\label{itm:p21p3-curve:classification}
Every positive $\mathrm d'$- and $\mathrm d''$-closed CLD current $T$
of bidegree $(0,0)$ on $X$ has a unique expression
\begin{equation}
\label{eq:p21p3-curve-classification}
                         T=\sum_i c_i\delta_{X_i},
                         \qquad c_i\in\mathbf R_{\geq 0}.
\end{equation}
The sum is locally finite. In particular, if $X$ is connected and
rig-smooth, then $T=c\delta_X$ for a unique
$c\in\mathbf R_{\geq 0}$.

\item\label{itm:p21p3-curve:bedford-taylor}
Let $W\subseteq X$ be open and let $U,V$ be arbitrary continuous
CLD-plurisubharmonic functions on $W$. If
\begin{equation}
\label{eq:p21p3-curve-equal-curvature}
           \mathrm d'\mathrm d''[U]
           =\mathrm d'\mathrm d''[V],
\end{equation}
then, for every $T$ as in
\cref{itm:p21p3-curve:classification},
\begin{equation}
\label{eq:p21p3-curve-bt-conclusion}
             \mathrm d'\mathrm d''\bigl((U-V)T|_W\bigr)=0.
\end{equation}
Consequently every positive closed CLD current of bidegree $(0,0)$ in
this one-dimensional analytic setting is a Bedford--Taylor current in the
sense of \cite[Definition~8.6.2]{CLD25}.
\end{enumerate}
\end{proposition}

\begin{remark}
\label{rem:p21p3-curve-scope}
The classification and Bedford--Taylor assertions make no claim over a
non-algebraically-closed or trivially valued ground field, or for a reduced
curve whose component normalizations are not rig-smooth.  They also make
no claim in analytic dimension at least two.  The type-$1$ zero-atom lemma
itself does not require algebraic closedness.
\end{remark}
