\begin{theorem}
\label{thm:wanner-curve-comparison}
Let \(K\) be algebraically closed, complete, and nontrivially valued, let
\(C\) be a smooth proper algebraic curve, and let \(W\subset C^{\mathrm{an}}\)
be open.  For a continuous function \(u:W\to\mathbf R\),
\[
                  \dprimeddprime[u]\geq0
        \quad\Longleftrightarrow\quad
                  u\text{ is Thuillier-subharmonic}.
\]
Consequently every continuous CLD-psh function on \(W\) is
Thuillier-subharmonic.
\end{theorem}

% Source: manuscript/parts/part2/ch19_cld_examples.tex.
\begin{corollary}
\label{cor:ch19-continuous-zero-current-harmonic}
Under the hypotheses of \Cref{thm:wanner-curve-comparison}, let
$h:W\to\mathbf R$ be continuous.  If
\[
                         \dprimeddprime[h]=0,
\]
then $h$ is Thuillier harmonic, hence harmonic in the sense of
Gubler--Jell--Rabinoff and classically pluriharmonic in the sense of
Gubler--Rabinoff.
\end{corollary}

\begin{lemma}[First-current rigidity on a published curve]
\label{lem:p062-curve-first-current-rigidity}
Let \(K\), \(C\), and \(W\subset C^{\mathrm{an}}\) be as in
\Cref{thm:wanner-curve-comparison}.  If \(b\in C^0(W,\mathbf R)\) satisfies
\[
          \mathrm d'\mathrm d''[b]=0,\qquad \mathrm d'[b]=0,
\]
then \(b\) is locally constant.
\end{lemma}

\begin{proposition}
\label{prop:p062-open-polyannulus-rigid-base}
Let \(K\) be a complete non-Archimedean field with nontrivial absolute
value, let \(m\geq1\), and let \(B\) be a good, topologically separated,
boundaryless, pure \(m\)-dimensional strict \(K\)-analytic space.  Assume
that
\begin{enumerate}
\item every \(b\in C^0(B,\mathbf R)\) satisfying
  \(\mathrm d'\mathrm d''[b]=0\) is classically pluriharmonic; and
\item every \(b\in C^0(B,\mathbf R)\) satisfying
  \(\mathrm d'\mathrm d''[b]=0\) and \(\mathrm d'[b]=0\) is locally
  constant.
\end{enumerate}
For \(s\geq0\), let
\[
 A_i=\{z\in(\mathbf G_{\mathrm m,K})^{\mathrm{an}}:
                     r_i<|T_i(z)|<R_i\},
 \qquad 0<r_i<R_i,\quad r_i,R_i\in|K^\times|,
\]
and put
\[
 X_s=B\times_K\prod_{i=1}^sA_i,
 \qquad p:X_s\longrightarrow B,
 \qquad q_i:X_s\longrightarrow A_i.
\]
If \(h\in C^0(X_s,\mathbf R)\) satisfies
\(\mathrm d'\mathrm d''[h]=0\), then there are unique continuous
functions \(b,a_1,\ldots,a_s:B\to\mathbf R\) such that
\[
 h=p^*b+\sum_{i=1}^s(a_i\circ p)q_i^*\log|T_i|,
 \qquad \mathrm d'\mathrm d''[b]=0,
\]
and every \(a_i\) is locally constant.  In particular, \(h\) is
classically pluriharmonic.  Moreover, every
\(u\in C^0(X_s,\mathbf R)\) satisfying
\[
             \mathrm d'\mathrm d''[u]=0,
             \qquad \mathrm d'[u]=0
\]
is locally constant.
\end{proposition}

\begin{proposition}
\label{prop:p062-mixed-exact-bundle-tower-transfer}
Let $K$ be a complete non-Archimedean field with nontrivial absolute
value, let $B_0$ be a positive-dimensional good, topologically separated,
pure strict $K$-analytic space, and let $\ell\geq1$.  Suppose that
\[
 B_\ell\xrightarrow{\pi_\ell}B_{\ell-1}
 \longrightarrow\cdots\longrightarrow B_1
 \xrightarrow{\pi_1}B_0
\]
is a finite tower in which every $B_j$ is a positive-dimensional good,
topologically separated, pure strict $K$-analytic space.  At every stage
assume that exactly one of the following descriptions is specified.
\begin{enumerate}
\item The morphism $\pi_j$ is a locally trivial standard strict
open-polydisc bundle as in
\Cref{prop:p062-open-polydisc-bundle-descent}; in particular, its positive
fibre dimension and chart radii satisfy the conditions there.
\item The base $B_{j-1}$ is locally holomorphically separated and
paracompact, and $\pi_j$ is a locally trivial analytic projective-space
bundle as in \Cref{prop:p062-projective-bundle-descent}, with fibre
$(\mathbf P_K^{q_j})^{\mathrm{an}}$ for some $q_j\geq1$.
\item The base $B_{j-1}$ is locally holomorphically separated and
paracompact, and $\pi_j$ is a locally trivial smooth proper split-toric
bundle as in
\Cref{prop:p062-smooth-proper-split-toric-bundle-descent}, with fibre
$X_{\Sigma_j}^{\mathrm{an}}$ for a smooth proper split toric $K$-variety
$X_{\Sigma_j}$ of dimension $q_j\geq1$.
\item The field $K$ is algebraically closed, the base $B_{j-1}$ is locally
holomorphically separated and paracompact, and $\pi_j$ is a locally trivial
proper smooth curve bundle as in
\Cref{prop:p062-proper-smooth-curve-bundle-descent}, with fibre
$D_j^{\mathrm{an}}$ for a nonempty geometrically connected smooth proper
algebraic $K$-curve $D_j$.
\end{enumerate}
Put $\Pi=\pi_1\circ\cdots\circ\pi_\ell$.  If
$h\in C^0(B_\ell,\mathbf R)$ satisfies
$\mathrm d'\mathrm d''[h]=0$, then there is a unique
$b\in C^0(B_0,\mathbf R)$ such that
\[
                  h=\Pi^*b,
                  \qquad \mathrm d'\mathrm d''[b]=0.
\]
The following properties transfer from $B_0$ to $B_\ell$.
\begin{enumerate}
\item If every continuous zero-current function on $B_0$ is classically
pluriharmonic, then every such function on $B_\ell$ is classically
pluriharmonic.
\item If every continuous zero-current function on $B_0$ is
$\mathbf R$-harmonic, then every such function on $B_\ell$ is
$\mathbf R$-harmonic, hence locally $\mathbf R$-PL and
Gubler--Jell--Rabinoff harmonic.
\item If every $u\in C^0(B_0,\mathbf R)$ satisfying
\[
             \mathrm d'\mathrm d''[u]=0,
             \qquad \mathrm d'[u]=0
\]
is locally constant, then the same implication holds with $B_\ell$ in
place of $B_0$.
\end{enumerate}
\end{proposition}
