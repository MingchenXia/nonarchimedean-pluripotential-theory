\begin{proof}[of
\Cref{prop:p062-stage-local-polyannulus-smooth-rigidity-transfer}]
Fix $U\in\mathcal U$ and use its displayed product chart.  The induced
analytic structure on $U$ is good, topologically separated, boundaryless,
pure, and strict.  Conditions \textup{(2)} and \textup{(3)} therefore imply
the two base hypotheses of
\Cref{prop:p062-open-polyannulus-rigid-base}.  Applied to
$h|_{\pi^{-1}(U)}$, that proposition gives unique continuous functions
$b_U,a_{U,1},\ldots,a_{U,s_U}$ on $U$ such that
\begin{equation}
\label{eq:p062-stage-local-annular-formula}
 h|_{\pi^{-1}(U)}
 =\pi^*b_U+
   \sum_{i=1}^{s_U}(a_{U,i}\circ\pi)\log|T_{U,i}|,
 \qquad \mathrm d'\mathrm d''[b_U]=0,
\end{equation}
and every $a_{U,i}$ is locally constant.  The same proposition already
makes the restricted function classically pluriharmonic.

By condition \textup{(2)}, $b_U$ is ordinary CLD-smooth with zero smooth
Hessian.  Hence so is $\pi^*b_U$ by
\Cref{prop:cld-form-pullback}.  Every annular coordinate $T_{U,i}$ is
invertible, so
\Cref{lem:ch15-log-smooth,prop:cld-form-pullback} show that its logarithm
in \eqref{eq:p062-stage-local-annular-formula} is ordinary CLD-smooth with
zero smooth Hessian.  Multiplication by the locally constant coefficient
$a_{U,i}\circ\pi$ preserves both properties.  Thus $h$ is ordinary
CLD-smooth with zero smooth Hessian on every $\pi^{-1}(U)$, and hence on
$Y$.

If also $\mathrm d'[h]=0$, the final assertion of
\Cref{prop:p062-open-polyannulus-rigid-base} makes
$h|_{\pi^{-1}(U)}$ locally constant for every $U$.  This is a local
property, so $h$ is locally constant on $Y$.  The proof used only the
individual product charts, which proves the final scope assertion.
\end{proof}

\begin{proof}[of
\Cref{thm:p213-residual-picard-stage-local-annular-tower}]
We prove by induction on $j$ the following three assertions on $B_j$:
every continuous zero-current function is classically pluriharmonic; every
such function is ordinary CLD-smooth with zero smooth Hessian; and adding
vanishing of its first current forces it to be locally constant.

For $j=0$, the first assertion is
\Cref{cor:ch19-continuous-zero-current-harmonic}, the second is
\Cref{prop:p213-residual-picard-smooth-curve}, and the third is
\Cref{lem:p062-curve-first-current-rigidity}.

Suppose the three assertions hold on $B_{j-1}$.  If $\pi_j$ is an exact
bundle stage, apply
\Cref{prop:p062-mixed-exact-bundle-tower-transfer} with a tower of length
one.  A zero-current function $h$ on $B_j$ has the form
\[
                              h=\pi_j^*b
\]
for a unique zero-current function $b$ on $B_{j-1}$.  The first transfer
assertion of that proposition and the induction hypothesis make $h$
classically pluriharmonic.  The induction smoothness assertion and
\Cref{prop:cld-form-pullback} make $h$ ordinary CLD-smooth with zero smooth
Hessian.  The third transfer assertion gives first-current rigidity on
$B_j$.

If $\pi_j$ is an annular stage, its specified cover and the induction
ambient hypotheses place it exactly in
\Cref{prop:p062-stage-local-polyannulus-smooth-rigidity-transfer}.  That
proposition supplies all three assertions on $B_j$.  This completes the
induction and proves the conclusions at $j=\ell$.

Notice that the induction supplies whole-space conclusions on each $B_j$;
it does not promote them to all ordinary opens of $B_j$.  The separately
stated conditions on every member of $\mathcal U_j$ are therefore used
essentially, proving the final qualification.
\end{proof}

\begin{proof}[of
\Cref{cor:p213-residual-picard-stage-local-annular-tower-bt}]
Put $h=u-v$.  Equality of the Hessian currents gives
$\mathrm d'\mathrm d''[h]=0$.  The tower theorem makes $h$ ordinary
CLD-smooth with zero smooth Hessian.  Apply \Cref{cor:ch16-smooth-h}.
\end{proof}

\begin{proof}[of
\Cref{cor:p213-residual-picard-stage-local-annular-tower-cover-bt}]
Put $h=u-v$.  On every member $B$ of the displayed cover, apply
\Cref{thm:p213-residual-picard-stage-local-annular-tower} to $h|_B$.
Thus $h$ is ordinary CLD-smooth with zero smooth Hessian locally on $V$.
These are sheaf-local properties, so they hold on $V$.  Apply
\Cref{cor:ch16-smooth-h} directly to $T,u,v$ on $V$.
\end{proof}

\begin{proof}[of
\Cref{cor:p213-residual-picard-stage-local-annular-tower-basis-bt}]
For an ordinary open $V\subseteq X$, the basis members contained in $V$
form a cover to which
\Cref{cor:p213-residual-picard-stage-local-annular-tower-cover-bt}
applies.
\end{proof}
