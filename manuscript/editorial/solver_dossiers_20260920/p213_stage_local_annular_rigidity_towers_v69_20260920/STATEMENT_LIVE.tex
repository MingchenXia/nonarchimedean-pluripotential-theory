\begin{proposition}
\label{prop:p062-stage-local-polyannulus-smooth-rigidity-transfer}
Let $K$ be a complete non-Archimedean field with nontrivial absolute value,
let $m\geq1$, and let $B$ be a good, topologically separated,
boundaryless, pure $m$-dimensional strict $K$-analytic space.  Let
$\pi:Y\to B$ be a morphism, where $Y$ is a good, topologically separated,
boundaryless, pure strict $K$-analytic space.  Suppose that $B$ has an
analytic open cover $\mathcal U$ such that, for every $U\in\mathcal U$,
the following conditions hold.
\begin{enumerate}
\item There is an integer $s_U\geq1$, together with radii
\[
 0<r_{U,i}<R_{U,i},\qquad r_{U,i},R_{U,i}\in|K^\times|
 \quad(1\leq i\leq s_U),
\]
and an isomorphism over $U$
\[
 \pi^{-1}(U)\simeq
 U\times_K\prod_{i=1}^{s_U}
 \{z\in(\mathbf G_{\mathrm m,K})^{\mathrm{an}}:
                  r_{U,i}<|T_{U,i}(z)|<R_{U,i}\}.
\]
\item Every $b\in C^0(U,\mathbf R)$ satisfying
$\mathrm d'\mathrm d''[b]=0$ is classically pluriharmonic and ordinary
CLD-smooth, with $\mathrm d'\mathrm d''b=0$ as a smooth form.
\item Every $b\in C^0(U,\mathbf R)$ satisfying
\[
             \mathrm d'\mathrm d''[b]=0,
             \qquad \mathrm d'[b]=0
\]
is locally constant.
\end{enumerate}
Then every $h\in C^0(Y,\mathbf R)$ satisfying
$\mathrm d'\mathrm d''[h]=0$ is classically pluriharmonic and ordinary
CLD-smooth, with $\mathrm d'\mathrm d''h=0$ as a smooth form.  Moreover,
every such $h$ with $\mathrm d'[h]=0$ is locally constant.  No compatibility
among the displayed annular coordinates and no global logarithmic
coefficient formula are asserted.
\end{proposition}

\begin{theorem}
\label{thm:p213-residual-picard-stage-local-annular-tower}
Let $K,C,W$ be as in
\Cref{prop:p213-residual-picard-smooth-curve}, and assume
\eqref{eq:p213-residual-picard-vanishing}.  Let $\ell\geq1$, and suppose
that
\[
 B_\ell\xrightarrow{\pi_\ell}B_{\ell-1}
 \longrightarrow\cdots\longrightarrow B_1
 \xrightarrow{\pi_1}B_0=W
\]
is a finite tower in which every $B_j$ is a positive-dimensional good,
topologically separated, boundaryless, pure strict $K$-analytic space.
At every stage assume that exactly one of the following descriptions is
specified.
\begin{enumerate}
\item The morphism $\pi_j$ is one of the four exact bundle types appearing
in \Cref{prop:p062-mixed-exact-bundle-tower-transfer}, with all hypotheses
of the corresponding source proposition retained.
\item The morphism $\pi_j$ is an analytically locally trivial finite
standard strict open-polyannulus bundle, and there is a trivializing
analytic open cover $\mathcal U_j$ of $B_{j-1}$ such that every
$U\in\mathcal U_j$ satisfies conditions \textup{(2)} and \textup{(3)} of
\Cref{prop:p062-stage-local-polyannulus-smooth-rigidity-transfer}.
\end{enumerate}
Then every $h\in C^0(B_\ell,\mathbf R)$ satisfying
$\mathrm d'\mathrm d''[h]=0$ is classically pluriharmonic and ordinary
CLD-smooth, with $\mathrm d'\mathrm d''h=0$ as a smooth form.  Moreover,
every such $h$ with $\mathrm d'[h]=0$ is locally constant.

The stage-local hypotheses on the members of $\mathcal U_j$ are not
claimed to follow merely from the corresponding whole-space conclusions
on $B_{j-1}$.  In particular, the theorem does not assert the result for
an arbitrary locally trivial annular stage over an arbitrary intermediate
base open.
\end{theorem}

\begin{corollary}
\label{cor:p213-residual-picard-stage-local-annular-tower-bt}
In the setting of
\Cref{thm:p213-residual-picard-stage-local-annular-tower}, assume that
$B=B_\ell$ has pure dimension $n\geq2$.  Let $T$ be a positive
$\mathrm d'$- and $\mathrm d''$-closed ordinary CLD current of bidegree
$(p,p)$, $0\leq p\leq n-1$, and let $u,v\in C^0(B,\mathbf R)$ satisfy
\[
                         \mathrm d'\mathrm d''[u]
                         =\mathrm d'\mathrm d''[v].
\]
Then
\[
                         \mathrm d'\mathrm d''(uT)
                         =\mathrm d'\mathrm d''(vT).
\]
In particular, this holds for every continuous psh pair in the
continuous-potential predicate of P21.3.
\end{corollary}

\begin{corollary}
\label{cor:p213-residual-picard-stage-local-annular-tower-cover-bt}
Let $K$ be algebraically closed, complete, and nontrivially valued, let
$X$ be a good, topologically separated, pure, boundaryless strict
$K$-analytic space of dimension $n\geq2$, and let $V\subseteq X$ be
ordinary open.  Suppose that $V$ has an ordinary open cover by sets $B$
which, with their induced analytic structures, satisfy the geometric
hypotheses of
\Cref{cor:p213-residual-picard-stage-local-annular-tower-bt}.  Then every
positive closed ordinary CLD current $T$ of bidegree $(p,p)$ on $V$,
$0\leq p\leq n-1$, has Bedford--Taylor independence for every continuous
psh pair $u,v$ on $V$ with equal Hessian currents.
\end{corollary}

\begin{corollary}
\label{cor:p213-residual-picard-stage-local-annular-tower-basis-bt}
Let $K$ and $X$ be as in
\Cref{cor:p213-residual-picard-stage-local-annular-tower-cover-bt}.
Suppose that the ordinary topology of $X$ has a basis of open sets $B$ such
that every $B$, with its induced analytic structure, satisfies the
geometric hypotheses of
\Cref{cor:p213-residual-picard-stage-local-annular-tower-bt}.  Then the
continuous-pair Bedford--Taylor predicate in P21.3 holds on every ordinary
open subset of $X$ for every positive closed ordinary CLD current.
\end{corollary}
