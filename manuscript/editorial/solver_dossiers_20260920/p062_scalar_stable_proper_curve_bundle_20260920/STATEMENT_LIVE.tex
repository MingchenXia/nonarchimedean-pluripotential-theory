% Book-local P06.2 scalar-stable proper-curve-bundle extension,
% 20 September 2026.  No novelty or priority claim is made.

\begin{proposition}
\label{prop:p062-scalar-stable-proper-curve-bundle-descent}
Let $K$ be a complete non-Archimedean field with nontrivial absolute value,
let $m\geq1$, and let $B$ be a good, topologically separated, locally
holomorphically separated, paracompact, pure $m$-dimensional strict
$K$-analytic space.  Let $D$ be a nonempty geometrically connected smooth
proper algebraic curve over $K$, put $F=D^{\mathrm{an}}$, and let
$\pi:Y\to B$ be a morphism from a good, topologically separated, pure
$(m+1)$-dimensional strict $K$-analytic space.  Suppose that $B$ has an
ordinary analytic open cover by sets $U$ for which there is an isomorphism
over $U$
\[
                         \pi^{-1}(U)\simeq U\times_KF.
\]

Let $h\in C^0(Y,\mathbf R)$ satisfy
$\mathrm d'\mathrm d''[h]=0$.  Assume in addition the following
fibrewise scalar-stability condition.  For every rigid point $x\in B$, let
$h_x$ be the restriction of $h$ to
$Y_x\simeq D_{\mathscr H(x)}^{\mathrm{an}}$.  For one completed algebraic
closure $\Omega_x$ of $\mathscr H(x)$, the scalar extension
$h_{x,\Omega_x}$ satisfies
\begin{equation}
             \mathrm d'\mathrm d''[h_{x,\Omega_x}]=0
             \quad\text{on }D_{\Omega_x}^{\mathrm{an}}.
 \label{eq:p062-scalar-stable-curve-fibre}
\end{equation}
Then there is a unique $b\in C^0(B,\mathbf R)$ such that
\begin{equation}
                         h=\pi^*b,
                         \qquad
                         \mathrm d'\mathrm d''[b]=0.
 \label{eq:p062-scalar-stable-curve-descent}
\end{equation}
Moreover, $\mathrm d'[h]=0$ implies $\mathrm d'[b]=0$.
The proposition does not assert that
\eqref{eq:p062-scalar-stable-curve-fibre} follows automatically from the
downstairs current equality.
\end{proposition}
