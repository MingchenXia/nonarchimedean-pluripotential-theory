\begin{proposition}
\label{prop:p071-double-node-support-plane-reduction}
Let $E=F=\mathbf R^2$ and put
\[
 v_1=w_1=(1,0),\qquad
 v_2=w_2=(0,1),\qquad
 v_3=w_3=(-1,-1).
\]
Let $S$ and $T$ be the corresponding open exact trivalent stars with
outward parametrizations $p_i(r)=rv_i$ and $q_j(s)=sw_j$ for
$0\leq r,s<R$, where $R>0$, and write $o=(\eta_S,\eta_T)$.  For a real-valued
continuous function $u$ on $S\times T$, the following are equivalent.

\begin{enumerate}
\item For every finite family
\[
 z_a=(p_{i_a}(r_a),q_{j_a}(s_a)),\qquad
 \lambda_a>0,\qquad \sum_a\lambda_a=1,
\]
for which there are $M,N\geq0$ satisfying
\begin{align}
 \sum_{a:i_a=i}\lambda_ar_a&=M
       &&(i=1,2,3),\
 \sum_{a:j_a=j}\lambda_as_a&=N
       &&(j=1,2,3),
 \label{eq:p071-support-plane-balanced-data}
\end{align}
one has
\begin{equation}
                       u(o)\leq\sum_a\lambda_au(z_a).
 \label{eq:p071-support-plane-jensen}
\end{equation}

\item The same assertion holds for families containing at most five
atoms.

\item There are linear forms $A\in E^*$ and $B\in F^*$ such that
\begin{equation}
 u(p_i(r),q_j(s))
 \geq u(o)+A(rv_i)+B(sw_j)
 \label{eq:p071-support-plane-minorant}
\end{equation}
for every $i,j\in\{1,2,3\}$ and $0\leq r,s<R$.
\end{enumerate}

The support bound five is sharp for the barycentric reduction.  For every
$0<\rho<R$, the five points
\begin{equation}
 \begin{aligned}
 z_1&=(p_1(\rho),\eta_T),&
 z_2&=(p_2(\rho),\eta_T),&
 z_3&=(p_3(\rho),q_1(\rho)),\\
 z_4&=(\eta_S,q_2(\rho)),&
 z_5&=(\eta_S,q_3(\rho))
 \end{aligned}
 \label{eq:p071-support-plane-sharp-atoms}
\end{equation}
have barycentre $o$ with weights $1/5$, but $o$ is not in the convex hull
of any proper subfamily.  Thus four atoms cannot replace five in a
universal reduction of balanced double-node barycentres.
\end{proposition}
