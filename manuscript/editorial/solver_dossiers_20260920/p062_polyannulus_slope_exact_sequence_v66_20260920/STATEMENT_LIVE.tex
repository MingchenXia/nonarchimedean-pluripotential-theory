% Book-local P06.2 polyannulus slope exact sequence, 20 September 2026.
Retain the setting and notation of
\Cref{prop:p062-polyannulus-bundle-slope-local-system,%
def:p062-zero-current-radiance-class}.  Thus
\[
 \begin{aligned}
 \mathscr P_W(U)
 &=\{b\in C^0(U,\mathbf R):\mathrm d'\mathrm d''[b]=0\},\\
 \rho_0(\pi)
 &=[(c_{\alpha\beta})]
   \in\check H^1\bigl(W,\mathscr P_W(\mathbb V_{\pi,\mathbf R})\bigr).
 \end{aligned}
\]
We again record the transition data used below.  In the frame indexed by
$\alpha$,
\begin{align*}
 L_\beta
 &=\pi^*c_{\alpha\beta}
   +(M_{\alpha\beta}\circ\pi)L_\alpha,\\
 c_{\alpha\gamma}
 &=c_{\beta\gamma}+M_{\beta\gamma}c_{\alpha\beta},
 &
 a_\alpha&=M_{\alpha\beta}^{\mathsf T}a_\beta.
\end{align*}
Every continuous zero-current function has unique chartwise coefficients
\begin{equation}
\label{eq:p062-slope-exact-local-form}
 h=\pi^*b_\alpha+(a_\alpha\circ\pi)^{\mathsf T}L_\alpha,
 \qquad
 b_\alpha=b_\beta+a_\beta^{\mathsf T}c_{\alpha\beta},
\end{equation}
where $b_\alpha\in\mathscr P_W(U_\alpha)$ and $a_\alpha$ is locally
constant.  Conversely, after refinement, coefficients with these
properties define a continuous zero-current function by
\eqref{eq:p062-slope-exact-local-form}.  This local converse is the one for
the displayed curve--polyannulus product charts; it is not asserted for an
arbitrary annular fibration.

After passage to a common refinement, a second set of annular coordinates
is related to the first by
\[
 \widehat L_\alpha
 =\pi^*e_\alpha+(N_\alpha\circ\pi)L_\alpha,
 \qquad
 N_\alpha\in\operatorname{GL}_s(\mathbf Z),
 \qquad
 e_\alpha\in\mathscr P_W(U_\alpha)^s,
\]
with $N_\alpha$ locally constant.  Its transition data satisfy
\[
 \widehat M_{\alpha\beta}
 =N_\beta M_{\alpha\beta}N_\alpha^{-1},
 \qquad
 \widehat c_{\alpha\beta}
 =N_\beta c_{\alpha\beta}
  +e_\beta-\widehat M_{\alpha\beta}e_\alpha.
\]

We also record the vanishing criterion for the radiance class.  If
$\rho_0(\pi)=0$, then, after refinement, there are
$d_\alpha\in\mathscr P_W(U_\alpha)^s$ such that
\begin{equation}
\label{eq:p062-slope-exact-radiance-gauge}
 c_{\alpha\beta}+d_\beta-M_{\alpha\beta}d_\alpha=0.
\end{equation}
The vectors
\[
 \widetilde L_\alpha=L_\alpha+\pi^*d_\alpha
\]
then satisfy
$\widetilde L_\beta=(M_{\alpha\beta}\circ\pi)\widetilde L_\alpha$ and
glue to a continuous section $\Lambda_\pi$ of the associated flat real
vector bundle on $Y$.  For every parallel dual section $a$, the local
pairings
$(a_\alpha\circ\pi)^{\mathsf T}\widetilde L_\alpha$ glue to a continuous
zero-current function on $Y$ with slope $a$; this is the
preceding local converse with base coefficient
$a_\alpha^{\mathsf T}d_\alpha$.

Put
\[
 \mathscr Z(Y)=
 \{h\in C^0(Y,\mathbf R):\mathrm d'\mathrm d''[h]=0\}.
\]

\begin{proposition}
\label{prop:p062-polyannulus-slope-exact-sequence}
The unique local slope vectors define a linear map
\[
 \sigma_\pi:\mathscr Z(Y)
 \longrightarrow H^0(W,\mathbb V_{\pi,\mathbf R}^{\vee}).
\]
For a parallel dual section $a$, with local coordinates $a_\alpha$, the
scalar functions
\[
 r_{\alpha\beta}(a)=a_\beta^{\mathsf T}c_{\alpha\beta}
 \in\mathscr P_W(U_{\alpha\beta})
\]
form a Cech one-cocycle.  Its class depends only on $a$ and the canonical
radiance class, and defines a linear contraction map
\[
 \partial_{\rho_0}:H^0(W,\mathbb V_{\pi,\mathbf R}^{\vee})
 \longrightarrow\check H^1(W,\mathscr P_W),
 \qquad
 a\longmapsto\langle a,\rho_0(\pi)\rangle.
\]
There is a natural exact sequence of real vector spaces
\begin{equation}
\label{eq:p062-polyannulus-slope-exact-sequence}
 0\longrightarrow\mathscr P_W(W)
 \xrightarrow{\ \pi^*\ }\mathscr Z(Y)
 \xrightarrow{\ \sigma_\pi\ }
 H^0(W,\mathbb V_{\pi,\mathbf R}^{\vee})
 \xrightarrow{\ \partial_{\rho_0}\ }
 \check H^1(W,\mathscr P_W).
\end{equation}
Equivalently, a parallel slope $a$ occurs in a global continuous
zero-current function if and only if
$\langle a,\rho_0(\pi)\rangle=0$.

If $\partial_{\rho_0}$ is injective, every continuous zero-current function
on $Y$ is the pullback of a unique element of $\mathscr P_W(W)$.  In
particular, this holds when
$H^0(W,\mathbb V_{\pi,\mathbf R}^{\vee})=0$.

If $\rho_0(\pi)=0$, any zero-current affine gauge satisfying
\eqref{eq:p062-slope-exact-radiance-gauge} gives a splitting of
\eqref{eq:p062-polyannulus-slope-exact-sequence} and hence a
gauge-dependent isomorphism
\[
 \mathscr Z(Y)\simeq
 \mathscr P_W(W)\oplus
 H^0(W,\mathbb V_{\pi,\mathbf R}^{\vee}).
\]
None of these assertions trivializes the integral local system or produces
global analytic annular coordinates.
\end{proposition}
