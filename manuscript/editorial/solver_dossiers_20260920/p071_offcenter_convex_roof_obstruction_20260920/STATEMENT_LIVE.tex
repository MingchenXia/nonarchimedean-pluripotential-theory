% Book-local P07.1 obstruction, 20 September 2026.
% This isolated statement makes no novelty or priority claim.

\begin{proposition}
\label{prop:p071-offcenter-convex-roof-obstruction}
Let $E=F=\mathbf R^2$ and put
\[
 v_1=w_1=(1,0),\qquad
 v_2=w_2=(0,1),\qquad
 v_3=w_3=(-1,-1).
\]
Let $S\subset E$ and $T\subset F$ be the corresponding exact trivalent
stars with outward parametrizations $p_i(r)=rv_i$ and $q_j(s)=sw_j$ on
$[0,R)$, where $R>13/4$.  Consider the local two-star generator class
$\mathcal G_0$ whose members are real-valued, have smooth jointly convex
restrictions $g_{ij}$ to the nine branch rectangles, have common smooth
restrictions on both families of node axes, and satisfy
\begin{align}
 \sum_{i=1}^3\partial_rg_{ij}(0,s)&=0
       &&(j=1,2,3),
 \label{eq:p071-offcenter-first-balance}\\
 \sum_{j=1}^3\partial_sg_{ij}(r,0)&=0
       &&(i=1,2,3).
 \label{eq:p071-offcenter-second-balance}
\end{align}

There is a member of $\mathcal G_0$ which is not convex for ambient
barycentres on $S\times T$.  More precisely, set
\begin{equation}
 g_{ij}(r,s)=\frac12A_ir^2+C_{ij}rs+\frac12B_js^2,
 \label{eq:p071-offcenter-quadratic}
\end{equation}
where
\begin{equation}
 \begin{aligned}
 A&=(625,81,256),& B&=(144,16,256),\\
 C&=\begin{pmatrix}
 84&-100&16\\
 108&36&-144\\
 -192&64&128
 \end{pmatrix}.&&
 \end{aligned}
 \label{eq:p071-offcenter-coefficients}
\end{equation}
The nine functions glue to a member $g\in\mathcal G_0$.

Let
\[
 z_0=(p_2(1/4),q_2(2))
\]
and take the following five atoms and weights:
\begin{equation}
\begin{array}{c|c}
 \lambda_a&z_a\\ \hline
 12/43&(\eta_S,q_2(5/2))\\
 13/43&(\eta_S,q_2(11/4))\\
 7/43 &(p_1(1/4),q_2(13/4))\\
 10/43&(p_2(5/4),q_3(1/4))\\
 1/43 &(p_3(7/4),q_1(5/2)).
\end{array}
\label{eq:p071-offcenter-atoms}
\end{equation}
Under the displayed embeddings in $E\times F$, one has
\begin{equation}
                 z_0=\sum_a\lambda_az_a,
 \label{eq:p071-offcenter-barycentre}
\end{equation}
but
\begin{equation}
 g(z_0)=\frac{1681}{32}
   >\frac{57945}{1376}
   =\sum_a\lambda_ag(z_a),
 \qquad
 g(z_0)-\sum_a\lambda_ag(z_a)=\frac{7169}{688}.
 \label{eq:p071-offcenter-violation}
\end{equation}
Consequently the generated two-star sheaf cannot be characterized by
unrestricted ambient-barycentric Jensen inequalities.  In particular, a
convex-roof extension to the ambient convex hull cannot by itself prove
sufficiency of the mixed double-junction conditions.  This proposition
does not settle that sufficiency question and does not retire P07.1.
\end{proposition}
