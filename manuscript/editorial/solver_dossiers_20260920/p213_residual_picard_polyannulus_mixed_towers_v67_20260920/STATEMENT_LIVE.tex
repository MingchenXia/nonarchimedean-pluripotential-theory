\begin{theorem}
\label{thm:p213-residual-picard-open-polycylinder-mixed-tower}
Let $K,C,W$ be as in
\Cref{prop:p213-residual-picard-smooth-curve}, and assume
\eqref{eq:p213-residual-picard-vanishing}.  Fix integers $s\geq1$ and
$r\geq0$.  Let $\pi_0:Y_0\to W$ be a morphism, where $Y_0$ is a good,
topologically separated, pure $(s+r+1)$-dimensional strict $K$-analytic
space.  Suppose that $W$ has an analytic open cover by sets $U$ for which,
with radii
\[
 \begin{gathered}
  0<r_{U,i}<R_{U,i},\qquad
  r_{U,i},R_{U,i}\in|K^\times|\quad(1\leq i\leq s),\\
  \rho_{U,j}\in|K^\times|\quad(1\leq j\leq r),
 \end{gathered}
\]
there is an isomorphism over $U$
\begin{equation}
\label{eq:p213-residual-picard-open-polycylinder-chart}
 \pi_0^{-1}(U)\simeq
 U\times_K\prod_{i=1}^s
 \{z\in(\mathbf G_{\mathrm m,K})^{\mathrm{an}}:
              r_{U,i}<|T_{U,i}(z)|<R_{U,i}\}
 \times_K\prod_{j=1}^rD^-_{\rho_{U,j}}.
\end{equation}
For $r=0$, the final product is understood to be a point.

Put $B_0=Y_0$.  Let $\ell\geq0$, and, when $\ell\geq1$, suppose that
\[
 B_\ell\xrightarrow{\pi_\ell}B_{\ell-1}
 \longrightarrow\cdots\longrightarrow B_1
 \xrightarrow{\pi_1}B_0
\]
is a finite tower satisfying the stagewise hypotheses of
\Cref{prop:p062-mixed-exact-bundle-tower-transfer}.  Write
$\Pi=\pi_1\circ\cdots\circ\pi_\ell$ when $\ell\geq1$ and
$\Pi=\operatorname{id}_{Y_0}$ when $\ell=0$.

If $h\in C^0(B_\ell,\mathbf R)$ satisfies
$\dprimeddprime[h]=0$, then there is a unique
$g\in C^0(Y_0,\mathbf R)$ such that
\[
                    h=\Pi^*g,
                    \qquad \dprimeddprime[g]=0.
\]
Both $g$ and $h$ are ordinary CLD-smooth and have zero smooth Hessian.
No compatibility of the annular coordinates in
\eqref{eq:p213-residual-picard-open-polycylinder-chart} is required on
overlaps; in particular, no vanishing of the integral monodromy or of the
zero-current radiance class is assumed.
\end{theorem}

\begin{corollary}
\label{cor:p213-residual-picard-open-polycylinder-mixed-tower-bt}
In the setting of
\Cref{thm:p213-residual-picard-open-polycylinder-mixed-tower}, assume that
$B=B_\ell$ is boundaryless of pure dimension $n\geq2$.  Let $T$ be a
positive $\mathrm d'$- and $\mathrm d''$-closed ordinary CLD current of
bidegree $(p,p)$, $0\leq p\leq n-1$, and let
$u,v\in C^0(B,\mathbf R)$ satisfy
\[
                         \dprimeddprime[u]=\dprimeddprime[v].
\]
Then
\[
                         \dprimeddprime(uT)=\dprimeddprime(vT).
\]
In particular, this holds for every continuous psh pair in the
continuous-potential predicate of P21.3.
\end{corollary}

\begin{corollary}
\label{cor:p213-residual-picard-open-polycylinder-mixed-tower-cover-bt}
Let $K$ be algebraically closed, complete, and nontrivially valued, let
$X$ be a good, topologically separated, pure, boundaryless strict
$K$-analytic space of dimension $n\geq2$, and let $V\subseteq X$ be
ordinary open.  Suppose that $V$ has an ordinary open cover by sets $B$
which, with their induced analytic structures, satisfy the geometric
hypotheses of
\Cref{cor:p213-residual-picard-open-polycylinder-mixed-tower-bt}.  Then
every positive closed ordinary CLD current $T$ of bidegree $(p,p)$ on $V$,
$0\leq p\leq n-1$, has Bedford--Taylor independence for every continuous
psh pair $u,v$ on $V$ with equal Hessian currents.
\end{corollary}

\begin{corollary}
\label{cor:p213-residual-picard-open-polycylinder-mixed-tower-basis-bt}
Let $K$ and $X$ be as in
\Cref{cor:p213-residual-picard-open-polycylinder-mixed-tower-cover-bt}.
Suppose that the ordinary topology of $X$ has a basis of open sets $B$ such
that every $B$, with its induced analytic structure, satisfies the
geometric hypotheses of
\Cref{cor:p213-residual-picard-open-polycylinder-mixed-tower-bt}.  Then the
continuous-pair Bedford--Taylor predicate in P21.3 holds on every ordinary
open subset of $X$ for every positive closed ordinary CLD current.
\end{corollary}
