\begin{proof}[of
\Cref{thm:p213-residual-picard-open-polycylinder-mixed-tower}]
If $\ell\geq1$, apply
\Cref{prop:p062-mixed-exact-bundle-tower-transfer} to descend $h$ uniquely
to a continuous function $g$ on $Y_0$ with
\[
                         h=\Pi^*g,
                         \qquad \dprimeddprime[g]=0.
\]
For $\ell=0$, take $g=h$; uniqueness is then tautological.

It remains to prove ordinary smoothness locally on $Y_0$.  Fix a chart
\eqref{eq:p213-residual-picard-open-polycylinder-chart}, and let
\[
 Z_U=U\times_K\prod_{i=1}^s
 \{z\in(\mathbf G_{\mathrm m,K})^{\mathrm{an}}:
              r_{U,i}<|T_{U,i}(z)|<R_{U,i}\}.
\]
The open set $U$ and every standard strict annular or disc factor are good,
topologically separated, pure, and strict.  Their finite products, including
each intermediate base obtained while removing the disc factors, retain
these properties and have positive dimension.
Successive applications of \Cref{prop:p062-open-disc-descent} to the $r$
open-disc factors give a continuous function $\bar g_U$ on $Z_U$ such that
\[
 g|_{\pi_0^{-1}(U)}=\operatorname{pr}_{Z_U}^*\bar g_U,
 \qquad \dprimeddprime[\bar g_U]=0;
\]
when $r=0$, this means simply $\bar g_U=g|_{\pi_0^{-1}(U)}$ under the
displayed chart.

Apply \Cref{thm:p062-open-polyannulus-products-over-curves} to $\bar g_U$.
Writing $p_U:Z_U\to U$ and $q_{U,i}:Z_U\to A_{U,i}$ for the projections,
there are a continuous zero-current function $b_U$ on $U$ and locally
constant functions $a_{U,1},\ldots,a_{U,s}$ on $U$ such that
\begin{equation}
\label{eq:p213-residual-picard-local-polycylinder-formula}
 \bar g_U=p_U^*b_U+
 \sum_{i=1}^s(a_{U,i}\circ p_U)q_{U,i}^*\log|T_{U,i}|.
\end{equation}
The residual-Picard criterion
\Cref{prop:p213-residual-picard-smooth-curve} applies on $U$ and makes
$b_U$ ordinary CLD-smooth with zero smooth Hessian.  After shrinking in
$U$, all $a_{U,i}$ are constant.  Each $T_{U,i}$ is an invertible analytic
function on its annulus, so \Cref{lem:ch15-log-smooth} makes
$\log|T_{U,i}|$ ordinary CLD-smooth with zero smooth Hessian.  Therefore
\eqref{eq:p213-residual-picard-local-polycylinder-formula} makes
$\bar g_U$ ordinary CLD-smooth with zero smooth Hessian.  Pullback along
$\operatorname{pr}_{Z_U}$ has the same properties by
\Cref{prop:cld-form-pullback}.  The displayed charts cover $Y_0$, so $g$
is ordinary CLD-smooth with zero smooth Hessian on $Y_0$.

Finally, functorial pullback of ordinary CLD forms commutes with both
differentials by \Cref{prop:cld-form-pullback}.  Hence
$h=\Pi^*g$ is ordinary CLD-smooth and has zero smooth Hessian.  The proof
was chartwise and imposed no overlap condition on the annular coordinates,
which proves the final scope assertion.
\end{proof}

\begin{proof}[of
\Cref{cor:p213-residual-picard-open-polycylinder-mixed-tower-bt}]
Put $h=u-v$.  The equality of Hessian currents gives
$\dprimeddprime[h]=0$.  The tower theorem makes $h$ ordinary CLD-smooth
with zero smooth Hessian.  Apply \Cref{cor:ch16-smooth-h}.
\end{proof}

\begin{proof}[of
\Cref{cor:p213-residual-picard-open-polycylinder-mixed-tower-cover-bt}]
Put $h=u-v$.  On every member $B$ of the displayed cover, apply
\Cref{thm:p213-residual-picard-open-polycylinder-mixed-tower} to $h|_B$.
Thus $h$ is ordinary CLD-smooth with zero smooth Hessian locally on $V$.
These are sheaf-local properties, so they hold on $V$.  Apply
\Cref{cor:ch16-smooth-h} directly to $T,u,v$ on $V$.
\end{proof}

\begin{proof}[of
\Cref{cor:p213-residual-picard-open-polycylinder-mixed-tower-basis-bt}]
For an ordinary open $V\subseteq X$, the basis members contained in $V$
form a cover to which
\Cref{cor:p213-residual-picard-open-polycylinder-mixed-tower-cover-bt}
applies.
\end{proof}
