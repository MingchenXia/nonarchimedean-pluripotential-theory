\begin{lemma}
\label{lem:p062-open-annulus-unit-integral-slope}
Let $F$ be an algebraically closed complete non-Archimedean field with
nontrivial absolute value, let $0<r<R$ belong to $|F^\times|$, and put
\[
 A=\{z\in(\mathbf G_{\mathrm m,F})^{\mathrm{an}}:r<|T(z)|<R\}.
\]
For every $f\in\mathcal O(A)^\times$ there are unique
$\kappa\in\mathbf R$ and $n\in\mathbf Z$ such that
\begin{equation}
\label{eq:p062-open-annulus-unit-integral-slope}
                 \log|f|=\kappa+n\log|T|
                 \quad\text{on }A.
\end{equation}
\end{lemma}

\begin{proposition}
\label{prop:p062-polyannulus-bundle-slope-local-system}
Let $K$ be algebraically closed, complete, and nontrivially valued, let
$C$ be a smooth proper algebraic curve, and let $W\subset C^{\mathrm{an}}$
be open.  Let $s\geq1$, and let $\pi:Y\to W$ be a morphism, where $Y$ is a
good, topologically separated, pure $(s+1)$-dimensional strict $K$-analytic
space.  Suppose that $W$ has an analytic open cover
$(U_\alpha)_{\alpha\in I}$ for which, for radii
\[
 0<r_{\alpha,i}<R_{\alpha,i},
 \qquad r_{\alpha,i},R_{\alpha,i}\in|K^\times|
 \quad(1\leq i\leq s),
\]
there is an isomorphism over $U_\alpha$
\[
 \pi^{-1}(U_\alpha)\simeq
 U_\alpha\times_K\prod_{i=1}^s
 \{z\in(\mathbf G_{\mathrm m,K})^{\mathrm{an}}:
            r_{\alpha,i}<|T_{\alpha,i}(z)|<R_{\alpha,i}\}.
\]
Put
\[
 L_\alpha=
 \begin{pmatrix}
  \log|T_{\alpha,1}|\\[-2pt]
  \vdots\\[-2pt]
  \log|T_{\alpha,s}|
 \end{pmatrix}
 \quad\text{on }\pi^{-1}(U_\alpha).
\]
On every nonempty overlap $U_{\alpha\beta}=U_\alpha\cap U_\beta$ there
are unique maps
\[
 M_{\alpha\beta}:U_{\alpha\beta}\longrightarrow
      \operatorname{GL}_s(\mathbf Z),
 \qquad
 c_{\alpha\beta}:U_{\alpha\beta}\longrightarrow\mathbf R^s
\]
such that $M_{\alpha\beta}$ is locally constant, every component of
$c_{\alpha\beta}$ is continuous with zero $\mathrm d'\mathrm d''$-current,
and
\begin{equation}
\label{eq:p062-annular-coordinate-transition}
 L_\beta
 =\pi^*c_{\alpha\beta}+
   (M_{\alpha\beta}\circ\pi)L_\alpha
 \quad\text{on }\pi^{-1}(U_{\alpha\beta}).
\end{equation}
On every triple overlap they satisfy
\begin{equation}
\label{eq:p062-annular-affine-cocycle}
 M_{\alpha\gamma}=M_{\beta\gamma}M_{\alpha\beta},
 \qquad
 c_{\alpha\gamma}
 =c_{\beta\gamma}+M_{\beta\gamma}c_{\alpha\beta}.
\end{equation}
Consequently, the matrices define a rank-$s$ local system of lattices
$\mathbb{V}_{\pi,\mathbf Z}$ on $W$, well defined up to canonical
isomorphism under refinement or replacement of the trivializing atlas.
After putting
\[
 \mathbb{V}_{\pi,\mathbf R}
 =\mathbb{V}_{\pi,\mathbf Z}\otimes_{\mathbf Z}\mathbf R,
\]
the pairs $(M_{\alpha\beta},c_{\alpha\beta})$ define an affine
$\mathbb{V}_{\pi,\mathbf R}$-cocycle whose translation terms are
zero-current functions.

If $h\in C^0(Y,\mathbf R)$ satisfies
$\mathrm d'\mathrm d''[h]=0$, its unique chartwise expressions are
\begin{equation}
\label{eq:p062-annular-local-normal-form}
 h=\pi^*b_\alpha+
   (a_\alpha\circ\pi)^{\mathsf T}L_\alpha,
 \qquad
 \mathrm d'\mathrm d''[b_\alpha]=0,
\end{equation}
where $a_\alpha:U_\alpha\to\mathbf R^s$ is locally constant.  On overlaps,
\begin{equation}
\label{eq:p062-annular-coefficient-transition}
 a_\alpha=M_{\alpha\beta}^{\mathsf T}a_\beta,
 \qquad
 b_\alpha=b_\beta+a_\beta^{\mathsf T}c_{\alpha\beta}.
\end{equation}
Thus the slope vectors define a global locally constant section of
$\mathbb{V}_{\pi,\mathbf R}^\vee$.  The failure of the chartwise logarithmic
coefficient data to glue in the chosen coordinates is recorded by the
integral linear monodromy $M_{\alpha\beta}$ and the zero-current translation
cocycle $c_{\alpha\beta}$; no choice of compatible global annular coordinates
is asserted.
\end{proposition}
