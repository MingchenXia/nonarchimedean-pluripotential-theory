\begin{theorem}\label{thm:wanner-curve-comparison}
Let \(K\) be algebraically closed, complete, and nontrivially valued, let
\(C\) be a smooth proper algebraic curve, and let \(W\subset C^{\mathrm{an}}\)
be open.  For a continuous function \(u:W\to\mathbf R\),
\[
                  \dprimeddprime[u]\geq0
        \quad\Longleftrightarrow\quad
                  u\text{ is Thuillier-subharmonic}.
\]
Consequently every continuous CLD-psh function on \(W\) is
Thuillier-subharmonic.
\end{theorem}

\begin{proposition}
\label{prop:ch08-annulus-skeleton}
Let $a\in K^\circ$ with $v(a)=\ell>0$ and let
\[
 A(\ell)=\{z\in\mathbf A^{1,\An}_K:0\leq v(z)\leq\ell\}
\]
be the corresponding closed annulus, the analytic generic fibre of
$\Spf R\{x,y\}/(xy-a)$ with $x=z$ and $y=a/z$.  Then:
\begin{enumerate}
\item the map $s\mapsto\zeta_{0,\rme^{-s}}$ identifies $[0,\ell]$ with a
closed subset $\Gamma_{A}\subset A(\ell)$, and the retraction
$r(z)=\zeta_{0,|z|}$ is a continuous idempotent onto it;
\item for $f$ in the annulus algebra, $|f(z)|\leq|f(r(z))|$, and
$s\mapsto-\log|f(\zeta_{0,\rme^{-s}})|$ is concave and piecewise affine with
integral slopes;
\item $\Gamma_A$ carries the path metric of \cref{def:ch08-path-metric}, in
which it has length $\ell=v(a)$; in particular the edge length is the
valuation of the deformation parameter, and is independent of the model.
\end{enumerate}
\end{proposition}

\begin{lemma}
\label{lem:ch15-log-smooth}
Let $f$ be an invertible analytic function on an open set $U\subset X$.  Then
$\log|f|$ is a smooth function on $U$ in the sense of
\Cref{chap:cld-forms}, and
\[
 \dprime\log|f|=-\dprime t,\qquad
 \ddprime\log|f|=-\ddprime t,\qquad
 \dprimeddprime\log|f|=0 ,
\]
where $t$ is the affine coordinate on $\mathbf R$ pulled back by the moment
map $f$, that is, $t=-\log|f|$.
\end{lemma}

\begin{theorem}
\label{thm:p062-open-polyannulus-products-over-curves}
Let \(K\), \(C\), and \(W\subset C^{\mathrm{an}}\) be as in
\Cref{thm:wanner-curve-comparison}.  Let \(s\geq0\), and for
\(1\leq i\leq s\) put
\[
 A_i=A^-_{r_i,R_i}
 =\{z\in(\mathbf G_{\mathrm m,K})^{\mathrm{an}}:
                  r_i<|T_i(z)|<R_i\},
 \qquad 0<r_i<R_i,\quad r_i,R_i\in|K^\times|.
\]
Write
\[
 X_s=W\times_K\prod_{i=1}^sA_i,
 \qquad p:X_s\longrightarrow W,
 \qquad q_i:X_s\longrightarrow A_i.
\]
If \(h\in C^0(X_s,\mathbf R)\) satisfies
\(\mathrm d'\mathrm d''[h]=0\), then there are unique continuous functions
\(b,a_1,\ldots,a_s:W\to\mathbf R\) such that
\[
 h=p^*b+\sum_{i=1}^s(a_i\circ p)q_i^*\log|T_i|.
\tag{\(\heartsuit\)}\label{eq:p062-open-polyannulus-curve-formula}
\]
Here \(\mathrm d'\mathrm d''[b]=0\), every \(a_i\) is locally constant,
and \(h\) is classically pluriharmonic.  In fact it is locally
\(\mathbf R\)-PL and Gubler--Jell--Rabinoff harmonic.
\end{theorem}
