% Book-local P07.1 corollary, 20 September 2026.
% This records the open-Euclidean restriction already implicit in the
% Stone--exact-star--polyhedral theorem.  It makes no novelty or priority claim.

\begin{corollary}
\label{cor:p071-stone-infinite-star-euclidean-collapse}
Let $Z$ be a locally compact Hausdorff space with a basis of clopen
subsets, let $B\ne\varnothing$, let $X_B$ be the exact
Chambert--Loir--Ducros star of
\Cref{thm:p071-infinite-star-collapse-candidate}, and let
$\Omega\subset\mathbf R^d$ be a nonempty ordinary open set.  Put
\[
                         X=Z\times X_B\times\Omega.
\]
Endow $X$ with the restrictions of the generator, first-stage, full
generated, and Stone-factorized comparison sheaves from
\Cref{thm:p071-stone-infinite-star-polyhedral-collapse} for
$P=\mathbf R^d$.  Then
\begin{equation}
                 \mathcal P_{X,1}=\mathcal P_X=\mathcal C_X.
 \label{eq:p071-stone-infinite-euclidean-collapse}
\end{equation}
Every germ has a pointwise decreasing approximation by generators on one
fixed smaller product neighbourhood.  At a node, that neighbourhood may be
chosen as $C\times U'\times V'$, where $C\subset Z$ is compact clopen,
$U'\subset X_B$ is standard, and $V'\Subset\Omega$ is open and convex.
The comparison condition retains one countable set of active full branches
common to the compact Stone family and to the Euclidean variable.

In particular, the conclusion holds when $\Omega$ is open and convex, with
the joint branch--Euclidean generator and comparison germs obtained by
specializing the polyhedral definitions to $P=\mathbf R^d$.  Thus a
possibly nonmetrizable locally constant Stone parameter, an arbitrarily
large exact CLD branch set, and an active open Euclidean factor may occur
simultaneously.  This does not allow nonconstant affine functions on the
Stone factor, non-CLD or nonambient infinite stars, separately convex
generators, or nonconvex or branched chamber gluings, and therefore does not
retire P07.1.
\end{corollary}
