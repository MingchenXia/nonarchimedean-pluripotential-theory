% Book-local P06.2 zero-current radiance obstruction, 20 September 2026.
Retain the setting and notation of
\Cref{prop:p062-polyannulus-bundle-slope-local-system}, and let
$\mathscr P_W$ be the sheaf
\[
 \mathscr P_W(U)=
 \{b\in C^0(U,\mathbf R):\mathrm d'\mathrm d''[b]=0\}.
\]
We record explicitly the conclusions of that proposition used below.  As
standing data, the trivializing atlas carries locally constant maps
$M_{\alpha\beta}:U_{\alpha\beta}\to\operatorname{GL}_s(\mathbf Z)$,
vectors $c_{\alpha\beta}\in\mathscr P_W(U_{\alpha\beta})^s$, and logarithmic
vectors $L_\alpha$ such that
\[
 L_\beta=\pi^*c_{\alpha\beta}
 +(M_{\alpha\beta}\circ\pi)L_\alpha,
 \qquad
 M_{\alpha\gamma}=M_{\beta\gamma}M_{\alpha\beta},
 \qquad
 c_{\alpha\gamma}
 =c_{\beta\gamma}+M_{\beta\gamma}c_{\alpha\beta}.
\]
Here $c_{\alpha\alpha}=0$ and
$c_{\beta\alpha}=-M_{\beta\alpha}c_{\alpha\beta}$.  After passage to a
common refinement, a second atlas is related to the first by
\[
 \widehat L_\alpha
 =\pi^*e_\alpha+(N_\alpha\circ\pi)L_\alpha,
 \qquad
 N_\alpha\in\operatorname{GL}_s(\mathbf Z),
 \qquad
 e_\alpha\in\mathscr P_W(U_\alpha)^s,
\]
with $N_\alpha$ locally constant.  Finally, every continuous zero-current
function has unique chartwise coefficients
\[
 h=\pi^*b_\alpha+(a_\alpha\circ\pi)^{\mathsf T}L_\alpha,
 \qquad
 a_\alpha=M_{\alpha\beta}^{\mathsf T}a_\beta,
 \qquad
 b_\alpha=b_\beta+a_\beta^{\mathsf T}c_{\alpha\beta},
\]
where $b_\alpha\in\mathscr P_W(U_\alpha)$ and $a_\alpha$ is locally constant.
Let $\mathbb V_{\pi,\mathbf R}$ be the real local system with transition
matrices $M_{\alpha\beta}$.

\begin{definition}
\label{def:p062-zero-current-radiance-class}
Write $\mathscr P_W(\mathbb V_{\pi,\mathbf R})$ for the twist of
$\mathscr P_W$ by this local system: in the frame indexed by $\alpha$, a
section is a vector of functions in $\mathscr P_W$, and its coordinates
transform by $d_\beta=M_{\alpha\beta}d_\alpha$.

The translation vectors $c_{\alpha\beta}$ define a Cech class
\begin{equation}
\label{eq:p062-zero-current-radiance-class}
 \rho_0(\pi)=[(c_{\alpha\beta})]
 \in
 \check H^1\bigl(W,\mathscr P_W(\mathbb V_{\pi,\mathbf R})\bigr),
\end{equation}
where the right-hand side is the direct limit over refinements of analytic
open covers.  We call $\rho_0(\pi)$ the zero-current radiance class of the
polyannulus bundle.
\end{definition}

\begin{proposition}
\label{prop:p062-polyannulus-radiance-obstruction}
The class $\rho_0(\pi)$ is independent of the chosen trivializing atlas and
annular coordinates under the canonical identification of the resulting
local systems.  It vanishes if and only if, after possibly refining the
atlas, there are vectors
\[
 d_\alpha\in\mathscr P_W(U_\alpha)^s
\]
for which the zero-current affine gauges
\begin{equation}
\label{eq:p062-radiance-gauged-logarithms}
 \widetilde L_\alpha=L_\alpha+\pi^*d_\alpha
\end{equation}
satisfy the purely linear transition law
\begin{equation}
\label{eq:p062-radiance-linear-transition}
 \widetilde L_\beta
 =(M_{\alpha\beta}\circ\pi)\widetilde L_\alpha.
\end{equation}
Let $\mathscr C_Y^0$ denote the sheaf of continuous real-valued functions on
$Y$.  When these equivalent conditions hold, the $\widetilde L_\alpha$ glue
to a global continuous section
\[
 \Lambda_\pi\in H^0(Y,\mathcal E_\pi),
 \qquad
 \mathcal E_\pi
 =\mathscr C_Y^0\otimes_{\mathbf R}\pi^{-1}\mathbb V_{\pi,\mathbf R},
\]
of the flat real vector bundle associated with the pulled-back local system.

For any such choice of gauges and every $h\in C^0(Y,\mathbf R)$ with
$\mathrm d'\mathrm d''[h]=0$, let
$a\in H^0(W,\mathbb V_{\pi,\mathbf R}^\vee)$ be the parallel dual section
given by the standing coefficient laws.  Then there
is a unique $b\in\mathscr P_W(W)$ such that
\begin{equation}
\label{eq:p062-radiance-global-coefficient-formula}
 h=\pi^*b+\langle\pi^{-1}a,\Lambda_\pi\rangle.
\end{equation}
In particular, vanishing of
$\check H^1(W,\mathscr P_W(\mathbb V_{\pi,\mathbf R}))$ removes the entire
zero-current translation obstruction.  This conclusion concerns affine
gauges of logarithmic absolute values; it does not assert that the
$d_\alpha$ are logarithms of analytic units or that the bundle admits global
annular coordinates.
\end{proposition}
