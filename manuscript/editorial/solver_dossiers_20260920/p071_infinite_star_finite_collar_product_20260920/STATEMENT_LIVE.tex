% Book-local P07.1 theorem, 20 September 2026.
% This theorem combines the exact infinite CLD star with the finite
% constant-collar endpoint of CLD Example 6.7.2.  It makes no novelty or
% priority claim.

\begin{theorem}
\label{thm:p071-infinite-star-finite-collar-product-collapse}
Let $B\ne\varnothing$, let $X_B$ be the exact
Chambert--Loir--Ducros star of
\Cref{thm:p071-infinite-star-collapse-candidate}, and let
\[
                  S_+=[0,R_0),\qquad 0<R_0\leq+\infty,
\]
carry the finite constant-collar affine structure of
\Cref{thm:p071-cld-constant-collar-endpoint}.  Put
$X=X_B\times S_+$.

Define the exact product generator sheaf $\mathcal G_{+,0}$ as follows.
On $X_B\times(0,R_0)$ it is the one-dimensional Euclidean product
generator sheaf of
\Cref{thm:p071-infinite-star-euclidean-collapse}.  On a product
neighbourhood meeting $X_B\times\{0\}$, require in addition that a
generator be independent of the $S_+$-coordinate on some initial collar,
locally uniformly in the $X_B$-variable.  Thus the finite-activity and
balance conditions at the star node, and the terminal-collar condition at
every full star branch endpoint, are exactly those of the cited theorem.
Let $\mathcal P_{+,1}$ be the sheaf of local pointwise decreasing limits
of sequences in $\mathcal G_{+,0}$, and let $\mathcal P_+$ be the smallest
subsheaf of $\mathbf R\cup\{-\infty\}$-valued functions which contains
$\mathcal G_{+,0}$ and is closed under such limits.

Define a comparison sheaf $\mathcal J_+$ by the following local
conditions.  On an ordinary branch interval times a half-interval in
$S_+$, every branch function, with its boundary values inserted, is
extended-real jointly convex.  At a full star branch endpoint, require the
exact terminal trace of
\Cref{thm:p071-infinite-star-euclidean-collapse}.  In the $S_+$-direction
require the fibrewise boundary-minimum condition
\begin{equation}
                    u(x,0)\leq u(x,s)
       \qquad(x\in X_B,\ 0\leq s<R_0)
\label{eq:p071-star-finite-collar-boundary-minimum}
\end{equation}
whenever the two points lie in the same displayed product chart.

Near a point $(\eta,0)$ at the star node, choose a standard star
neighbourhood $U$, with finite exceptional set $A\subset B$ and
exceptional collars $(-R_a,0]$, and a half-interval $[0,r)\subset S_+$.
Write
\[
             u_b(t,s)=u(p_b(t),s),\qquad c(s)=u(\eta,s).
\]
Require one countable set $K\subset B\setminus A$ such that
\begin{equation}
 u_b(t,s)=c(s)
 \quad
 \bigl(b\in B\setminus(A\cup K),\ t\in[-\infty,0],\
       0\leq s<r\bigr).
\label{eq:p071-star-finite-collar-countable-activity}
\end{equation}
For a finite set $F\subset B\setminus A$, numbers $r_b>0$ for $b\in F$
and $0<\rho_a<R_a$ for $a\in A$, put
\[
          \Lambda=\sum_{b\in F}\frac1{r_b}
                    +\sum_{a\in A}\frac1{\rho_a}.
\]
Whenever $\Lambda>0$, $c(s_0)\in\mathbf R$, and
$s_0,s_b,s_a\in[0,r)$ satisfy
\begin{equation}
 \Lambda s_0=
       \sum_{b\in F}\frac{s_b}{r_b}
        +\sum_{a\in A}\frac{s_a}{\rho_a},
\label{eq:p071-star-finite-collar-barycentre}
\end{equation}
require
\begin{equation}
 \Lambda c(s_0)\leq
       \sum_{b\in F}\frac{u_b(-r_b,s_b)}{r_b}
        +\sum_{a\in A}\frac{u_a(-\rho_a,s_a)}{\rho_a}.
\label{eq:p071-star-finite-collar-jensen}
\end{equation}
Away from $s=0$, these are exactly the comparison conditions of
\Cref{thm:p071-infinite-star-euclidean-collapse}; away from the star node,
the node conditions are omitted.

Then $\mathcal J_+$ is a sheaf and
\begin{equation}
                  \mathcal P_{+,1}=\mathcal P_+=\mathcal J_+.
\label{eq:p071-star-finite-collar-collapse}
\end{equation}
Every germ in these sheaves has a pointwise decreasing approximation by
members of $\mathcal G_{+,0}$ on one fixed smaller product neighbourhood.
In particular, every approximant has an exact initial constant collar,
although the collar widths may tend to zero.

Thus an arbitrary exact CLD branch set and the nonambient finite endpoint
encoded by \Cref{thm:p071-cld-constant-collar-endpoint} may occur
simultaneously, with mixed
branch--interval Hessian terms on the finite stratum.  The theorem does not
cover the constant-collar endpoint at $-\infty$, products of two branched
stars, non-CLD infinite stars, nonambient branch structures, or nonconstant
affine directions on an additional Stone factor.  Consequently it narrows
but does not retire P07.1.
\end{theorem}
