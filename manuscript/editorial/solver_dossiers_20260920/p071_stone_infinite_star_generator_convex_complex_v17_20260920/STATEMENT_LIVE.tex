% Book-local P07.1 corollary, 20 September 2026.
% This transports the Stone--exact-star--polyhedral collapse through local
% generator-preserving convex developments.  It makes no novelty or priority
% claim.

\begin{corollary}
\label{cor:p071-stone-infinite-star-generator-convex-collapse}
Let $\Sigma$ be a possibly infinite Hausdorff face-to-face polyhedral
complex, not assumed locally finite, equipped with a sheaf
$\mathcal G_{\Sigma,0}$ of real-valued smooth psh generators.  Suppose that
every $p\in|\Sigma|$ has an open neighbourhood $U$, a finite-dimensional
real affine space $A$, a nonempty convex polyhedron $P\subset A$, a
relatively open convex subset $V\subset P$, and a homeomorphism
\[
                          \varphi:U\xrightarrow{\sim}V
\]
which carries the complete germ of $\mathcal G_{\Sigma,0}$ in both
directions to the single-chamber smooth psh generator sheaf on $V$.

Let $Z$ be a locally compact Hausdorff space with a basis of clopen subsets,
let $B\ne\varnothing$, and let $X_B$ be the exact
Chambert--Loir--Ducros star of
\Cref{thm:p071-infinite-star-collapse-candidate}.  Put
\[
                         X=Z\times X_B\times|\Sigma|
\]
and equip $X$ with the corresponding product generator sheaf
$\mathcal G_{X,0}$.  Let $\mathcal P_{X,1}$ be its first local decreasing
sequential stage and let $\mathcal P_X$ be its full generated psh sheaf.
Then
\begin{equation}
                         \mathcal P_{X,1}=\mathcal P_X.
 \label{eq:p071-stone-infinite-star-generator-convex-collapse}
\end{equation}
Every germ has a pointwise decreasing approximation by members of
$\mathcal G_{X,0}$ on one fixed smaller product neighbourhood contained in
one displayed convex-development chart.  At a node of $X_B$, one countable
set of active full branches works simultaneously for the compact Stone
family and for every point of the smaller developed polyhedral
neighbourhood.

If $\mathcal C_X$ is defined chartwise by transporting the restriction of
the Stone-factorized comparison sheaf in
\Cref{thm:p071-stone-infinite-star-polyhedral-collapse}, then this
definition is independent of the chosen convex-development chart and
\[
                         \mathcal P_{X,1}=\mathcal P_X=\mathcal C_X.
\]
Thus arbitrary incidence, infinitely many cells through every
neighbourhood, and global chamber monodromy may coexist with a possibly
nonmetrizable locally constant Stone parameter and an arbitrarily large
exact CLD star.  The result does not cover a germ without a
generator-preserving convex development, in particular a branched or
singular lower link, nor does it cover nonconstant affine directions on the
Stone factor, non-CLD or nonambient stars, or separately convex generators.
Consequently it narrows but does not retire P07.1.
\end{corollary}
