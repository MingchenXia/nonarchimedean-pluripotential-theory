\begin{proposition}
\label{prop:p062-polyannulus-bundle-slope-local-system}
Let $K$, $C$, $W$, $\pi:Y\to W$, and $s\geq1$ be as in
\Cref{cor:p062-open-polyannulus-bundles-over-curves}.  Fix a trivializing
cover $(U_\alpha)_{\alpha\in I}$ as in that corollary, choose annular
coordinates $T_{\alpha,1},\ldots,T_{\alpha,s}$ in every chart, and put
\[
 L_\alpha=
 \begin{pmatrix}
  \log|T_{\alpha,1}|\\[-2pt]
  \vdots\\[-2pt]
  \log|T_{\alpha,s}|
 \end{pmatrix}
 \quad\text{on }\pi^{-1}(U_\alpha).
\]
On every nonempty overlap $U_{\alpha\beta}=U_\alpha\cap U_\beta$ there
are unique maps
\[
 M_{\alpha\beta}:U_{\alpha\beta}\longrightarrow
      \operatorname{GL}_s(\mathbf R),
 \qquad
 c_{\alpha\beta}:U_{\alpha\beta}\longrightarrow\mathbf R^s
\]
such that $M_{\alpha\beta}$ is locally constant, every component of
$c_{\alpha\beta}$ is continuous with zero $\mathrm d'\mathrm d''$-current,
and
\begin{equation}
\label{eq:p062-annular-coordinate-transition}
 L_\beta
 =\pi^*c_{\alpha\beta}+
   (M_{\alpha\beta}\circ\pi)L_\alpha
 \quad\text{on }\pi^{-1}(U_{\alpha\beta}).
\end{equation}
On every triple overlap they satisfy
\begin{equation}
\label{eq:p062-annular-affine-cocycle}
 M_{\alpha\gamma}=M_{\beta\gamma}M_{\alpha\beta},
 \qquad
 c_{\alpha\gamma}
 =c_{\beta\gamma}+M_{\beta\gamma}c_{\alpha\beta}.
\end{equation}
Consequently, the matrices define a rank-$s$ real local system
$\mathbb{V}_\pi$ on $W$, well defined up to canonical isomorphism under
refinement or replacement of the trivializing atlas, and the pairs
$(M_{\alpha\beta},c_{\alpha\beta})$ define an affine
$\mathbb{V}_\pi$-cocycle whose translation terms are zero-current functions.

If $h\in C^0(Y,\mathbf R)$ satisfies
$\mathrm d'\mathrm d''[h]=0$, its unique chartwise expressions are
\begin{equation}
\label{eq:p062-annular-local-normal-form}
 h=\pi^*b_\alpha+
   (a_\alpha\circ\pi)^{\mathsf T}L_\alpha,
 \qquad
 \mathrm d'\mathrm d''[b_\alpha]=0,
\end{equation}
where $a_\alpha:U_\alpha\to\mathbf R^s$ is locally constant.  On overlaps,
\begin{equation}
\label{eq:p062-annular-coefficient-transition}
 a_\alpha=M_{\alpha\beta}^{\mathsf T}a_\beta,
 \qquad
 b_\alpha=b_\beta+a_\beta^{\mathsf T}c_{\alpha\beta}.
\end{equation}
Thus the slope vectors define a global locally constant section of the dual
local system $\mathbb{V}_\pi^\vee$.  The failure of the chartwise logarithmic
coefficient data to glue in the chosen coordinates is recorded by the linear
monodromy $M_{\alpha\beta}$ and the zero-current translation cocycle
$c_{\alpha\beta}$; no choice of compatible global annular coordinates and no
integrality of the matrices are asserted.
\end{proposition}
