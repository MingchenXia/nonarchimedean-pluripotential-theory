% Book-local P06.2 smooth proper split-toric bundle descent, 20 September 2026.
% The proof uses only the regular affine toric atlas and strict-disc descent.
\begin{proposition}
\label{prop:p062-smooth-proper-split-toric-bundle-descent}
Let $K$ be a complete non-Archimedean field with nontrivial absolute
value, let $m,q\geq1$, and let $B$ be a good, topologically separated,
locally holomorphically separated, paracompact, pure $m$-dimensional strict
$K$-analytic space.  Let $X_\Sigma$ be a smooth proper split toric
$K$-variety of dimension $q$.  Let $\pi:Y\to B$ be a morphism, where
$Y$ is a good, topologically separated, pure $(m+q)$-dimensional strict
$K$-analytic space.  Suppose that $B$ has an analytic open cover by
$U$'s for which there is an isomorphism over $U$
\[
                  \pi^{-1}(U)\simeq U\times_KX_\Sigma^{\mathrm{an}}.
\]
If $h\in C^0(Y,\mathbf R)$ satisfies
$\mathrm d'\mathrm d''[h]=0$, then there is a unique
$b\in C^0(B,\mathbf R)$ such that
\[
                         h=\pi^*b,
                         \qquad
                         \mathrm d'\mathrm d''[b]=0.
\]
Moreover, $\mathrm d'[h]=0$ implies $\mathrm d'[b]=0$.
\end{proposition}
