\begin{remark}
\label{rem:ch08-harmonic-vs-log}
Write $\mathcal F_C$ for the sheaf of $\mathbf R$-vector spaces generated
by germs $\log|f|$, with $f$ invertible analytic, and $\mathcal H_C$ for
Thuillier's sheaf of harmonic germs.  Then
$\mathcal F_C\subseteq\mathcal H_C$, the quotient is supported at the
type~$2$ points, and at such a point $x$ with associated residue curve
$C_x$ one has
\[
 (\mathcal H_C/\mathcal F_C)_x
 \simeq\operatorname{Pic}^0(C_x)\otimes_{\mathbf Z}\mathbf R.
\]
\end{remark}

\begin{theorem}
\label{thm:wanner-curve-comparison}
Let $K$ be algebraically closed, complete, and nontrivially valued, let
$C$ be a smooth proper algebraic curve, and let $W\subset C^{\mathrm{an}}$
be open.  For a continuous function $u:W\to\mathbf R$,
\[
                  \mathrm d'\mathrm d''[u]\geq0
        \quad\Longleftrightarrow\quad
                  u\text{ is Thuillier-subharmonic}.
\]
Consequently every continuous CLD-psh function on $W$ is
Thuillier-subharmonic.
\end{theorem}

% Source: manuscript/parts/part2/ch19_cld_examples.tex.
\begin{corollary}
\label{cor:ch19-continuous-zero-current-harmonic}
Under the hypotheses of \Cref{thm:wanner-curve-comparison}, let
$h:W\to\mathbf R$ be continuous.  If
\[
                         \mathrm d'\mathrm d''[h]=0,
\]
then $h$ is Thuillier harmonic, hence harmonic in the sense of
Gubler--Jell--Rabinoff and classically pluriharmonic in the sense of
Gubler--Rabinoff.
\end{corollary}

\begin{proposition}
\label{prop:p213-residual-picard-smooth-curve}
Let $K$ be algebraically closed, complete, and nontrivially valued, let
$C$ be a smooth proper algebraic $K$-curve, and let
$W\subset C^{\mathrm{an}}$ be open.  Suppose that
$h\in C^0(W,\mathbf R)$ satisfies
\[
                         \mathrm d'\mathrm d''[h]=0.
\]
For a type~$2$ point $x\in W$, let $C_x$ be its smooth projective residue
curve and let
\[
 \operatorname{ob}_x(h)\in
 \operatorname{Pic}^0(C_x)\otimes_{\mathbf Z}\mathbf R
\]
be the residual class of the harmonic germ of $h$ at $x$.  Then the
following are equivalent:
\begin{enumerate}
\item $h$ is ordinary CLD-smooth near $x$ and
      $\mathrm d'\mathrm d''h=0$ there as a smooth form;
\item $\operatorname{ob}_x(h)=0$.
\end{enumerate}
At points of types $1$, $3$, and $4$, the germ of $h$ is automatically
ordinary CLD-smooth with zero smooth Hessian.
\end{proposition}

\begin{proposition}
\label{prop:classical-psh-calculus}
Classically psh functions form a sheaf and a convex cone, are stable under
finite maxima and decreasing nets, scalar extension, and analytic pullback.
If $u$ attains a local maximum at an interior point, then $u$ is locally
constant there.  A real-valued $h$ is pluriharmonic in this theory when
both $h$ and $-h$ are classically psh.
\end{proposition}

\begin{proposition}
\label{prop:p062-proper-smooth-curve-bundle-descent}
Let $K$ be an algebraically closed, complete non-Archimedean field with
nontrivial absolute value, let $m\geq1$, and let $B$ be a good,
topologically separated, locally holomorphically separated, paracompact,
pure $m$-dimensional strict $K$-analytic space.  Let $D$ be a nonempty
geometrically connected smooth proper algebraic curve over $K$, and put
$F:=D^{\mathrm{an}}$.  Let $\pi:Y\to B$ be a morphism, where $Y$ is a
good, topologically separated, pure $(m+1)$-dimensional strict
$K$-analytic space.  Suppose that $B$ has an ordinary analytic open cover
by $U$'s for which there is an isomorphism over $U$
\[
                         \pi^{-1}(U)\simeq U\times_KF.
\]
If $h\in C^0(Y,\mathbf R)$ satisfies
$\mathrm d'\mathrm d''[h]=0$, then there is a unique
$b\in C^0(B,\mathbf R)$ such that
\[
                         h=\pi^*b,
                         \qquad
                         \mathrm d'\mathrm d''[b]=0.
\]
Moreover, $\mathrm d'[h]=0$ implies $\mathrm d'[b]=0$.
\end{proposition}

% Source: Gubler--Jell--Rabinoff, Forms on Berkovich spaces based on
% harmonic tropicalizations, Corollary 7.15 in the published version.
\begin{corollary}
\label{cor:p213-gjr-harmonic-zero-current}
Let $Y$ be a good separated strict analytic space and let $h$ be a
Gubler--Jell--Rabinoff harmonic function on $Y$.  Then the restriction of
the current $\mathrm d'\mathrm d''[h]$ to $Y\setminus\partial Y$ is zero.
In particular, if $Y$ is boundaryless, then
$\mathrm d'\mathrm d''[h]=0$ on $Y$.
\end{corollary}

\begin{proposition}
\label{prop:ch18-harmonic-functorial}
Let $\varphi:Y\to X$ be a morphism of good strict $K$-analytic spaces and
let $h$ be an $\mathbf R$-harmonic function on an open $U\subseteq X$.
Then $\varphi^*h=h\circ\varphi$ is $\mathbf R$-harmonic on
$\varphi^{-1}(U)$.
\end{proposition}

% Source: manuscript/parts/part2/ch13_forms.tex.
\begin{proposition}
\label{prop:cld-form-pullback}
Let $g:Y\to X$ be a morphism of topologically separated analytic spaces.
There is a functorial morphism
\[
                 g^*:A_X^{p,q}\longrightarrow g_*A_Y^{p,q}
\]
commuting with wedge, $J$, $\mathrm d'$, and $\mathrm d''$.
\end{proposition}
