\begin{proposition}
\label{prop:p213-zero-current-smoothness-not-open-hereditary}
Let $K$ be algebraically closed, complete, and nontrivially valued, and let
$D$ be a geometrically connected smooth proper algebraic $K$-curve.  Suppose
that $x\in D^{\mathrm{an}}$ is a type~$2$ point such that
\begin{equation}
\label{eq:p213-nonhereditary-residual-class}
 \operatorname{Pic}^0(D_x)\otimes_{\mathbf Z}\mathbf R\ne0.
\end{equation}
Put
\[
 P=(\mathbf P^1_K)^{\mathrm{an}},
 \qquad X=P\times_KD^{\mathrm{an}}.
\]
Then the following assertions hold.
\begin{enumerate}
\item Every $H\in C^0(X,\mathbf R)$ satisfying
$\mathrm d'\mathrm d''[H]=0$ is constant.  In particular, every such
function is classically pluriharmonic and ordinary CLD-smooth with zero
smooth Hessian, and adding $\mathrm d'[H]=0$ forces local constancy.
\item There are an ordinary open neighbourhood $U\subset D^{\mathrm{an}}$
of $x$ and a function $g\in C^0(U,\mathbf R)$ such that
\[
                     \mathrm d'\mathrm d''[g]=0
\]
but $g$ is not ordinary CLD-smooth near $x$.  For
\[
 \Omega=P\times_KU,
 \qquad G=\operatorname{pr}_U^*g,
\]
one has $\mathrm d'\mathrm d''[G]=0$, whereas $G$ is not ordinary
CLD-smooth near $(p,x)$ for any $p\in P(K)$.
\end{enumerate}
Consequently, whole-space ordinary smooth zero-current regularity is not
inherited by arbitrary ordinary open subsets, even on a smooth proper
surface which is a trivial proper smooth curve bundle over a
residual-Picard base.  Thus the stage-local chart hypotheses in
\Cref{thm:p213-residual-picard-stage-local-annular-tower} cannot be removed
by invoking only the corresponding whole-space conclusions on an exact
intermediate base.
\end{proposition}

\begin{corollary}
\label{cor:p213-good-reduction-elliptic-open-nonheredity}
The situation in
\Cref{prop:p213-zero-current-smoothness-not-open-hereditary} occurs.  One
may take
\[
                K=\widehat{\overline{\mathbf C((t))}}
\]
with its $t$-adic absolute value and let $D$ be the base change to $K$ of
an elliptic curve $E$ over $\mathbf C$.  At the type~$2$ point $x$
defined by the good-reduction component, one has $D_x\simeq E$ and
\[
 \operatorname{Pic}^0(D_x)\otimes_{\mathbf Z}\mathbf R\ne0.
\]
\end{corollary}
