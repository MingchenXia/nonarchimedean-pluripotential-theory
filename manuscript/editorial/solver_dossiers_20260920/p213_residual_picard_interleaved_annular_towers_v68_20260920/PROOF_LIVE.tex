\begin{proof}[of
\Cref{thm:p213-residual-picard-curve-local-interleaved-tower}]
Fix $U\in\mathcal U$, and write $\Pi_U:B_{\ell,U}\to U$ for the
restricted composite.  Locality of currents gives
\[
                 \dprimeddprime[h|_{B_{\ell,U}}]=0.
\]
Apply
\Cref{thm:p062-interleaved-polyannulus-mixed-bundle-towers} to the
base-changed tower.  In the notation of that theorem, there are a continuous
function $b_U:U\to\mathbf R$ and locally constant functions
$a_{U,j,k}:U\to\mathbf R$ such that
\begin{equation}
\label{eq:p213-residual-picard-interleaved-local-formula}
 h|_{B_{\ell,U}}
 =\Pi_U^*b_U+
   \sum_{j\in J}\sum_{k=1}^{s_j}
      (a_{U,j,k}\circ\Pi_U)L_{U,j,k},
 \qquad \dprimeddprime[b_U]=0.
\end{equation}
When $J$ is empty, the double sum is understood to be zero.

The residual-Picard criterion
\Cref{prop:p213-residual-picard-smooth-curve} makes $b_U$ ordinary
CLD-smooth with zero smooth Hessian.  Hence the same is true of
$\Pi_U^*b_U$ by \Cref{prop:cld-form-pullback}.  Each $L_{U,j,k}$ is an
iterated pullback of $\log|T_{U,j,k}|$ from a standard strict open annulus.
The annular coordinate is invertible, so
\Cref{lem:ch15-log-smooth,prop:cld-form-pullback} makes $L_{U,j,k}$
ordinary CLD-smooth with zero smooth Hessian.  Since $a_{U,j,k}$ is locally
constant, its pullback is locally constant; thus every summand in
\eqref{eq:p213-residual-picard-interleaved-local-formula} is ordinary
CLD-smooth with zero smooth Hessian.  The same holds for the displayed
finite sum.

The sets $B_{\ell,U}$ cover $B_\ell$.  Ordinary CLD smoothness and the
vanishing of the smooth Hessian are local, so they hold for $h$ on
$B_\ell$.  The argument uses the annular choices only on each
$B_{\ell,U}$ and therefore imposes no compatibility on overlaps.  Without
such compatibility the local formulas need not define a global coefficient
formula, which proves the final scope assertion.
\end{proof}

\begin{proof}[of
\Cref{cor:p213-residual-picard-curve-local-interleaved-tower-bt}]
Put $h=u-v$.  Equality of the Hessian currents gives
$\dprimeddprime[h]=0$.  The tower theorem makes $h$ ordinary CLD-smooth
with zero smooth Hessian.  Apply \Cref{cor:ch16-smooth-h}.
\end{proof}

\begin{proof}[of
\Cref{cor:p213-residual-picard-curve-local-interleaved-tower-cover-bt}]
Put $h=u-v$.  On every member $B$ of the displayed cover, apply
\Cref{thm:p213-residual-picard-curve-local-interleaved-tower} to $h|_B$.
Thus $h$ is ordinary CLD-smooth with zero smooth Hessian locally on $V$.
These are sheaf-local properties, so they hold on $V$.  Apply
\Cref{cor:ch16-smooth-h} directly to $T,u,v$ on $V$.
\end{proof}

\begin{proof}[of
\Cref{cor:p213-residual-picard-curve-local-interleaved-tower-basis-bt}]
For an ordinary open $V\subseteq X$, the basis members contained in $V$
form a cover to which
\Cref{cor:p213-residual-picard-curve-local-interleaved-tower-cover-bt}
applies.
\end{proof}
