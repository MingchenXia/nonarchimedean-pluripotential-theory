\begin{theorem}
\label{thm:p213-residual-picard-curve-local-interleaved-tower}
Let $K,C,W$ be as in
\Cref{prop:p213-residual-picard-smooth-curve}, and assume
\eqref{eq:p213-residual-picard-vanishing}.  Let $\ell\geq1$, and suppose
that
\[
 B_\ell\xrightarrow{\pi_\ell}B_{\ell-1}
 \longrightarrow\cdots\longrightarrow B_1
 \xrightarrow{\pi_1}B_0=W
\]
is a finite tower of $K$-analytic spaces.  For $0\leq j\leq\ell$, put
$\Pi_j=\pi_1\circ\cdots\circ\pi_j$, with $\Pi_0$ the identity, and set
$B_{j,U}=\Pi_j^{-1}(U)$ for $U\subseteq W$.

Suppose that $W$ has an analytic open cover $\mathcal U$ such that, for
every $U\in\mathcal U$, the base-changed tower
\[
 B_{\ell,U}\longrightarrow B_{\ell-1,U}
 \longrightarrow\cdots\longrightarrow B_{1,U}
 \longrightarrow B_{0,U}=U
\]
satisfies all hypotheses of
\Cref{thm:p062-interleaved-polyannulus-mixed-bundle-towers}, with $U$ in
place of $W$.  Require the set of annular stages and the number of annular
factors at each such stage to be independent of $U$, but allow their
standard strict annuli, coordinates, and product trivializations to depend
on $U$.  No compatibility of those choices on overlaps is assumed.

If $h\in C^0(B_\ell,\mathbf R)$ satisfies
$\dprimeddprime[h]=0$, then $h$ is ordinary CLD-smooth and
$\dprimeddprime h=0$ as a smooth form.  No global logarithmic coefficient
formula is asserted.
\end{theorem}

\begin{corollary}
\label{cor:p213-residual-picard-curve-local-interleaved-tower-bt}
In the setting of
\Cref{thm:p213-residual-picard-curve-local-interleaved-tower}, assume that
$B=B_\ell$ has pure dimension $n\geq2$.  Let $T$ be a positive
$\mathrm d'$- and $\mathrm d''$-closed ordinary CLD current of bidegree
$(p,p)$, $0\leq p\leq n-1$, and let $u,v\in C^0(B,\mathbf R)$ satisfy
\[
                         \dprimeddprime[u]=\dprimeddprime[v].
\]
Then
\[
                         \dprimeddprime(uT)=\dprimeddprime(vT).
\]
In particular, this holds for every continuous psh pair in the
continuous-potential predicate of P21.3.
\end{corollary}

\begin{corollary}
\label{cor:p213-residual-picard-curve-local-interleaved-tower-cover-bt}
Let $K$ be algebraically closed, complete, and nontrivially valued, let
$X$ be a good, topologically separated, pure, boundaryless strict
$K$-analytic space of dimension $n\geq2$, and let $V\subseteq X$ be
ordinary open.  Suppose that $V$ has an ordinary open cover by sets $B$
which, with their induced analytic structures, satisfy the geometric
hypotheses of
\Cref{cor:p213-residual-picard-curve-local-interleaved-tower-bt}.  Then
every positive closed ordinary CLD current $T$ of bidegree $(p,p)$ on $V$,
$0\leq p\leq n-1$, has Bedford--Taylor independence for every continuous
psh pair $u,v$ on $V$ with equal Hessian currents.
\end{corollary}

\begin{corollary}
\label{cor:p213-residual-picard-curve-local-interleaved-tower-basis-bt}
Let $K$ and $X$ be as in
\Cref{cor:p213-residual-picard-curve-local-interleaved-tower-cover-bt}.
Suppose that the ordinary topology of $X$ has a basis of open sets $B$ such
that every $B$, with its induced analytic structure, satisfies the
geometric hypotheses of
\Cref{cor:p213-residual-picard-curve-local-interleaved-tower-bt}.  Then the
continuous-pair Bedford--Taylor predicate in P21.3 holds on every ordinary
open subset of $X$ for every positive closed ordinary CLD current.
\end{corollary}
