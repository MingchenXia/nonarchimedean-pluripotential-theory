\begin{proposition}
\label{prop:p213-residual-picard-smooth-curve}
Let $K$ be algebraically closed, complete, and nontrivially valued, let
$C$ be a smooth proper algebraic $K$-curve, and let
$W\subset C^{\mathrm{an}}$ be open.  Suppose that
$h\in C^0(W,\mathbf R)$ satisfies
\[
                         \dprimeddprime[h]=0.
\]
For a type~$2$ point $x\in W$, let $C_x$ be its smooth projective
residue curve and let
\[
 \operatorname{ob}_x(h)\in
 \operatorname{Pic}^0(C_x)\otimes_{\mathbf Z}\mathbf R
\]
be the residual class of the harmonic germ of $h$ at $x$.  Then the
following are equivalent:
\begin{enumerate}
\item $h$ is ordinary CLD-smooth near $x$ and
      $\dprimeddprime h=0$ there as a smooth form;
\item $\operatorname{ob}_x(h)=0$.
\end{enumerate}
At points of types $1$, $3$, and $4$, the germ of $h$ is automatically
ordinary CLD-smooth with zero smooth Hessian.

Consequently, if
\begin{equation}
\label{eq:p213-residual-picard-vanishing}
       \operatorname{Pic}^0(C_x)\otimes_{\mathbf Z}\mathbf R=0
       \qquad\text{for every type~$2$ point }x\in W,
\end{equation}
then $h$ is ordinary CLD-smooth on $W$ and
$\dprimeddprime h=0$ as a smooth form.  The hypothesis
\eqref{eq:p213-residual-picard-vanishing} holds, in particular, when the
residue field $\widetilde K$ is algebraic over a finite field, or when
$C$ is a Mumford curve.
\end{proposition}

\begin{theorem}
\label{thm:p062-interleaved-polyannulus-mixed-bundle-towers}
Let $K$, $C$, and $W\subset C^{\mathrm{an}}$ be as in
\Cref{thm:wanner-curve-comparison}, let $\ell\geq1$, and put $B_0=W$.
Suppose that
\[
 B_\ell\xrightarrow{\pi_\ell}B_{\ell-1}
 \longrightarrow\cdots\longrightarrow B_1
 \xrightarrow{\pi_1}B_0
\]
is a finite tower in which every $B_j$ is a positive-dimensional good,
topologically separated, boundaryless, pure strict $K$-analytic space.  At
every stage assume that exactly one of the following descriptions is
specified.
\begin{enumerate}
\item The morphism $\pi_j$ is a locally trivial standard strict
open-polydisc bundle as in
\Cref{prop:p062-open-polydisc-bundle-descent}; in particular, its positive
fibre dimension and chart radii satisfy the conditions there.
\item The base $B_{j-1}$ is locally holomorphically separated and
paracompact, and $\pi_j$ is a locally trivial analytic projective-space
bundle as in \Cref{prop:p062-projective-bundle-descent}, with fibre
$(\mathbf P_K^{q_j})^{\mathrm{an}}$ for some $q_j\geq1$.
\item The base $B_{j-1}$ is locally holomorphically separated and
paracompact, and $\pi_j$ is a locally trivial smooth proper split-toric
bundle as in
\Cref{prop:p062-smooth-proper-split-toric-bundle-descent}, with fibre
$X_{\Sigma_j}^{\mathrm{an}}$ for a smooth proper split toric $K$-variety
$X_{\Sigma_j}$ of dimension $q_j\geq1$.
\item The base $B_{j-1}$ is locally holomorphically separated and
paracompact, and $\pi_j$ is a locally trivial proper smooth curve bundle as
in \Cref{prop:p062-proper-smooth-curve-bundle-descent}, with fibre
$D_j^{\mathrm{an}}$ for a nonempty geometrically connected smooth proper
algebraic $K$-curve $D_j$.
\item There are an integer $s_j\geq1$ and standard strict open annuli
\[
 A_{j,k}=\{z\in(\mathbf G_{\mathrm m,K})^{\mathrm{an}}:
                 r_{j,k}<|T_{j,k}(z)|<R_{j,k}\},
 \qquad 0<r_{j,k}<R_{j,k},\quad r_{j,k},R_{j,k}\in|K^\times|,
\]
for $1\leq k\leq s_j$, together with an isomorphism over $B_{j-1}$
\[
 B_j\simeq B_{j-1}\times_K\prod_{k=1}^{s_j}A_{j,k}
\]
under which $\pi_j$ is the first projection.  Write
$q_{j,k}:B_j\to A_{j,k}$ for the corresponding projection.
\end{enumerate}
Let $J\subseteq\{1,\ldots,\ell\}$ be the set of stages of the fifth
type, put
\[
 \Pi=\pi_1\circ\cdots\circ\pi_\ell:B_\ell\longrightarrow W,
\]
and, for $j\in J$, put
\[
 \rho_j=\pi_{j+1}\circ\cdots\circ\pi_\ell:B_\ell\longrightarrow B_j,
 \qquad
 L_{j,k}=\rho_j^*q_{j,k}^*\log|T_{j,k}|,
\]
where $\rho_\ell$ is the identity when $\ell\in J$.  If
$h\in C^0(B_\ell,\mathbf R)$ satisfies
$\mathrm d'\mathrm d''[h]=0$, then there are unique continuous functions
\[
 b:W\to\mathbf R,
 \qquad a_{j,k}:W\to\mathbf R\quad(j\in J,\ 1\leq k\leq s_j)
\]
such that
\[
 h=\Pi^*b+
   \sum_{j\in J}\sum_{k=1}^{s_j}(a_{j,k}\circ\Pi)L_{j,k},
 \qquad \mathrm d'\mathrm d''[b]=0,
\]
and every $a_{j,k}$ is locally constant.  In particular, $h$ is
classically pluriharmonic, locally $\mathbf R$-PL, and
Gubler--Jell--Rabinoff harmonic.
\end{theorem}

\begin{lemma}
\label{lem:ch15-log-smooth}
Let $f$ be an invertible analytic function on an open set $U\subset X$.
Then $\log|f|$ is a smooth function on $U$ in the sense of ordinary CLD
forms, and
\[
 \dprime\log|f|=-\dprime t,\qquad
 \ddprime\log|f|=-\ddprime t,\qquad
 \dprimeddprime\log|f|=0,
\]
where $t$ is the affine coordinate on $\mathbf R$ pulled back by the moment
map $f$, that is, $t=-\log|f|$.
\end{lemma}

\begin{proposition}
\label{prop:cld-form-pullback}
Let $g:Y\to X$ be a morphism of topologically separated analytic spaces.
There is a functorial morphism
\[
                 g^*:A_X^{p,q}\longrightarrow g_*A_Y^{p,q}
\]
commuting with wedge, $J$, $\dprime$, and $\ddprime$.
\end{proposition}

\begin{corollary}
\label{cor:ch16-smooth-h}
Let $U$ be open in a good, topologically separated, pure, boundaryless
strict analytic space, let $T$ be a positive closed ordinary CLD current on
$U$, and let $u,v$ be continuous functions on $U$ satisfying
$\dprimeddprime[u]=\dprimeddprime[v]$.  If $h=u-v$ is ordinary CLD-smooth
and $\dprimeddprime h=0$, then
$\dprimeddprime(hT)=0$, equivalently
$\dprimeddprime(uT)=\dprimeddprime(vT)$.
\end{corollary}

\begin{problem}
\label{prob:frontier-cross-fan-upstairs}
Let $K$ be a complete non-Archimedean field, let $n\geq2$, and let $X$
be a good separated strict boundaryless pure $n$-dimensional
$K$-analytic space.  Determine whether every positive
$\mathrm d'$- and $\mathrm d''$-closed CLD current $T$ of bidegree
$(p,p)$, $0\leq p\leq n-1$, has the following property: for every open
$W\subseteq X$ and all continuous psh functions $U,V$ on $W$,
\[
                  \mathrm d'\mathrm d''[U]
                  =\mathrm d'\mathrm d''[V]
\]
implies
\[
 \mathrm d'\mathrm d''\bigl(U\,T|_W\bigr)
 =\mathrm d'\mathrm d''\bigl(V\,T|_W\bigr).
\]
If this fails, construct a counterexample.
\end{problem}
