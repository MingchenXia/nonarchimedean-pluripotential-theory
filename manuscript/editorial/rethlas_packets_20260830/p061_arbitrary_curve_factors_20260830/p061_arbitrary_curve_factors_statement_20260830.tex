\begin{lemma}
\label{lem:p061-nilpotent-rpl-invariance}
Let \(K\) be a complete non-Archimedean field with nontrivial absolute
value, and let
\[
                         \iota:X_0\longrightarrow X
\]
be a nilpotent closed immersion of good strict \(K\)-analytic spaces.  Thus
its coherent defining ideal is locally nilpotent.  The map \(\iota\)
identifies the underlying topological spaces and their rational
G-topologies.  If \(g:X\to\mathbf R\) is continuous, then
\[
 g\text{ is }\mathbf R\text{-PL on }X
 \quad\Longleftrightarrow\quad
 \iota^*g\text{ is }\mathbf R\text{-PL on }X_0.
\]
\end{lemma}

\begin{lemma}
\label{lem:p061-product-factor-reduction}
Let \(m\geq0\), let \(L\) be a complete non-Archimedean field, and let
\(A_1,\ldots,A_m,B\) be strict \(L\)-affinoid algebras.  Let
\(I_i\subset A_i\) and \(I_B\subset B\) be nilpotent ideals.  Put
\[
\begin{aligned}
 P&=A_1\widehat\otimes_L\cdots\widehat\otimes_LA_m
       \widehat\otimes_LB,\\
 Q&=(A_1/I_1)\widehat\otimes_L\cdots
       \widehat\otimes_L(A_m/I_m)\widehat\otimes_L(B/I_B).
\end{aligned}
\]
For \(m=0\), these formulas mean \(P=B\) and \(Q=B/I_B\).
Then the canonical homomorphism \(P\to Q\) is surjective with nilpotent
kernel
\[
                         J=I_1P+\cdots+I_mP+I_BP.
\]
Consequently
\[
                         \mathcal M(Q)\longrightarrow\mathcal M(P)
\]
is a nilpotent closed immersion and a homeomorphism.  If
\(V=\mathcal M(P_V)\) is a strict affinoid domain in \(\mathcal M(P)\),
then its inverse image is
\(\mathcal M(P_V/JP_V)\); in particular, the base-changed immersion is
again nilpotent and a homeomorphism.
\end{lemma}

\begin{theorem}
\label{thm:p061-arbitrary-curve-factors-ph-rpl}
Let \(m\geq0\), and let \(K\) be a complete non-Archimedean field with
nontrivial absolute value.  Let \(C_1,\ldots,C_m\) be arbitrary good
strict \(K\)-analytic
curves, not assumed reduced or geometrically reduced.  Let \(Y\) be a
good strict \(K\)-analytic space whose reduction
\(Y_{\mathrm{red}}\) is quasi-smooth.  Assume that
\[
              X=C_1\times_K\cdots\times_KC_m\times_KY
\]
is good, with the convention \(X=Y\) when \(m=0\).  Then every
classically pluriharmonic function
\[
                              h:X\longrightarrow\mathbf R
\]
is \(\mathbf R\)-PL.  Equivalently, \(h\) is G-locally a finite real
linear combination of logarithms of analytic units.  Moreover, \(h\) is
harmonic in the sense of Gubler--Jell--Rabinoff.

The same conclusion holds on a good strict analytic space that is
analytically locally isomorphic to an open analytic subspace of such a
finite product.
\end{theorem}

\begin{remark}
\label{rem:p061-arbitrary-curve-factors-firewall}
\Cref{thm:p061-arbitrary-curve-factors-ph-rpl} removes reducedness and
geometric reducedness from the finite curve factors and allows a nilpotent
thickening of the quasi-smooth factor.  It makes no claim for a reduced
singular factor of analytic dimension at least two, an arbitrary
nonproduct singular germ, an infinite product, a nongood or nonstrict
space, a trivially valued field, or a merely continuous function with zero
Chambert--Loir--Ducros current.  The coefficient group cannot in general be
reduced from \(\mathbf R\) to \(\mathbf Q\) or \(\mathbf Z\), by
\Cref{prop:p061-integral-pl-counterexample}.
\end{remark}
