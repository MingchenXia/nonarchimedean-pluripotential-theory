\begin{lemma}
\label{lem:p061-rational-section-tube}
Let \(K\) be a complete non-Archimedean field with nontrivial absolute
value, let \(A\) and \(B\) be strict \(K\)-affinoid algebras, and let
\(a\in\mathcal M(A)(K)\).  Put
\[
 P=A\widehat\otimes_KB,
 \qquad
 \epsilon=\epsilon_a\widehat\otimes 1:P\longrightarrow B.
\]
The following assertions hold.
\begin{enumerate}
\item If \(G\in\ker\epsilon\) and \(\eta>0\), there is a strict affinoid
neighborhood \(U\) of \(a\) in \(\mathcal M(A)\) such that
\[
          \sup_{U\times_K\mathcal M(B)}|G|<\eta.
\]
\item For every \(z\in\mathcal M(B)\), the product domains
\(U\times_KV\), where \(U\) and \(V\) run through strict affinoid
neighborhoods of \(a\) and \(z\), form a neighborhood basis at the point
\(s_a(z)\in\mathcal M(P)\).
\item If \(f\) is an analytic unit on a product affinoid neighborhood
\(U_0\times_KV\) of \(s_a(z)\), then, after shrinking \(U_0\) around
\(a\), one has, with \(p_V:U_0\times_KV\to V\) the second projection,
\begin{equation}
\label{eq:p061-unit-constant-transverse}
              |f|=p_V^*|s_a^*f|
              \quad\text{on }U_0\times_KV.
\end{equation}
\end{enumerate}
\end{lemma}

\begin{lemma}
\label{lem:p061-parameterized-rigid-point}
Let \(K\) be an algebraically closed complete non-Archimedean field with
nontrivial absolute value.  Let \(S\) and \(Z\) be good strict
\(K\)-analytic spaces, let \(a\in S(K)\), and let \(z\in Z\).  Assume
that \(S\times_KZ\) is pure-dimensional on a neighborhood of
\(s_a(z)\).  Write
\[
 p_S:S\times_KZ\longrightarrow S,
 \qquad
 p_Z:S\times_KZ\longrightarrow Z,
 \qquad
 s_a:Z\longrightarrow S\times_KZ
\]
for the projections and the section induced by \(a\).  If an
\(\mathbf R\)-PL function \(u\) is defined on a neighborhood of
\(s_a(z)\), then there are strict affinoid neighborhoods \(U\) of \(a\)
and \(V\) of \(z\) such that
\begin{equation}
\label{eq:p061-parameterized-rigid-point}
 u|_{U\times_KV}=p_Z^*\bigl(s_a^*u|_V\bigr).
\end{equation}
In particular, the function on the right-hand side of
\eqref{eq:p061-parameterized-rigid-point} is \(\mathbf R\)-PL.
\end{lemma}

\begin{proposition}
\label{prop:p061-curve-product-normalization-descent}
Let \(K\) be as in \Cref{lem:p061-parameterized-rigid-point}, let \(C\)
be a reduced good strict \(K\)-analytic curve, and let \(Z\) be a good
strict pure-dimensional \(K\)-analytic space.  Assume that
\(C\times_KZ\) is good.  Denote the normalization by
\[
                         \nu:C^\nu\longrightarrow C
\]
and put \(q=\nu\times\mathrm{id}_Z\).  Let
\(g:C\times_KZ\to\mathbf R\) be continuous.  If \(q^*g\) is
\(\mathbf R\)-PL, then \(g\) is \(\mathbf R\)-PL.
\end{proposition}

\begin{theorem}
\label{thm:p061-curve-product-ph-rpl}
Let \(K\) be a complete non-Archimedean field with nontrivial absolute
value.  Let \(C\) be a geometrically reduced good strict \(K\)-analytic
curve, and let \(Y\) be a good strict quasi-smooth \(K\)-analytic space.
Assume that \(X=C\times_KY\) is good.  Then every classically
pluriharmonic function
\[
                              h:X\longrightarrow\mathbf R
\]
is \(\mathbf R\)-PL.  Equivalently, \(h\) is G-locally a finite real
linear combination of logarithms of analytic units.  Moreover, \(h\) is
harmonic in the sense of Gubler--Jell--Rabinoff.

The same conclusion holds on a good strict analytic space that is
analytically locally isomorphic to an open analytic subspace of such a
product.
\end{theorem}

\begin{corollary}
\label{cor:p061-finite-curve-products-ph-rpl}
Let \(K\) be a complete non-Archimedean field with nontrivial absolute
value.  Let \(C_1,\ldots,C_m\) be geometrically reduced good strict
analytic curves over \(K\), and let \(Y\) be a good strict quasi-smooth
\(K\)-analytic space.  Assume that
\[
                   C_1\times_K\cdots\times_KC_m\times_KY
\]
is good.  Every classically pluriharmonic function on this product is
\(\mathbf R\)-PL and harmonic in the sense of
Gubler--Jell--Rabinoff.

The same conclusion holds on a good strict analytic space that is
analytically locally isomorphic to an open analytic subspace of such a
finite product.
\end{corollary}

\begin{example}
\label{ex:p061-nodal-cuspidal-products}
Suppose that \(K\) is a complete algebraically closed non-Archimedean
field of characteristic zero with nontrivial absolute value.  The
conclusions above apply, for every good strict quasi-smooth \(Y\), to
finite products of the curves
\[
 \mathcal M\bigl(K\langle S,T\rangle/(ST)\bigr)
 \quad\text{and}\quad
 \mathcal M\bigl(K\langle S,T\rangle/(T^2-S^3)\bigr)
\]
with \(Y\), as well as to their open analytic subspaces.  Thus they cover
nodal and cuspidal cylinders, and products of such cylinders, including
all their non-quasi-smooth strata.
\end{example}
