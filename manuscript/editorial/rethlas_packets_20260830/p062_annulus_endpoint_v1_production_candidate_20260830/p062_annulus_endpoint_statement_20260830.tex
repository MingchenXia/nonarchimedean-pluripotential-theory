% Exact frozen candidate: one-dimensional endpoint variants only.
% No product or higher-dimensional endpoint assertion is made.

\begin{proposition}\label{prop:p062-annulus-endpoint-arbitrary-field}
Let \(K\) be a complete non-Archimedean field with nontrivial absolute
value, and let
\(
             0<r<R
\)
belong to \(|K^\times|\).  Consider any one of the three standard strict
analytic domains
\[
 \begin{aligned}
 A_{r,R}^{\mathrm{oc}}
   &=\{x\in(\mathbf G_{\mathrm m,K})^{\mathrm{an}}:
                                      r<|T(x)|\leq R\},\\
 A_{r,R}^{\mathrm{co}}
   &=\{x\in(\mathbf G_{\mathrm m,K})^{\mathrm{an}}:
                                      r\leq|T(x)|<R\},\\
 A_{r,R}^{\mathrm{cc}}
   &=\{x\in(\mathbf G_{\mathrm m,K})^{\mathrm{an}}:
                                      r\leq|T(x)|\leq R\}.
 \end{aligned}
\]
Write \(A\) for the selected domain and
\(\operatorname{Int}(A)\) for its absolute analytic interior.  If
\(h\in C^0(A,\mathbf R)\) satisfies
\begin{equation}\label{eq:p062-annulus-endpoint-zero-current}
                 \mathrm d'\mathrm d''[h]=0
                 \qquad\text{on }\operatorname{Int}(A)
\end{equation}
as a Chambert--Loir--Ducros current, then there are unique
\(a,b\in\mathbf R\) such that
\begin{equation}\label{eq:p062-annulus-endpoint-affine}
                         h=a\log|T|+b
                         \qquad\text{on }A.
\end{equation}

This assertion is only one-dimensional.  It makes no endpoint claim for a
product, a polycylinder, or any higher-dimensional analytic domain.
\end{proposition}
