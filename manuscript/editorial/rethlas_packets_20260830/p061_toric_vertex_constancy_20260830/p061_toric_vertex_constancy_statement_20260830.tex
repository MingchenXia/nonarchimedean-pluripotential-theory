% Exact research-candidate statement for the toric fixed-point scope of P06.1.
\begin{theorem}
\label{thm:p061-toric-fixed-point-constancy}
Let \(K\) be a complete non-Archimedean field with nontrivial absolute
value.  Let \(N\) be a lattice, put \(M=N^{\vee}\), and let
\(\sigma\subset N_{\mathbf R}\) be a full-dimensional strongly convex
rational polyhedral cone.  Set
\[
 S:=\sigma^{\vee}\cap M,
 \qquad
 X_{\sigma}:=\operatorname{Spec}K[S],
\]
and denote by \(x_{\sigma}\in X_{\sigma}^{\mathrm{an}}\) the
\(K\)-rational torus-fixed point at which every monomial
\(\chi^u\), \(u\in S\setminus\{0\}\), vanishes.

Let \(\Omega\subset X_{\sigma}^{\mathrm{an}}\) be an open neighbourhood
of \(x_{\sigma}\).  Every classically pluriharmonic function
\(h:\Omega\to\mathbf R\) is constant on an analytic neighbourhood of
\(x_{\sigma}\).  More precisely, choose
\(\varpi\in K\) with \(0<|\varpi|<1\) and
\(v\in N\cap\operatorname{int}(\sigma)\), and put
\(\ell(u):=\langle u,v\rangle\).  For \(n\geq1\), let
\begin{equation}
\label{eq:p061-toric-weighted-affinoid}
 U_n:=\left\{x\in X_{\sigma}^{\mathrm{an}}:
       |\chi^u(x)|\leq |\varpi|^{n\ell(u)}
       \text{ for every }u\in S\right\}.
\end{equation}
It is equivalent in \eqref{eq:p061-toric-weighted-affinoid} to impose the
inequalities for any finite system of monoid generators of \(S\).  The
\(U_n\) form a basis of strict affinoid neighbourhoods of
\(x_{\sigma}\); every \(U_n\) has a singleton Shilov boundary; and
\(h|_{U_n}\) is constant for all sufficiently large \(n\).

Consequently, the germ of \(h\) at \(x_{\sigma}\) is
\(\mathbf R\)-PL and harmonic in the sense of
Gubler--Jell--Rabinoff.  These conclusions require neither simpliciality
of \(\sigma\) nor a restriction on the residue characteristic of \(K\).
\end{theorem}
