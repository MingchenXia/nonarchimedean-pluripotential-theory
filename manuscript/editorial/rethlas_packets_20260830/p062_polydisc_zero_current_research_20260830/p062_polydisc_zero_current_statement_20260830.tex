\begin{proposition}\label{prop:p062-polydisc-trace-separation-candidate}
Let \(K\) be an algebraically closed complete non-Archimedean field with
nontrivial absolute value, let \(n\geq 1\), and let
\[
 \begin{aligned}
 D^-_{\mathbf R}
 &:=\bigl\{x\in (\mathbf A^n_K)^{\an}:|T_i(x)|<R_i
       \text{ for }1\leq i\leq n\bigr\},\\
 D^+_{\mathbf R}
 &:=\bigl\{x\in (\mathbf A^n_K)^{\an}:|T_i(x)|\leq R_i
       \text{ for }1\leq i\leq n\bigr\},\\
 \mathbf R&=(R_1,\ldots,R_n)\in |K^\times|^n.
 \end{aligned}
\]
For \(a\in D^-_{\mathbf R}(K)\) and
\(\mathbf r=(r_1,\ldots,r_n)\) with \(0<r_i<R_i\), denote by
\(\eta_{a,\mathbf r}\) the Gauss point of the closed polydisc with centre
\(a\) and polyradius \(\mathbf r\). Suppose that
\(h\in C^0(D^-_{\mathbf R},\mathbb R)\) has the following property: for
every \(a\in D^-_{\mathbf R}(K)\) and every
\(\boldsymbol\rho\) with \(0<\rho_i<R_i\), the function
\begin{equation}\label{eq:p062-polydisc-radial-trace-candidate}
 \tau_{a,\boldsymbol\rho}(u_1,\ldots,u_n)
 :=h\bigl(\eta_{a,(\mathrm e^{-u_1},\ldots,
                         \mathrm e^{-u_n})}\bigr),
 \qquad u_i>-\log\rho_i,
\end{equation}
has zero distributional Lagerberg Hessian. Then \(h\) is constant.
The same conclusion holds on the closed polydisc \(D^+_{\mathbf R}\) if
the trace hypothesis is imposed for all centres in
\(D^+_{\mathbf R}(K)\) and all \(0<\rho_i<R_i\).
\end{proposition}

\begin{theorem}\label{thm:p062-polydisc-zero-current-candidate}
In the setting of \cref{prop:p062-polydisc-trace-separation-candidate},
let \(h\in C^0(D^-_{\mathbf R},\mathbb R)\). If
\begin{equation}\label{eq:p062-polydisc-zero-current-candidate}
             \mathrm d'\mathrm d''[h]=0
\end{equation}
as a Chambert--Loir--Ducros current on \(D^-_{\mathbf R}\), then \(h\)
is constant. In particular, \(h\) is weakly smooth,
\(\mathrm d'\mathrm d''h=0\) as a weakly smooth form, harmonic in the
sense of Gubler--Jell--Rabinoff, and classically pluriharmonic.

If \(h\in C^0(D^+_{\mathbf R},\mathbb R)\) and
\eqref{eq:p062-polydisc-zero-current-candidate} holds on the analytic
interior of \(D^+_{\mathbf R}\), then \(h\) is likewise constant on
\(D^+_{\mathbf R}\).
\end{theorem}
