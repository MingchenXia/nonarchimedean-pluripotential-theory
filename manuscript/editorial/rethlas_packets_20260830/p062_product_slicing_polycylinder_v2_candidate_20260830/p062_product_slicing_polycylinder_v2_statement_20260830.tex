\begin{proposition}\label{prop:p062-product-fibre-zero-current-candidate}
Let \(K\) be a complete non-Archimedean field with nontrivial absolute
value. Let \(B\) be a standard strict open polycylinder over \(K\), let
\(Z\) be either a standard strict open disc or a standard strict open
annulus, and put \(X=B\times_K Z\). If
\(h\in C^0(X,\mathbf R)\) satisfies
\begin{equation}\label{eq:p062-product-zero-current-candidate}
                    \mathrm d'\mathrm d''[h]=0
\end{equation}
as a Chambert--Loir--Ducros current on \(X\), then, for every \(x\in B\),
the restriction
\[
       h_x:=h|_{X_x}\in C^0(X_x,\mathbf R),
       \qquad X_x\simeq Z_{\mathscr H(x)},
\]
satisfies
\begin{equation}\label{eq:p062-fibre-zero-current-candidate}
             \mathrm d'\mathrm d''[h_x]=0
\end{equation}
as a Chambert--Loir--Ducros current over the complete residue field
\(\mathscr H(x)\).
\end{proposition}

\begin{proposition}\label{prop:p062-arbitrary-field-annulus-candidate}
Let \(K\) be as above, let \(0<r<R\) belong to \(|K^\times|\), and put
\[
 A^-_{r,R}:=\{x\in(\mathbf G_{\mathrm m,K})^{\mathrm{an}}:
                         r<|T(x)|<R\}.
\]
If \(h\in C^0(A^-_{r,R},\mathbf R)\) satisfies
\(
 \mathrm d'\mathrm d''[h]=0
\), then there are unique \(a,b\in\mathbf R\) such that
\begin{equation}\label{eq:p062-annulus-affine-candidate}
                         h=a\log|T|+b.
\end{equation}
The same conclusion holds on any standard strict annulus obtained by
making either or both endpoint inequalities non-strict, provided the
current equation is imposed on its absolute analytic interior.
\end{proposition}

\begin{theorem}\label{thm:p062-arbitrary-field-polycylinder-candidate}
Let \(K\) be as above, and let \(p,q\) be nonnegative integers. For
\(1\leq i\leq p\), let
\[
 D_i=\{x\in(\mathbf A^1_K)^{\mathrm{an}}:|T_i(x)|<R_i\},
 \qquad R_i\in|K^\times|,
\]
and, for \(1\leq j\leq q\), let
\[
 A_j=\{x\in(\mathbf G_{\mathrm m,K})^{\mathrm{an}}:
                  r_j<|S_j(x)|<R'_j\},
\]
where
\[
                  0<r_j<R'_j,
 \qquad r_j,R'_j\in|K^\times|.
\]
Put
\[
                         P=\prod_{i=1}^pD_i\times_K
                           \prod_{j=1}^qA_j.
\]
If \(h\in C^0(P,\mathbf R)\) satisfies
\(
 \mathrm d'\mathrm d''[h]=0
\), then there are unique real constants
\(c,\lambda_1,\ldots,\lambda_q\) such that
\begin{equation}\label{eq:p062-polycylinder-affine-candidate}
                   h=c+\sum_{j=1}^q\lambda_j\log|S_j|.
\end{equation}
In particular, \(h\) is ordinary CLD-smooth, weakly smooth, harmonic in
the sense of Gubler--Jell--Rabinoff, and classically pluriharmonic.

The same assertion holds when any of the displayed factor inequalities
is made non-strict, if the current equation is imposed on the absolute
analytic interior of the resulting closed or half-closed polycylinder.
\end{theorem}
