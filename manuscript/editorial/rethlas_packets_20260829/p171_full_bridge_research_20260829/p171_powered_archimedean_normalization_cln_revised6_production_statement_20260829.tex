\begin{lemma}
\label{lem:p171-production-rho-convention}
Let \(A\) be an integral Banach ring and, for \(x\in\cM(A)\), put
\begin{equation}
 \rho(x):=\frac{\log\max\{1,\abs{2}_x\}}{\log 2}.
 \label{eq:p171-production-rho}
\end{equation}
The function \(\rho\) is continuous.  One has \(\rho(x)=0\) exactly when
the absolute value on \(\mathcal H(x)\) is non-Archimedean.  If
\(\rho(x)>0\), then \(0<\rho(x)\leq1\), and there is a character into
\(\C\), unique up to complex conjugation, such that, after completion,
\begin{equation}
 \abs{a}_x=\abs{\chi_x(a)}_\infty^{\rho(x)}.
 \label{eq:p171-production-character}
\end{equation}
Thus \(\mathcal H(x)\) is \(\R\) or \(\C\) endowed with the
\(\rho(x)\)-th power of its usual absolute value.

Fix \(K_\rho=(K,\abs{\cdot}_\infty^\rho)\), where
\(K\in\{\R,\C\}\) and \(0<\rho\leq1\).  For a metric weight
\(\phi_\rho\) obtained by restriction in the Pille--Schneider convention,
define
\begin{equation}
 \overline\phi:=\rho^{-1}\phi_\rho,
 \qquad
 \norm e_{\overline\phi,\infty}
 :=\norm e_{\phi_\rho,\rho}^{1/\rho}.
 \label{eq:p171-production-unpower}
\end{equation}
Under the canonical homeomorphism from the usual real or complex
analytification to the \(K_\rho\)-analytification, this is the metric
transport determined by Pille--Schneider Proposition~3.2.  On the
psh/DFS closure used by their construction, it preserves semipositivity and
uniform approximation, and their fibrewise Monge--Ampere measure is the
Boucksom--Eriksson Archimedean measure of the unpowered metrics.  This is an
unpacking of the cited fibre convention, not an independently normalized
definition of a new complex \(dd^c\)-operator.
\end{lemma}

\begin{theorem}
\label{thm:p171-production-powered-normalization}
Let \(0<\rho\leq1\), let
\(K_\rho=(K,\abs{\cdot}_\infty^\rho)\) with
\(K\in\{\R,\C\}\), and let \(X/K_\rho\) be projective of dimension \(n\).
Nilpotents, several top-dimensional components, and their generic
multiplicities are allowed.  Let \(L_0,\ldots,L_n\) carry continuous
psh-regularizable metrics \(\phi_{i,\rho}\), and put
\(\overline\phi_i=\rho^{-1}\phi_{i,\rho}\).  Then
\begin{equation}
 \MA_\rho(\phi_{1,\rho},\ldots,\phi_{n,\rho})
 =
 \MA_\infty(\overline\phi_1,\ldots,\overline\phi_n).
 \label{eq:p171-production-PN1}
\end{equation}
Both sides use
\[
 [X]_n=\sum_Z
 \operatorname{length}(\mathcal O_{X,\eta_Z})[Z],
\]
so every top-dimensional generic length is retained.

If \(e\neq0\) belongs to the Deligne line
\(I_X(L_0,\ldots,L_n)\), then
\begin{equation}
 \lambda_\rho
 (e;\phi_{0,\rho},\ldots,\phi_{n,\rho})
 =
 \rho\,
 \lambda_\infty
 (e;\overline\phi_0,\ldots,\overline\phi_n).
 \label{eq:p171-production-PN2}
\end{equation}
Thus the recursively defined local height acquires exactly one factor
\(\rho\), rather than one factor in each recursive dimension.

For \(K=\R\), the measures are defined by push-forward from
complexification.  Equations
\eqref{eq:p171-production-PN1}--\eqref{eq:p171-production-PN2} use that
push-forward convention and introduce no additional factor \(2\).
\end{theorem}

\begin{lemma}
\label{lem:p171-production-dimension-zero}
Let \(X=\operatorname{Spec}B\) be finite and flat over \(K_\rho\), and let
\(\phi_{0,\rho}\) be a continuous dpsh metric on \(L_0\).  Put
\(\overline\phi_0=\rho^{-1}\phi_{0,\rho}\).  Under the positive algebraic
identification of the Deligne line with the norm line, one has
\begin{equation}
 \log\norm{\langle s\rangle}_{\langle\phi_{0,\rho}\rangle,\rho}
 =
 \rho\,
 \log\norm{\langle s\rangle}_{\langle\overline\phi_0\rangle,\infty}
 \label{eq:p171-production-Z0}
\end{equation}
for every trivializing section \(s\) of \(L_0\).  If
\(B=\prod_jB_j\), with residue field \(K_j\) and Artin length \(m_j\), the
zero-cycle measure is
\[
 \delta_X=\sum_jm_j[K_j:K]\,\delta_{x_j}.
\]
Hence \eqref{eq:p171-production-Z0} retains every residue degree and every
nonreduced Artin length.  Over \(\R\) it uses the real-fibre push-forward
convention and has no factor \(2\).
\end{lemma}

\begin{theorem}
\label{thm:p171-production-weighted-PG}
Let \(A\) be a Banach ring and \(U\subset\cM(A)\) open.  Fix finitely many
blocks
\[
 \mathbf U_i=(U_{i0},\ldots,U_{i,N_i-1}),\qquad N_i\geq1,
\]
nonnegative block degrees \(d_i\), a finite set \(S\) of multiexponents of
block degree \((d_i)_i\), and real weights \(w_\alpha\).  Let
\(c_\alpha\in\mathcal O(U)\), assume that the coefficient tuple is nowhere
zero, and assume locally that all \(c_\alpha\) belong to one Banach algebra
of analytic functions on a Banach analytic domain.

For \(\rho(x)=0\), set
\begin{equation}
 \Phi(x)=
 \max_{\alpha\in S}
 \bigl(\log\abs{c_\alpha(x)}_x+w_\alpha\bigr),
 \qquad \log0:=-\infty.
 \label{eq:p171-production-PG-zero}
\end{equation}
For \(\rho(x)>0\), choose a character satisfying
\eqref{eq:p171-production-character} and put
\begin{equation}
 F_x=
 \sum_{\alpha\in S}
 \chi_x(c_\alpha)\exp(w_\alpha/\rho(x))\mathbf U^\alpha,
 \qquad
 \Phi(x)=\rho(x)\ph(F_x),
 \label{eq:p171-production-PG-positive}
\end{equation}
where \(\ph\) is Philippon's logarithmic sphere measure with its harmonic
correction.  Then \(\Phi\) is independent of the representing character
and is continuous on \(U\), including across \(\rho=0\).

For every nowhere-vanishing analytic function \(\lambda\),
\begin{equation}
 \Phi(\lambda F)=\log\abs\lambda+\Phi(F).
 \label{eq:p171-production-PG-homogeneity}
\end{equation}
Consequently, if \(R=eF_e\) is an \(\mathcal L\)-valued polynomial and
\(e'=ue\), then
\begin{equation}
 F_{e'}=u^{-1}F_e,
 \qquad
 \Phi_{e'}=\Phi_e-\log\abs u.
 \label{eq:p171-production-PG-frame}
\end{equation}

More specifically, let \(P\neq0\) be a multihomogeneous resultant in
blocks of dimensions \(r_i+1\) and degrees \(\delta_i\), expressed in
coordinates orthonormal for the unpowered Hermitian norms.  For
\(\rho>0\),
\begin{equation}
 \rho\ph(P)
 =
 \rho\int_{\prod_iS^{2r_i+1}}\log\abs{P(z)}\,\mathrm d\eta(z)
 +
 \frac{\rho}{2}
 \sum_i\delta_i\sum_{\ell=1}^{r_i}\frac1\ell.
 \label{eq:p171-production-powered-PH}
\end{equation}
If \(\cR=e^{\otimes Q}F_e\), with \(Q\geq1\), and \(e'=ue\), then on all
of \(\cM(A)\),
\begin{equation}
 F_{e'}=u^{-Q}F_e,
 \qquad
 \Phi_{e'}=\Phi_e-Q\log\abs u.
 \label{eq:p171-production-frame-law}
\end{equation}
At \(\rho=0\), the value in
\eqref{eq:p171-production-frame-law} is defined by the weighted Gauss
maximum \eqref{eq:p171-production-PG-zero}.
\end{theorem}

\begin{lemma}
\label{lem:p171-production-powered-CLN}
Let \(X/K_\rho\) be as above.  For every \(0\leq i\leq n\), assume
explicitly that both metrics in the pair
\(\phi_{i,\rho},\phi'_{i,\rho}\) are continuous psh-regularizable and
semipositive metrics on \(L_i\).  Put
\[
 f_\rho=\phi_{0,\rho}-\phi'_{0,\rho}
\]
and
\begin{equation}
 C=
 \begin{cases}
  0,&n=0,\\[2mm]
  2\displaystyle\max_{1\leq i\leq n}
  (L_0L_1\cdots\widehat{L_i}\cdots L_n),&n\geq1.
 \end{cases}
 \label{eq:p171-production-CLN-constant}
\end{equation}
For every \(0<\rho\leq1\),
\begin{align}
 &\left|
 \int_X f_\rho\,
 \MA_\rho(\phi_{1,\rho},\ldots,\phi_{n,\rho})
 -
 \int_X f_\rho\,
 \MA_\rho(\phi'_{1,\rho},\ldots,\phi'_{n,\rho})
 \right|\notag\\
 &\hspace{25mm}\leq
 C\sum_{i=1}^n
 \norm{\phi_{i,\rho}-\phi'_{i,\rho}}_\infty.
 \label{eq:p171-production-powered-CLN}
\end{align}
The constant is independent of \(\rho\).  When \(n=0\), there are no
metric slots to change, the two integrals coincide, and both sides are
zero.  No estimate is asserted here for an arbitrary continuous test
function outside a difference of two continuous psh-regularizable
semipositive slot-zero metrics.
\end{lemma}

\begin{corollary}
\label{cor:p171-production-family-CLN}
Let \(A\) be an integral Banach ring, let \(X/A\) be flat and projective
with nonempty \(n\)-dimensional fibres, and suppose that the line bundles
and metric pairs in the preceding lemma come from \(X\).  Assume on every
fibre that both metrics in each pair, including the slot-zero pair, are
continuous psh-regularizable and semipositive.  The constant
\eqref{eq:p171-production-CLN-constant}, formed from the fibre-independent
mixed intersection numbers, works on every fibre for the test
\(\phi_0-\phi'_0\).

At points with \(\rho(x)>0\), this is
\eqref{eq:p171-production-powered-CLN}.  At points with \(\rho(x)=0\), it
is the non-Archimedean FS-pair estimate and its continuous semipositive
extension.  No division by \(\rho\) is made on the endpoint locus.  The
claim remains restricted to differences of admissible semipositive
slot-zero metrics, and includes \(n=0\) with constant \(0\).
\end{corollary}

\begin{corollary}
\label{cor:p171-production-product-net}
Retain the family of the preceding corollary and assume the following
finite-level continuity statement: for every DFS test function
\(\psi_q\) and every tuple of tropical Fubini--Study metrics
\((\phi_{i,a_i})_{i=1}^n\), the function
\[
 x\longmapsto
 \int_{X_x^{\rm an}}\psi_q\,
 \MA_x(\phi_{1,a_1},\ldots,\phi_{n,a_n})
\]
is continuous on \(\cM(A)\).

Let \(M=(L_1\cdots L_n)\geq0\) be the common fibre mass.  Let
\(\psi_q\to\psi\) be a uniformly convergent DFS net and write, for each
\(q\),
\[
 \psi_q=\alpha_q-\beta_q,
\]
where \(\alpha_q,\beta_q\) are tropical Fubini--Study metrics on one
semiample line bundle \(H_q\).  Let \(C_q\) be the CLN constant associated
to \(H_q,L_1,\ldots,L_n\), with \(C_q=0\) for \(n=0\).  For each \(i\),
let \(\phi_{i,a_i}\to\phi_i\) be a uniformly convergent net of tropical
Fubini--Study metrics.

Put
\[
 P=(0,1]^{\rm op}\times Q\times\prod_{i=1}^nA_i
\]
with its coordinatewise order, and let \(D\subset P\) consist of the
indices \(d=(\eps,q,\mathbf a)\) satisfying
\begin{align}
 (1+M)\norm{\psi_q-\psi}_\infty&\leq\eps/2,
 \label{eq:p171-production-product-test}\\
 (1+C_q)\sum_{i=1}^n
 \norm{\phi_{i,a_i}-\phi_i}_\infty&\leq\eps/2.
 \label{eq:p171-production-product-metrics}
\end{align}
Then \(D\) is directed and cofinal in \(P\).  For \(d\in D\), set
\[
 F_d(x)=
 \int_{X_x^{\rm an}}\psi_q\,
 \MA_x(\phi_{1,a_1},\ldots,\phi_{n,a_n}),
 \qquad
 F(x)=
 \int_{X_x^{\rm an}}\psi\,
 \MA_x(\phi_1,\ldots,\phi_n).
\]
Each \(F_d\) is continuous by the finite-level hypothesis, and
\begin{equation}
 \sup_{x\in\cM(A)}\abs{F_d(x)-F(x)}
 \leq
 M\norm{\psi_q-\psi}_\infty
 +
 C_q\sum_i\norm{\phi_{i,a_i}-\phi_i}_\infty
 \leq\eps.
 \label{eq:p171-production-product-bound}
\end{equation}
Consequently \(F_d\to F\) uniformly and \(F\) is continuous.  This
argument uses neither sequences nor a uniform bound on the constants
\(C_q\).  The arbitrary continuous limit \(\psi\) is never inserted into
the slot-zero CLN estimate; that estimate is applied only to the
admissible DFS decomposition pair \((\alpha_q,\beta_q)\).
\end{corollary}
