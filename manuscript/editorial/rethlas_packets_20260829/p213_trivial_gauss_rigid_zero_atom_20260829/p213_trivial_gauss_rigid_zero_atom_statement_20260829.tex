\begin{lemma}
\label{lem:p213-trivial-gauss-rigid-zero-atom}
Let $K$ be a complete trivially valued non-Archimedean field, let $X$ be
a good separated strict $K$-analytic space, and let $y\in X$ be a rigid
point.  Assume that
\[
                         y\in\operatorname{Int}(X/K),
\]
that the ordinary analytic local ring $\mathcal O_{X,y}$ is normal, and
that some open neighborhood $\Omega$ of $y$ is purely one-dimensional.
Let $u$ be a continuous CLD-plurisubharmonic function on $\Omega$, and put
\[
 \Omega^\circ:=\Omega\cap\operatorname{Int}(X/K),
 \qquad
 \mu_u:=\mathrm d'\mathrm d''[u|_{\Omega^\circ}].
\]
Then $\mu_u$ is a positive Radon measure and
\begin{equation}
\label{eq:p213-trivial-gauss-rigid-zero-atom}
                         \mu_u(\{y\})=0.
\end{equation}
The conclusion remains valid when $K$ is imperfect and when
$\mathscr H(y)/K$ is inseparable.
\end{lemma}

\begin{remark}
\label{rem:p213-trivial-gauss-rigid-zero-atom-index}
The proof does not use the printed exponent in
\cite[Corollary~16.8.6]{CLD25}.  With $p$ denoting the number of curvature
factors, that exponent must be $n-p$, not $p$; in particular, the corrected
corollary does not by itself exclude an atom at a point on a curve.
\end{remark}
