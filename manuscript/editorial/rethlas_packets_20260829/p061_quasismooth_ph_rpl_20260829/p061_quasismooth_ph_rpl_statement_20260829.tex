\begin{lemma}
\label{lem:p061-rigid-etale-rpl-descent}
Let \(K\) be a complete non-Archimedean field with nontrivial absolute
value.  Let \(U\) and \(V\) be strict \(K\)-affinoid rigid spaces, and let
\[
                         q:V\longrightarrow U
\]
be a covering for the rigid \(\acute{e}\)tale topology.  Write
\(U^{\mathrm B}\) and \(V^{\mathrm B}\) for the associated Berkovich
spaces.  If \(g:U^{\mathrm B}\to\mathbf R\) is continuous and
\(g\circ q^{\mathrm B}\) is \(\mathbf R\)-PL, then \(g\) is
\(\mathbf R\)-PL.
\end{lemma}

\begin{theorem}
\label{thm:p061-quasismooth-ph-rpl}
Let \(K\) be a complete non-Archimedean field with nontrivial absolute
value, and let \(X\) be a good, strict, quasi-smooth \(K\)-analytic space.
Then every classically pluriharmonic function
\[
                             h:X\longrightarrow\mathbf R
\]
is \(\mathbf R\)-PL.  Equivalently, \(h\) is G-locally a finite real
linear combination of functions \(\log|f|\), with \(f\) an analytic unit.
Moreover, \(h\) is harmonic in the sense of
Gubler--Jell--Rabinoff.  No global quasicompactness, separatedness, or
boundarylessness hypothesis is required.
\end{theorem}

\begin{proposition}
\label{prop:p061-integral-pl-counterexample}
The conclusion of \Cref{thm:p061-quasismooth-ph-rpl} cannot be strengthened
from \(\mathbf R\)-PL to integral PL, or even to \(\mathbf Q\)-PL.  More
precisely, choose
\(\pi\in K\) with \(0<|\pi|<1\), put
\[
 X=\mathcal M\bigl(K\langle T,S\rangle/(TS-\pi)\bigr),
 \qquad h=\sqrt2\log|T|,
\]
and regard \(X\) as the closed annulus \(|\pi|\leq |T|\leq1\).  Then \(X\)
is good, compact, strict, and quasi-smooth, and it has a strictly semistable
formal \(K^\circ\)-model.  The function \(h\) is classically
pluriharmonic and \(\mathbf R\)-PL, but it is not \(\mathbf Q\)-PL and hence
is not integral PL.
\end{proposition}
