\begin{lemma}
\label{lem:p21p3-componentwise-normal-rigid-zero-atom}
Let $K$ be a complete nontrivially valued non-Archimedean field, let $X$ be
a good separated strict $K$-analytic space, and let $y\in X$ be a rigid
point.  Assume that
\[
                         y\in\operatorname{Int}(X/K),
\]
and that the ordinary analytic local ring $\mathcal O_{X,y}$ is normal.
Let $\Omega\subseteq X$ be an open neighborhood of $y$ on which $X$ is
purely one-dimensional, and let $u$ be a continuous
CLD-plurisubharmonic function on $\Omega$.  Put
\[
 \Omega^\circ:=\Omega\cap\operatorname{Int}(X/K),
 \qquad
 \mu_u:=\mathrm d'\mathrm d''[u|_{\Omega^\circ}].
\]
Then $\mu_u$ is a positive Radon measure on $\Omega^\circ$ and has no atom
at $y$:
\begin{equation}
\label{eq:p213-normal-rigid-zero-atom}
                         \mu_u(\{y\})=0.
\end{equation}
\end{lemma}

\begin{remark}
\label{rem:p213-normal-rigid-zero-atom-scope}
Normality is required only for the local ring at $y$; neither $X$ nor
$\Omega$ is assumed normal away from $y$.  The point is required only to
lie in the analytic interior; the ambient space need not be boundaryless
away from a neighborhood of $y$.  The normality is ordinary normality, not
geometric normality.  Thus the statement does not restore rig-smoothness
over an imperfect ground field.  No assertion is made for a boundary point,
a nonrigid point, a nonnormal local ring, a trivially valued ground field,
or a discontinuous psh function.
\end{remark}

\begin{lemma}
\label{lem:p213-trivial-gauss-rigid-zero-atom}
Let $K$ be a complete trivially valued non-Archimedean field, let $X$ be
a good separated strict $K$-analytic space, and let $y\in X$ be a rigid
point.  Assume that
\[
                         y\in\operatorname{Int}(X/K),
\]
that the ordinary analytic local ring $\mathcal O_{X,y}$ is normal, and
that some open neighborhood $\Omega$ of $y$ is purely one-dimensional.
Let $u$ be a continuous CLD-plurisubharmonic function on $\Omega$, and put
\[
 \Omega^\circ:=\Omega\cap\operatorname{Int}(X/K),
 \qquad
 \mu_u:=\mathrm d'\mathrm d''[u|_{\Omega^\circ}].
\]
Then $\mu_u$ is a positive Radon measure and
\begin{equation}
\label{eq:p213-trivial-gauss-rigid-zero-atom}
                         \mu_u(\{y\})=0.
\end{equation}
The conclusion remains valid when $K$ is imperfect and when
$\mathscr H(y)/K$ is inseparable.
\end{lemma}

\begin{remark}
\label{rem:p213-trivial-gauss-rigid-zero-atom-index}
The proof does not use the printed exponent in
\cite[Corollary~16.8.6]{CLD25}.  With $p$ denoting the number of curvature
factors, that exponent must be $n-p$, not $p$; in particular, the corrected
corollary does not by itself exclude an atom at a point on a curve.
\end{remark}

\begin{lemma}
\label{lem:p213-variation-graph-source-cutoff}
Let $K$ be a complete non-Archimedean field, possibly trivially valued,
let $Y$ be a good separated normal boundaryless $K$-analytic space that is
purely one-dimensional, and let $y\in Y^{[2,3]}$.  Let
\[
                  F_0:W_0\longrightarrow\mathbf G_{m,K}^{n_0}
\]
be a moment on an open neighborhood of $y$.  After shrinking $W_0$, one
can append a finite block $S$ of analytic units and find an open
neighborhood $W$ of $y$ and a smooth function
\[
       0\leq\chi\leq1,\qquad
       \operatorname{supp}(\chi)\Subset W,\qquad
       \chi=1\text{ on a neighborhood of }y,
\]
with the following property.  Put $F=(F_0,S)$ and
$v=F_{\mathrm{trop}}(y)$.  There is an open neighborhood
$O\subset\mathbf R^N$ of $v$ such that, for every positive
$\mathrm d'$- and $\mathrm d''$-closed ordinary Chambert--Loir--Ducros
current $T$ of bidegree $(0,0)$ on $Y$, the compactly supported
pushforward
\begin{equation}
             R_{T,F,\chi}:=(F_{\mathrm{trop}})_*(\chi T)
\label{eq:p213-source-cutoff-pushforward}
\end{equation}
is positive on $\mathbf R^N$ and is $\mathrm d'$- and
$\mathrm d''$-closed on $O$.  Its support in $O$ is contained in a finite
rational fan with vertex $v$.

Every branch of $Y$ at $y$ on which at least one coordinate of $F_0$
varies maps to a ray germ distinct from the ray germ of every other
source branch.  Consequently the coefficient of
$R_{T,F,\chi}$ on such a ray is contributed by that source branch alone.
No target-only global tube and no assertion about the exact inverse image
of a ray is made.
\end{lemma}

\begin{lemma}
\label{lem:p213-euclidean-one-dimensional-current-revised}
Let $\Omega\subset\mathbf R^N$ be open, with its standard integral affine
structure, and let $\Sigma\subset\Omega$ be a locally finite embedded
rational polyhedral complex of pure dimension one.  Let $R$ be a positive
$\mathrm d'$- and $\mathrm d''$-closed supercurrent of bidimension $(1,1)$
on $\Omega$ whose support is contained in $\Sigma$.  After subdividing
$\Sigma$, there are constants $b_e\in\mathbf R_{\geq0}$, one for every
open edge $e$, such that
\begin{equation}
                         R=\sum_e b_e[e].
\label{eq:p213-euclidean-edge-decomposition-revised}
\end{equation}
The sum is locally finite and $[e]$ is the integration current calibrated
by the primitive integral tangent of $e$.  At every vertex $v\in\Omega$
one has
\begin{equation}
                    \sum_{e\ni v}b_eu_{e,v}=0,
\label{eq:p213-euclidean-vertex-balance-revised}
\end{equation}
where $u_{e,v}$ is the primitive outgoing tangent.  In particular, $R$
has no additional current supported at a vertex.
\end{lemma}

\begin{proposition}
\label{prop:p213-normal-curve-arbitrary-valuation-current-revised}
Let $K$ be a complete non-Archimedean field, possibly trivially valued,
and let $Y$ be a connected good separated normal boundaryless
$K$-analytic space that is purely one-dimensional.  Every positive
$\mathrm d'$- and $\mathrm d''$-closed ordinary Chambert--Loir--Ducros
current $T$ of bidegree $(0,0)$ on $Y$ is a nonnegative constant multiple
of the integration current:
\begin{equation}
                         T=c\,\delta_Y,
             \qquad c\in\mathbf R_{\geq0}.
\label{eq:p213-normal-curve-arbitrary-classification-revised}
\end{equation}
Consequently $T$ is a Bedford--Taylor current in the sense of
\cite[Definition~8.6.2]{CLD25}.
\end{proposition}

\begin{proposition}
\label{prop:p213-componentwise-arbitrary-valuation}
Let $K$ be a complete non-Archimedean field, possibly trivially valued,
and let $X$ be a good separated strict boundaryless $K$-analytic space,
purely of dimension one.  The space $X$ is not assumed reduced.  Write
\[
                 \iota:X_{\mathrm{red}}\hookrightarrow X
\]
for the canonical nilpotent closed immersion.  Let $(X_i)_i$ be the
irreducible components of $X_{\mathrm{red}}$, and write
\[
 \bar\nu_i:\widetilde X_i\longrightarrow X_i
       \hookrightarrow X_{\mathrm{red}},
 \qquad
 \nu_i:=\iota\circ\bar\nu_i,
 \qquad
 \delta_i:=(\nu_i)_*\delta_{\widetilde X_i}.
\]
Let $m_i\in\mathbf Z_{>0}$ be the coefficient of $[X_i]$ in the
fundamental cycle $[X]$.  No quasi-smoothness, rig-smoothness, or
exceptional-set hypothesis is imposed on $\widetilde X_i$.  Then the
following assertions hold.

\begin{enumerate}
\item\label{itm:p213-componentwise-av:calculus}
For every $(p,q)$, pullback by $\iota$ identifies the sheaves of ordinary
CLD-smooth forms on $X$ and $X_{\mathrm{red}}$, compatibly with wedge
product, $J$, $\mathrm d'$, $\mathrm d''$, support, and the three positivity
cones.  On compactly supported test forms this is an isomorphism of
topological vector spaces.  Its transpose identifies the CLD current
sheaves, their weak topologies, positivity, and
$\mathrm d'$- and $\mathrm d''$-closedness.  It identifies multiplication
by smooth functions for all currents.  For positive currents it also
identifies multiplication by locally bounded Borel functions, and by
linearity the same is true for currents locally expressible as differences
of positive currents.  No Borel product is asserted for an arbitrary
order-zero current.  Under the common underlying topological space, the
CLD-psh and continuous CLD-psh sheaves coincide.  The same pullback
identification holds for $G$-forms.

\item\label{itm:p213-componentwise-av:integration}
The integration currents retain the generic multiplicities:
\begin{equation}
 \label{eq:p213-componentwise-av-fundamental-cycles}
       \delta_X=\sum_i m_i\delta_i,
       \qquad
       \iota_*\delta_{X_{\mathrm{red}}}=\sum_i\delta_i.
\end{equation}
Both sums are locally finite.  Consequently
$\delta_X=\iota_*\delta_{X_{\mathrm{red}}}$ if and only if $X$ is
generically reduced along every $X_i$.

\item\label{itm:p213-componentwise-av:classification}
Every positive $\mathrm d'$- and $\mathrm d''$-closed ordinary CLD current
$T$ of bidegree $(0,0)$ on $X$ has a unique expression
\begin{equation}
 \label{eq:p213-componentwise-av-classification}
                         T=\sum_i c_i\delta_i,
                         \qquad c_i\in\mathbf R_{\geq0}.
\end{equation}
The sum is locally finite.

\item\label{itm:p213-componentwise-av:bedford-taylor}
Let $W\subseteq X$ be open, and let $U,V$ be continuous CLD-psh functions
on $W$.  Identify $W$ with its inverse image in $X_{\mathrm{red}}$, and,
for a continuous function $F$ on $W$, put
\[
 [F]_X:=F(\delta_X|_W),
 \qquad
 [F]_{X_{\mathrm{red}}}:=
      F(\delta_{X_{\mathrm{red}}}|_W).
\]
Then
\begin{equation}
 \label{eq:p213-componentwise-av-curvature-equivalence}
 \mathrm d'\mathrm d''[U]_X=\mathrm d'\mathrm d''[V]_X
 \quad\Longleftrightarrow\quad
 \mathrm d'\mathrm d''[U]_{X_{\mathrm{red}}}
  =\mathrm d'\mathrm d''[V]_{X_{\mathrm{red}}}.
\end{equation}
If these equivalent conditions hold, then every current in
\cref{itm:p213-componentwise-av:classification} satisfies
\begin{equation}
 \label{eq:p213-componentwise-av-bt-conclusion}
             \mathrm d'\mathrm d''\bigl((U-V)T|_W\bigr)=0.
\end{equation}
Thus every such current has Bedford--Taylor independence for continuous
psh pairs in the CLD sense.
\end{enumerate}
\end{proposition}
