\section*{Imported production-verified dependency}

\begin{proposition}
\label{prop:p21p3-gfc-smooth-rigidity}
Let $K$ be a complete nontrivially valued non-Archimedean field and let
$Y$ be a connected good separated strict rig-smooth $K$-analytic curve
without boundary.  Every positive $\mathrm d'$- and
$\mathrm d''$-closed CLD current $T$ of bidegree $(0,0)$ on $Y$ has the
form $T=c\delta_Y$ for a unique $c\in\mathbf R_{\geq0}$.
\end{proposition}
