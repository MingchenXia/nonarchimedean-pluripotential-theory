\begin{proposition}
\label{prop:p062-proper-full-rank-moment}
Let $K$ be a complete nontrivially valued non-Archimedean field, and let
$X$ be a good separated strict $K$-analytic space, pure of
dimension $n\geq 1$.  Let
\[
 f=(f_1,\ldots,f_n):X\longrightarrow(\mathbf G_{m,K}^{n})^{\mathrm{an}}
\]
be a pleasant moment in the sense of
\cite[Definition~14.1.7]{CLD25}.  Write
\[
 \tau=(\log|f_1|,\ldots,\log|f_n|),\qquad
 L=-\operatorname{id}_{\mathbf R^n},\qquad
 F=L\circ\tau=(-\log|f_1|,\ldots,-\log|f_n|).
\]
Put $\Sigma_F^{(n)}:=\Sigma_f^{(n)}$ and let
$\mu_{\Sigma_F^{(n)}}$ be its canonical calibration.  Define
\begin{equation}
\label{eq:p062-moment-explicit-pushforward-calibration}
 \Pi_F:=F(\Sigma_F^{(n)}),\qquad
 \mu_F:=
 (F|_{\Sigma_F^{(n)}})_*\mu_{\Sigma_F^{(n)}}.
\end{equation}
Put $e=|e_1\wedge\cdots\wedge e_n|$.  Assume that
\begin{enumerate}
\item\label{itm:p062-moment:image}
      \begin{equation}
      \label{eq:p062-moment-full-image}
      \Omega:=F(X)=\Pi_F=F(\Sigma_F^{(n)})
      \end{equation}
      is a nonempty connected open subset of $\mathbf R^n$;
\item\label{itm:p062-moment:proper}
      $F:X\to\Omega$ is proper in the following explicit sense: for every
      compact subset $C\subset\Omega$, the inverse image $F^{-1}(C)$ is
      compact in $X$;
\item\label{itm:p062-moment:calibration}
      there is a constant $m>0$ such that
      \begin{equation}
      \label{eq:p062-moment-constant-calibration}
       (F|_{\Sigma_F^{(n)}})_*\mu_{\Sigma_F^{(n)}}
       =\mu_F=m|e_1\wedge\cdots\wedge e_n|
       \quad\text{on }\Omega.
      \end{equation}
\end{enumerate}

Let $\varphi\in C^0(\Omega)$, put $h=\varphi\circ F$, and suppose that
\begin{equation}
\label{eq:p062-moment-zero-current}
                         \mathrm d'\mathrm d''[h]=0
\end{equation}
as a CLD current on $X$.  Then there are
$a_1,\ldots,a_n,b\in\mathbf R$ such that
\begin{equation}
\label{eq:p062-moment-affine-target}
                  \varphi(y)=b+\sum_{i=1}^n a_i y_i
                  \qquad(y\in\Omega),
\end{equation}
and hence
\begin{equation}
\label{eq:p062-moment-affine-source}
                  h=b+\sum_{i=1}^n a_i(-\log|f_i|).
\end{equation}
In particular, $h$ is ordinary CLD-smooth, $\mathbf R$-piecewise linear,
GJR harmonic, and classically pluriharmonic.

If $X$ is boundaryless, then the converse also holds: condition
\eqref{eq:p062-moment-zero-current} is equivalent to
\eqref{eq:p062-moment-affine-target}.
\end{proposition}

\begin{corollary}
\label{cor:p062-proper-full-rank-moment-automatic-calibration}
Retain all the hypotheses and notation of
\cref{prop:p062-proper-full-rank-moment}, except that
\cref{itm:p062-moment:calibration} is not assumed, and assume in addition
that $X$ is boundaryless.  Then the canonical
calibration $\mu_F$ has the form
\eqref{eq:p062-moment-constant-calibration} for a constant $m>0$.
Consequently, the conclusions
\eqref{eq:p062-moment-affine-target}--\eqref{eq:p062-moment-affine-source}
hold, and \eqref{eq:p062-moment-zero-current} is equivalent to
\eqref{eq:p062-moment-affine-target}.
\end{corollary}

\begin{remark}
\label{rem:p062-proper-full-rank-moment-scope}
The proposition concerns only continuous functions that factor through one
proper full-rank moment with full open calibrated image.  It does not assert
that an arbitrary continuous function with zero CLD
$\mathrm d'\mathrm d''$-current admits such a factorization, and it does not
resolve the unrestricted higher-dimensional problem.  If $\Omega$ is
disconnected, the same argument gives a possibly different affine formula
on each connected component, but not necessarily one formula on all of
$\Omega$.
\end{remark}
