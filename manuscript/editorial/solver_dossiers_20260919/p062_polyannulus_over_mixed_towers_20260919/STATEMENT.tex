\begin{corollary}
\label{cor:p062-open-polyannulus-over-mixed-bundle-towers}
Let $K$, $C$, and $W\subset C^{\mathrm{an}}$, and let
\[
 B_\ell\xrightarrow{\pi_\ell}B_{\ell-1}
 \longrightarrow\cdots\longrightarrow B_1
 \xrightarrow{\pi_1}B_0=W,
\]
be as in \Cref{thm:p062-mixed-bundle-towers-over-curves}.  Put
$\Pi=\pi_1\circ\cdots\circ\pi_\ell$, and assume in addition that
$B_\ell$ is boundaryless.  For $s\geq1$, let
\[
 A_i=\{z\in(\mathbf G_{\mathrm m,K})^{\mathrm{an}}:
                    r_i<|T_i(z)|<R_i\},
 \qquad 0<r_i<R_i,\quad r_i,R_i\in|K^\times|,
\]
and put
\[
 X=B_\ell\times_K\prod_{i=1}^sA_i,
 \qquad p:X\longrightarrow B_\ell,
 \qquad q_i:X\longrightarrow A_i,
 \qquad \overline\Pi=\Pi\circ p.
\]
If $h\in C^0(X,\mathbf R)$ satisfies
$\mathrm d'\mathrm d''[h]=0$, then there are unique continuous functions
$b,a_1,\ldots,a_s:W\to\mathbf R$ such that
\[
 h=\overline\Pi^*b+
   \sum_{i=1}^s(a_i\circ\overline\Pi)q_i^*\log|T_i|,
 \qquad \mathrm d'\mathrm d''[b]=0,
\]
and every $a_i$ is locally constant.  In particular, $h$ is classically
pluriharmonic, locally $\mathbf R$-PL, and Gubler--Jell--Rabinoff harmonic.
\end{corollary}
