% Book-local analytic P06.1 product consequence, 19 September 2026.
% This combines algebraic-split analytic toric charts with the arbitrary
% curve-factor product corollary.  It makes no novelty or priority claim.

\begin{corollary}
\label{cor:p061-analytic-toric-arbitrary-curve-factor-products}
Let $K$ be a complete non-Archimedean field with nontrivial absolute
value, let $m\geq0$, let $C_1,\ldots,C_m$ be arbitrary good strict
$K$-analytic curves, and let $Y$ be a good strict $K$-analytic space whose
reduction $Y_{\mathrm{red}}$ is quasi-smooth.  Let $X$ be a good strict
$K$-analytic space such that every point of $X$ has an ordinary open
neighbourhood $U$ with the following property: there are a complete
analytic extension $L/K$, obtained by completing an algebraic extension of
$K$, a split toric $L$-variety $\mathcal X$, and an $L$-analytic
isomorphism from $U_L$ onto an ordinary open analytic subspace of
$\mathcal X^{\mathrm{an}}$.

Assume that
\[
 X\times_K C_1\times_K\cdots\times_K C_m\times_KY
\]
is good.  Every classically pluriharmonic function on an ordinary open
analytic subspace of this product is $\mathbf R$-PL and harmonic in the
sense of Gubler--Jell--Rabinoff.  The same conclusion holds on every good
strict analytic space analytically locally isomorphic to such an open
subspace.

This does not construct the displayed splitting charts, treat charts that
split only over a nonalgebraic complete extension, or cover a nonproduct
singular germ.
\end{corollary}
