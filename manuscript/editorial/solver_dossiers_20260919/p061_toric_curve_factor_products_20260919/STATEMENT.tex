% Book-local P06.1 product consequence, 19 September 2026.
% This combines the all-orbits toric factorization with the arbitrary
% curve-factor theorem.  It makes no novelty or priority claim.

\begin{corollary}
\label{cor:p061-toric-arbitrary-curve-factor-products}
Let $K$ be a complete non-Archimedean field with nontrivial absolute
value, let $m\geq0$, let $C_1,\ldots,C_m$ be arbitrary good strict
$K$-analytic curves, and let $Y$ be a good strict $K$-analytic space whose
reduction $Y_{\mathrm{red}}$ is quasi-smooth.  Let $\mathcal X$ be a split
toric $K$-variety and assume that
\[
 Z=\mathcal X^{\mathrm{an}}\times_K C_1\times_K\cdots
      \times_K C_m\times_KY
\]
is good.  Every classically pluriharmonic function on an ordinary open
analytic subspace of $Z$ is $\mathbf R$-PL and harmonic in the sense of
Gubler--Jell--Rabinoff.  The same conclusion holds on every good strict
analytic space which is analytically locally isomorphic to such an open
subspace.

More generally, let $X$ be a $K$-variety for which there is a complete
analytic extension $L/K$, obtained by completing an algebraic extension of
$K$, such that $X_L$ is a split toric $L$-variety.  If the corresponding
product
\[
 X^{\mathrm{an}}\times_K C_1\times_K\cdots
      \times_K C_m\times_KY
\]
is good, the same two conclusions hold with $X$ in place of $\mathcal X$.

This does not cover a toric form without such an algebraic splitting
extension, a reduced singular factor of dimension at least two outside the
toric factor, or a nonproduct singular germ.
\end{corollary}
