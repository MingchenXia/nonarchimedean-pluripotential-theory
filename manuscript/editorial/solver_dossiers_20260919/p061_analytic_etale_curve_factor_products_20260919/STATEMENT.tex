% 19 September 2026.  No novelty or priority claim is made.

\begin{corollary}
\label{cor:p061-analytic-standard-etale-arbitrary-curve-factor-products}
Let $K$ be a complete non-Archimedean field with nontrivial absolute value,
let $m\geq0$, let $C_1,\ldots,C_m$ be arbitrary good strict $K$-analytic
curves, and let $Y$ be a good strict $K$-analytic space whose reduction
$Y_{\mathrm{red}}$ is quasi-smooth.  Let $X$ be a good strict $K$-analytic
space.  Suppose that every point of $X$ has an ordinary open neighbourhood
analytically isomorphic to an ordinary open subspace of a strict affinoid
space $\mathcal M(C)$ with the following presentation.  There are a
$K$-variety $\mathcal Z$, a complete analytic extension $L/K$ obtained by
completing an algebraic extension and such that $\mathcal Z_L$ is split
toric, a good strict quasi-smooth $K$-analytic space $Y_0$, a strict affinoid
domain $W=\mathcal M(A_0)$ in
$\mathcal Z^{\mathrm{an}}\times_KY_0$ which is an ordinary neighbourhood of
the image of the chosen source point, and
\[
 C=A_0\langle T,U\rangle/\bigl(g(T),h(T)U-1\bigr),
\]
where $g,h\in A_0[T]$, the polynomial $g$ is monic, and $g'(T)$ is
invertible in $C$.  Assume that
\[
 X\times_KC_1\times_K\cdots\times_KC_m\times_KY
\]
is good.

Every classically pluriharmonic function on an ordinary open analytic
subspace of this product is $\mathbf R$-PL and harmonic in the sense of
Gubler--Jell--Rabinoff.  The same conclusion holds on every good strict
analytic space analytically locally isomorphic to such an open subspace.

This does not assert that an arbitrary analytic etale map admits the
displayed standard-etale chart, treat a toric factor without the stated
algebraic splitting extension, or cover a locally twisted nonproduct germ.
\end{corollary}
