% Book-local P06.1 ordinary-etale product consequence, 19 September 2026.
% This carries arbitrary curve factors through the manufactured transverse
% toric tube.  It makes no novelty or priority claim.

\begin{corollary}
\label{cor:p061-etale-toric-arbitrary-curve-factor-products}
Let $K$ be a complete non-Archimedean field with nontrivial absolute
value, let $m\geq0$, let $C_1,\ldots,C_m$ be arbitrary good strict
$K$-analytic curves, and let $Y$ be a good strict $K$-analytic space whose
reduction $Y_{\mathrm{red}}$ is quasi-smooth.  Let $\mathcal X$ be a
separated finite-type $K$-scheme admitting, Zariski locally on the source,
ordinary etale charts
\[
 q:\mathcal U\longrightarrow\mathcal Z\times_K\mathcal Y_0,
\]
where $\mathcal Y_0$ is a smooth $K$-variety and, for every chart, there is
a complete analytic extension $L/K$, obtained by completing an algebraic
extension of $K$, such that $\mathcal Z_L$ is a split toric $L$-variety.
Assume that
\[
 \mathcal X^{\mathrm{an}}\times_K C_1\times_K\cdots
       \times_K C_m\times_KY
\]
is good.

Every classically pluriharmonic function on an ordinary open analytic
subspace of this product is $\mathbf R$-PL and harmonic in the sense of
Gubler--Jell--Rabinoff.  The same conclusion holds on every good strict
analytic space analytically locally isomorphic to such an open subspace.

This does not treat a genuinely log-etale chart outside the displayed
ordinary-etale scope, a toric factor without the stated algebraic splitting
extension, or a locally twisted nonproduct germ.
\end{corollary}
