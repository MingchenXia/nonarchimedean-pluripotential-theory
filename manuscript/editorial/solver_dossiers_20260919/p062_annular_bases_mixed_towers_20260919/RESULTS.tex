\begin{corollary}
\label{cor:p062-polyannulus-base-mixed-bundle-towers}
Let $K$, $C$, $W$, $s\geq1$, and $A_1,\ldots,A_s$ be as in
\Cref{thm:p062-open-polyannulus-products-over-curves}.  Put
\[
 B_0=W\times_K\prod_{i=1}^sA_i,
 \qquad p:B_0\longrightarrow W,
 \qquad q_i:B_0\longrightarrow A_i.
\]
Let
\[
 B_\ell\xrightarrow{\pi_\ell}B_{\ell-1}
 \longrightarrow\cdots\longrightarrow B_1
 \xrightarrow{\pi_1}B_0
\]
be a tower satisfying the stagewise hypotheses of
\Cref{prop:p062-mixed-exact-bundle-tower-transfer}, and set
$\Pi=\pi_1\circ\cdots\circ\pi_\ell$.  If
\[
 h\in C^0(B_\ell,\mathbf R),
 \qquad \mathrm d'\mathrm d''[h]=0,
\]
then there are unique continuous functions
$a_1,\ldots,a_s,b:W\to\mathbf R$ such that every $a_i$ is locally
constant, $\mathrm d'\mathrm d''[b]=0$, and
\[
 h=\Pi^*\left(p^*b+
       \sum_{i=1}^s(a_i\circ p)q_i^*\log|T_i|\right).
\]
In particular, $h$ is classically pluriharmonic, locally
$\mathbf R$-PL, and Gubler--Jell--Rabinoff harmonic.
\end{corollary}

\begin{proof}
By \Cref{prop:p062-mixed-exact-bundle-tower-transfer}, there is a unique
continuous zero-current function $g$ on $B_0$ with $h=\Pi^*g$.
Apply \Cref{thm:p062-open-polyannulus-products-over-curves} to $g$.
Its unique coefficient functions give the displayed formula, including
their zero-current and local-constancy properties; uniqueness of the formula
follows from the two cited uniqueness assertions.

The same product theorem makes $g$ classically pluriharmonic and
$\mathbf R$-harmonic.  The first two regularity-transfer assertions of
\Cref{prop:p062-mixed-exact-bundle-tower-transfer} therefore make $h$
classically pluriharmonic and $\mathbf R$-harmonic.  The latter is locally
$\mathbf R$-PL and Gubler--Jell--Rabinoff harmonic.
\end{proof}

\begin{corollary}
\label{cor:p062-open-polyannulus-bundle-mixed-bundle-towers}
Let $K$, $C$, $W$, $\pi_0:Y_0\to W$, and $s\geq1$ be as in
\Cref{cor:p062-open-polyannulus-bundles-over-curves}.  Let
\[
 B_\ell\xrightarrow{\pi_\ell}B_{\ell-1}
 \longrightarrow\cdots\longrightarrow B_1
 \xrightarrow{\pi_1}B_0=Y_0
\]
be a tower satisfying the stagewise hypotheses of
\Cref{prop:p062-mixed-exact-bundle-tower-transfer}.  If
\[
 h\in C^0(B_\ell,\mathbf R),
 \qquad \mathrm d'\mathrm d''[h]=0,
\]
then $h$ is classically pluriharmonic, locally $\mathbf R$-PL, and
Gubler--Jell--Rabinoff harmonic.
\end{corollary}

\begin{proof}
By \Cref{cor:p062-open-polyannulus-bundles-over-curves}, every continuous
zero-current function on $B_0$ is classically pluriharmonic and
$\mathbf R$-harmonic.  The first two regularity-transfer assertions of
\Cref{prop:p062-mixed-exact-bundle-tower-transfer} give the three stated
conclusions on $B_\ell$.
\end{proof}

\begin{proof}

$\mathbf R$-harmonic.  Apply the first two regularity-transfer assertions
of \Cref{prop:p062-mixed-exact-bundle-tower-transfer}.
\end{proof}
