% Book-local P07.1 theorem, 19 September 2026.
% This theorem combines the locally constant Stone parameter with the exact
% infinite CLD star and one active convex-polyhedral factor.  It makes no
% novelty or priority claim.

\begin{theorem}
\label{thm:p071-stone-infinite-star-polyhedral-collapse}
Let $Z$ be a locally compact Hausdorff space with a basis of clopen
subsets, let $B\ne\varnothing$, let $X_B$ be the exact
Chambert--Loir--Ducros star of
\Cref{thm:p071-infinite-star-collapse-candidate}, and let $P$ be a
nonempty convex polyhedron in a finite-dimensional real affine space.
Put
\[
                         T=X_B\times P,
             \qquad      X=Z\times T.
\]
On $T$, retain the exact product generator sheaf $\mathcal G_{T,0}$ and
the comparison sheaf $\mathcal J_T$ of
\Cref{thm:p071-infinite-star-polyhedral-collapse}.

For an ordinary open $W\subset X$, let $\mathcal G_{X,0}(W)$ be the
real-valued functions which locally, after shrinking the $Z$-variable to
a clopen neighbourhood, are independent of that variable and belong to
$\mathcal G_{T,0}$ in the remaining variables.  Let $\mathcal P_{X,1}$
be the sheaf of local pointwise decreasing limits of sequences in
$\mathcal G_{X,0}$, and let $\mathcal P_X$ be the smallest subsheaf of
$\mathbf R\cup\{-\infty\}$-valued functions which contains
$\mathcal G_{X,0}$ and is closed under such limits.

Define a comparison sheaf $\mathcal C_X$ by the following local
condition.  A section $u$ belongs to $\mathcal C_X$ near
$(z_0,x_0)\in Z\times T$ if there are a compact clopen neighbourhood
$C\subset Z$ of $z_0$, a product neighbourhood $V_T\subset T$ of
$x_0$, a compact metrizable zero-dimensional space $Y$, a continuous
surjection $q:C\twoheadrightarrow Y$, and an upper semicontinuous function
\[
 \bar u:Y\times V_T\longrightarrow\mathbf R\cup\{-\infty\}
\]
such that
\begin{equation}
 u(z,x)=\bar u(q(z),x),
 \qquad
 \bar u(\zeta,\cdot)\in\mathcal J_T(V_T)
 \quad(\zeta\in Y).
 \label{eq:p071-stone-infinite-polyhedral-factorization}
\end{equation}
Near a node point $(z_0,\eta,y_0)$, require in addition that
$V_T=U\times V$, where $U$ is a standard star neighbourhood with finite
exceptional set $A\subset B$ and $V$ is relatively open and convex in
$P$, and that there be one countable set $K\subset B\setminus A$,
independent of $\zeta$ and $y$, for which
\begin{equation}
 \bar u(\zeta,p_b(t),y)=\bar u(\zeta,\eta,y)
 \quad
 \bigl(\zeta\in Y,\ b\in B\setminus(A\cup K),\
       t\in[-\infty,0],\ y\in V\bigr).
 \label{eq:p071-stone-infinite-polyhedral-common-activity}
\end{equation}

Then $\mathcal C_X$ is a sheaf and
\begin{equation}
                 \mathcal P_{X,1}=\mathcal P_X=\mathcal C_X.
 \label{eq:p071-stone-infinite-polyhedral-collapse}
\end{equation}
More precisely, every germ in these sheaves has a pointwise decreasing
approximation by members of $\mathcal G_{X,0}$ on one fixed smaller
product neighbourhood.  At a node, that neighbourhood may be chosen as
$C\times U'\times V'$, with $C$ compact clopen, $U'$ standard, and
$V'$ relatively open and convex in $P$.

Thus a possibly nonmetrizable locally constant Stone parameter, an
arbitrarily large exact CLD branch set, and all affine directions, faces,
and corners of one convex polyhedron may occur simultaneously.  The
theorem still excludes nonconstant affine directions on the Stone factor,
non-CLD infinite stars, separately convex generators, nonambient branch
structures, and nonconvex or branched chamber gluings.  Consequently it
narrows but does not retire P07.1.
\end{theorem}
