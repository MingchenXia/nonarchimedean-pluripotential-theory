\begin{proposition}[Invariant toric continuous-pair independence, including boundary]
\label{prop:p213-invariant-toric-boundary-continuous-pairs-candidate}
Let $K$ be a complete non-Archimedean field, let $N$ be a lattice, let
$\Sigma$ be a rational fan in $N_{\mathbb R}$, and let $X_\Sigma$ be the
associated split toric $K$-variety.  Write
\[
 \operatorname{trop}:X_\Sigma^{\An}\longrightarrow N_\Sigma
\]
for its extended tropicalization.  Let $O\subseteq N_\Sigma$ be open, set
$V=\operatorname{trop}^{-1}(O)$, and let $S$ be the maximal compact torus.
Let $T$ be a positive $\dprime$- and $\ddprime$-closed CLD current of
bidegree $(p,p)$ on $V$, where $0\leq p\leq \dim X_\Sigma-1$.  If $u,v$ are
finite, continuous, $S$-invariant CLD-psh functions on $V$ satisfying
\[
 \dprimeddprime[u]=\dprimeddprime[v],
\]
then
\[
 \dprimeddprime(uT)=\dprimeddprime(vT).
\]
Consequently, every positive closed CLD current on $V$ has the
continuous-pair Bedford--Taylor independence property for finite,
$S$-invariant continuous psh pairs.
\end{proposition}
