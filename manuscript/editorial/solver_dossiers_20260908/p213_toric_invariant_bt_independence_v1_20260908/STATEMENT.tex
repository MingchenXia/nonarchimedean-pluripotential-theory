\begin{proposition}[Invariant split-toric continuous-pair independence]
\label{prop:p213-invariant-split-toric-continuous-pairs-candidate}
Let $K$ be a complete non-Archimedean field, let $\mathbb T$ be a split
$K$-torus of dimension $n$ with cocharacter lattice $N$, and put
\[
 \operatorname{trop}:\mathbb T^{\An}\longrightarrow N_{\mathbb R}.
\]
Let $O\subseteq N_{\mathbb R}$ be open and set
$V=\operatorname{trop}^{-1}(O)$.  Let $S$ be the maximal compact torus of
$\mathbb T^{\An}$, and let $T$ be a positive $\dprime$- and
$\ddprime$-closed CLD current of bidegree $(p,p)$ on $V$, where
$0\leq p\leq n-1$.  If $u,v$ are $S$-invariant continuous CLD-psh functions
on $V$ satisfying
\[
 \dprimeddprime[u]=\dprimeddprime[v],
\]
then
\[
 \dprimeddprime(uT)=\dprimeddprime(vT).
\]
Consequently, every positive closed CLD current on $V$ has the
continuous-pair Bedford--Taylor independence property for $S$-invariant
continuous psh pairs.
\end{proposition}
