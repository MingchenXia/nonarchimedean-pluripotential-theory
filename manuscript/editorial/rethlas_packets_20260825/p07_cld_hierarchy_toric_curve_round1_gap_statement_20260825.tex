\begin{theorem}
The following two assertions hold.

\begin{enumerate}
\item Let $N$ be a lattice, let $\Sigma$ be a rational polyhedral fan in
$N_{\mathbf R}$, and equip the tropical toric partial compactification
$N_\Sigma$ with its standard Lagerberg smooth structure.  On every open
subset of $N_\Sigma$ one has
\[
 \operatorname{Psh}^{\mathrm{CLD}}_{N_\Sigma,1}
 =\operatorname{Psh}^{\mathrm{CLD}}_{N_\Sigma}
 =\operatorname{Psh}^{\mathrm{tor}}_{N_\Sigma}.
\]

\item Let $X$ be a one-dimensional boundaryless affine tropical space.
Assume that every point has a neighbourhood isomorphic, as a calibrated
affine tropical space with smooth structure, to an open subset of the
support of a finite balanced one-dimensional polyhedral complex in some
$\mathbf R^N$, with the smooth structure induced by ambient smooth
functions.  Then
\[
                  \operatorname{Psh}_{X,1}
                  =\operatorname{Psh}_X.
\]
More precisely, every CLD-psh germ is represented on one fixed open
neighbourhood by the pointwise decreasing limit of a sequence of smooth
CLD-psh functions defined on that same neighbourhood.
\end{enumerate}
\end{theorem}
