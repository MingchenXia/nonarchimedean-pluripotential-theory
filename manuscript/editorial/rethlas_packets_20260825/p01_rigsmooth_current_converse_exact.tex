\paragraph{Imported inputs.}
We use the following results with their stated hypotheses.
\begin{enumerate}
\item Chambert--Loir--Ducros, Proposition~8.7.2: joint
Bedford--Taylor continuity under locally uniform convergence, including a
continuous outer factor.
\item Chambert--Loir--Ducros, Corollary~18.6.2: if
\[
 u_r=\log\max\{r,|f_1|,\ldots,|f_d|\},
\]
then
\[
 (\mathrm d'\mathrm d''u_r)^d=\delta_{f^{-1}(\eta_r)}
\]
on a boundaryless separated equidimensional analytic space.
\item Chambert--Loir--Ducros, Proposition~18.7.3, with the
proof-correct relative dimension $n-d$: if
$f=(f_1,\ldots,f_d):X\to\mathbf A^d$ is purely of relative dimension
$n-d$, the $f_i$ form a regular sequence, and $Y=f^{-1}(0)$, then for every
compactly supported $G$-form $\omega$ of bidegree $(n-d,n-d)$ with locally
bounded tropically Borel coefficients,
\[
 \langle\delta_{f^{-1}(t)},\omega\rangle
 =\langle\delta_Y,\omega\rangle
\]
for $t$ sufficiently close to the origin.
\item Chambert--Loir--Ducros, formula~(11.5.5.1): for a good germ,
\[
 \dim_{\mathrm{cent}}(X,z)+\dim_{\mathrm{Krull}}\mathcal O_{X,z}
 =\dim_zX.
\]
We also use the standard local property that if $X$ is rig-smooth at $z$,
then $\mathcal O_{X,z}$ is a regular local ring.  More precisely, Ducros,
\emph{Families of Berkovich spaces}, Corollary~5.3.5 identifies
quasi-smoothness with geometric regularity; ordinary regularity follows.
\item Gubler--Rabinoff, Theorem~4.1, Remark~4.2, Section~5.2, and the
proof of Theorem~5.3: a proper residual curve on which the residue line
bundle has negative degree admits a local irreducible analytic curve lift
through an interior point $z$; on the formal closure of that lift, the
residual component through $\operatorname{red}(z)$ is proper and maps
finitely and surjectively to the original residual curve.  The projection
formula multiplies its degree by a strictly positive generic degree.
\item The curve model slope formula used in Gubler--Rabinoff,
Proposition~5.1: the coefficient of
$\mathrm d'\mathrm d''[h|_C]$ at the component point is the degree of the
residual model line bundle, with the positive analytic multiplicity dictated
by the model cycle.
\end{enumerate}

\begin{proposition}
\label{prop:p01-lci-atom-slice-candidate}
Let $K$ be a complete nontrivially valued non-Archimedean field.  Let $X$ be
a good separated boundaryless strictly $K$-analytic space, pure of dimension
$n\geq2$, and let $h:X\to\mathbf R$ be a continuous $\mathbf R$-PL
function such that
\[
                         \mathrm d'\mathrm d''[h]\geq0.
\]
Let $V\subset X$ be an open analytic subspace and let
$f_1,\ldots,f_{n-1}\in\mathcal O_X(V)$ be a regular sequence.  Suppose that
$f=(f_1,\ldots,f_{n-1}):V\to\mathbf A^{n-1}$ is purely of relative
dimension one, and put $C=f^{-1}(0)$, with its analytic cycle
multiplicities.

If $\psi$ is a nonnegative smooth compactly supported function on $V$, then
\[
 \bigl\langle\mathrm d'\mathrm d''[h|_C],\psi|_C\bigr\rangle\geq0.
\]
In particular, if $\mathrm d'\mathrm d''[h|_C]$ is a discrete signed model
measure near $z\in C$ and there is such a $\psi$ which equals one near $z$
and whose support meets no other atom, then the coefficient at $z$ is
nonnegative.
\end{proposition}

\begin{proof}
Set $d=n-1$.  For $r>0$, put
\[
 u_r=\log\max\{r,|f_1|,\ldots,|f_d|\}.
\]
Let $M_\varepsilon$ be a smooth convex regularized maximum which is
nondecreasing in every variable, satisfies
\[
 \max_i s_i\leq M_\varepsilon(s)\leq\max_i s_i+\varepsilon,
\]
and does not depend on an entry lying at least $2\varepsilon$ below the
maximum.  Define locally
\[
 u_{r,\varepsilon}
 =M_\varepsilon(\log r,\log|f_1|,\ldots,\log|f_d|).
\]
The last property deletes every logarithm which is singular at the point
under consideration.  Hence $u_{r,\varepsilon}$ is a globally defined smooth
psh function on $V$, and
\[
 u_r\leq u_{r,\varepsilon}\leq u_r+\varepsilon.
\]
In particular, $u_{r,\varepsilon}\to u_r$ uniformly.

The form
\[
 \psi(\mathrm d'\mathrm d''u_{r,\varepsilon})^d
\]
is a smooth positive compactly supported $(d,d)$-form.  Ambient positivity
therefore gives
\begin{equation}
\label{eq:p01-positive-approximating-slice}
0\leq
\left\langle\mathrm d'\mathrm d''[h],
 \psi(\mathrm d'\mathrm d''u_{r,\varepsilon})^d\right\rangle.
\end{equation}
Since every factor $\mathrm d'\mathrm d''u_{r,\varepsilon}$ is
$\mathrm d'$- and $\mathrm d''$-closed, the definition of the differential
of a current and the Leibniz rule give
\begin{align}
\left\langle\mathrm d'\mathrm d''[h],
 \psi(\mathrm d'\mathrm d''u_{r,\varepsilon})^d\right\rangle
&=
\int_V h\,\mathrm d'\mathrm d''\!\left(
 \psi(\mathrm d'\mathrm d''u_{r,\varepsilon})^d\right)\\
&=
\int_V h\,\mathrm d'\mathrm d''\psi\wedge
 (\mathrm d'\mathrm d''u_{r,\varepsilon})^d.
\label{eq:p01-integration-by-parts-slice}
\end{align}

Apply Chambert--Loir--Ducros, Proposition~8.7.2 with constant current
$T=\delta_V$, with all $d$ psh factors equal to
$u_{r,\varepsilon}$, and with the continuous outer factor $h$.  Letting
$\varepsilon\to0$ in \eqref{eq:p01-integration-by-parts-slice} yields
\[
\lim_{\varepsilon\to0}
\int_V h\,\mathrm d'\mathrm d''\psi\wedge
 (\mathrm d'\mathrm d''u_{r,\varepsilon})^d
=
\int_V h\,\mathrm d'\mathrm d''\psi\wedge
 (\mathrm d'\mathrm d''u_r)^d.
\]
By Chambert--Loir--Ducros, Corollary~18.6.2, the last expression is
\[
 \left\langle\delta_{f^{-1}(\eta_r)},
 h\,\mathrm d'\mathrm d''\psi\right\rangle.
\]
The form
$h\,\mathrm d'\mathrm d''\psi$ is a compactly supported $G$-form of
bidegree $(1,1)$ with locally bounded tropically Borel coefficients: on a
finite $G$-cover of its support, the function $h$ is the pullback of a
continuous piecewise-affine function.  Chambert--Loir--Ducros,
Proposition~18.7.3 therefore gives, for all sufficiently small $r>0$,
\begin{align*}
 \left\langle\delta_{f^{-1}(\eta_r)},
 h\,\mathrm d'\mathrm d''\psi\right\rangle
 &=
 \left\langle\delta_C,h\,\mathrm d'\mathrm d''\psi\right\rangle\\
 &=
 \int_C h|_C\,\mathrm d'\mathrm d''(\psi|_C)\\
 &=
 \left\langle\mathrm d'\mathrm d''[h|_C],\psi|_C\right\rangle.
\end{align*}
Taking the limit in \eqref{eq:p01-positive-approximating-slice} proves the
first assertion.

For the last assertion, choose the stipulated cutoff.  The displayed
pairing is then exactly the coefficient of the discrete measure at $z$.
\end{proof}

\begin{theorem}
\label{thm:p01-rigsmooth-current-converse-candidate}
Let $K$ be a complete nontrivially valued non-Archimedean field, let $X$ be
a good separated boundaryless rig-smooth strictly $K$-analytic space, and
let $h:X\to\mathbf R$ be a real-valued $\mathbf R$-PL function.  If
\[
                         \mathrm d'\mathrm d''[h]\geq0,
\]
then $h$ is residue-nef, equivalently model semipositive, on $X$.
Consequently, on this class of spaces, an $\mathbf R$-PL function is model
semipositive if and only if its Chambert--Loir--Ducros current is positive.
\end{theorem}

\begin{proof}
The assertion is local, so we may work on a pure $n$-dimensional component.
For $n=1$, the claim is the curve model slope formula: positivity of the
discrete measure gives nonnegative degree on every residual component.
Assume $n\geq2$.

Suppose for contradiction that $h$ is not semipositive at a point $x$.
Choose the strictly affinoid neighbourhood, formal model, and proper
residual curve $\Gamma$ of negative degree which occur in the proof of
Gubler--Rabinoff, Theorem~5.3.  Their local lifting theorem supplies an
interior point $z$ reducing to the generic point of $\Gamma$, an analytic
neighbourhood $U$ of $z$, and an irreducible one-dimensional Zariski-closed
analytic subspace $C\subset U$ containing $z$.

The inclusion of the function field of $\Gamma$ into the graded residue
field at $z$ gives $d_K(z)\geq1$.  On the other hand, Gubler--Rabinoff,
Remark~4.2 applied to $C$ gives $d_K(z)\leq\dim C=1$.  Thus $d_K(z)=1$.
The central dimension is at least $d_K(z)$, while the curve germ through
$z$ shows that it is at most one.  Hence
\[
                 d_K(z)=\dim_{\mathrm{cent}}(X,z)=1.
\]
Since $X$ is rig-smooth, $\mathcal O_{X,z}$ is a regular local ring; by
Chambert--Loir--Ducros, formula~(11.5.5.1), its Krull dimension is $n-1$.
The same formula on the irreducible curve $C$ shows that
$\mathcal O_{C,z}$ has Krull dimension zero.  It is a zero-dimensional local
domain and hence a field.  Therefore the ideal of $C$ at $z$ is the maximal
ideal of $\mathcal O_{X,z}$.

Choose a regular system of parameters
$f_1,\ldots,f_{n-1}$ generating this maximal ideal.  By coherence and after
shrinking around $z$, they are analytic functions forming a regular
sequence and their common zero locus is $C$.  Shrinking once more, the map
$f=(f_1,\ldots,f_{n-1})$ is purely of relative dimension one.  Apply
\Cref{prop:p01-lci-atom-slice-candidate} on a boundaryless open
neighbourhood of $z$.

On the formal closure of $C$ used in the proof of Gubler--Rabinoff,
Theorem~5.3, the closure of $\operatorname{red}(z)$ is a proper residual
curve $\Gamma'$ mapping finitely and surjectively to $\Gamma$.  The model
slope formula identifies the coefficient of
$\mathrm d'\mathrm d''[h|_C]$ at $z$ with
\[
  m_z\deg\!\left(\mathcal L|_{\Gamma'}\right),
\]
where $m_z>0$ is the analytic cycle multiplicity.  The projection formula
gives
\[
 \deg\!\left(\mathcal L|_{\Gamma'}\right)
 =\deg(\Gamma'/\Gamma)\deg\!\left(\mathcal L|_\Gamma\right)<0.
\]
Thus this coefficient is negative.

The curve-model measure is discrete near $z$.  Choose a nonnegative smooth
ambient cutoff equal to one near $z$ whose compact support avoids all other
atoms.  The last assertion of
\Cref{prop:p01-lci-atom-slice-candidate} says that the same coefficient is
nonnegative, a contradiction.  Hence no negative residual curve exists and
$h$ is residue-nef.

The reverse implication in the final sentence is Gubler--Jell--Rabinoff,
Theorem~7.14.
\end{proof}

\paragraph{Scope warning.}
The argument does not prove the theorem for a singular ambient space.  At a
central-dimension-one point of a pure $n$-fold, the ordinary local ring has
Krull dimension $n-1$, but it need not contain a regular sequence of that
length.  Replacing the complete-intersection slice by arbitrary equations
requires a positive excess-intersection formula which is not among the
imported inputs.
