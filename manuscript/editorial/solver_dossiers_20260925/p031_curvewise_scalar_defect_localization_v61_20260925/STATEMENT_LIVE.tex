% P03.1 curvewise scalar-defect localization, live integration.

\begin{theorem}
\label{thm:p031-curvewise-scalar-defect-localization}
Let $K$ be an algebraically closed complete nontrivially valued
non-Archimedean field, let
$V\subseteq\mathbf A_K^{n,\mathrm{an}}$ be an ordinary open subset, and
let $(u_i)_{i\in I}$ be a nonempty locally upper-bounded family of
classically plurisubharmonic functions on $V$.  Put
\[
                         s:=\sup_{i\in I}u_i,
                    \qquad u:=s^*.
\]
Fix an analytic extension $K'/K$.  Write
\[
 \pi:=\pi_{K'/K}:V_{K'}\longrightarrow V,
 \qquad r:=\sigma_{K'/K}\circ\pi,
 \qquad v:=(s\circ\pi)^*.
\]
Let $C$ be a separated smooth connected strict $K'$-analytic curve and
let $g:C\to V_{K'}$ be a $K'$-analytic morphism.  Define
\[
 p_g:=(s\circ\pi\circ g)^*,
 \qquad h_g:=v\circ g,
 \qquad q_g:=u\circ\pi\circ g=v\circ r\circ g,
\]
and the two strict loci
\[
 \mathcal A_g:=\{c\in C:p_g(c)<h_g(c)\},
 \qquad
 \mathcal B_g:=\{c\in C:h_g(c)<q_g(c)\}.
\]
Then $p_g$ and $h_g$ are subharmonic on $C$, and
\begin{equation}
\label{eq:p031-curvewise-scalar-defect-chain}
                              p_g\leq h_g\leq q_g.
\end{equation}
Consequently,
\begin{equation}
\label{eq:p031-curvewise-total-defect-splitting}
                  \{p_g<q_g\}=\mathcal A_g\cup\mathcal B_g.
\end{equation}
Moreover, $q_g=h_g$ and hence $q_g$ is subharmonic on
\begin{equation}
\label{eq:p031-curvewise-scalar-good-locus}
                              C\setminus\overline{\mathcal B_g}.
\end{equation}
Thus, if
\[
 \mathcal F_g:=
 \{c\in C:q_g\text{ is not subharmonic on any neighborhood of }c\},
\]
then
\begin{equation}
\label{eq:p031-curvewise-failure-contained-in-scalar-defect}
                         \mathcal F_g\subseteq\overline{\mathcal B_g}.
\end{equation}
In particular, the intrinsic-to-ambient regularization gap
$\mathcal A_g$ cannot by itself support failure of the scalar-extension
curve test; such failure requires accumulation of the genuine canonical
retraction defect $\mathcal B_g$.
\end{theorem}

\begin{corollary}
\label{cor:p031-scalar-defect-curve-witness}
In the setting of
\Cref{thm:p031-curvewise-scalar-defect-localization}, if $u=s^*$ is not
classically plurisubharmonic on $V$, then there exist an analytic extension
$K'/K$, a separated smooth connected strict $K'$-analytic curve $C$, a
$K'$-analytic morphism $g:C\to V_{K'}$, and a point $c\in C$ such that
\[
 c\in\overline{
   \{d\in C:
      (s\circ\pi_{K'/K})^*(g(d))
       <u(\pi_{K'/K}(g(d)))\}
 }.
\]
Equivalently, every neighborhood of $c$ contains a point where the
extended-field regularization is strictly smaller than its value after the
canonical scalar retraction.  This is a necessary witness condition, not a
converse or a removability theorem.
\end{corollary}
