% P05.1 separable product-envelope proof, 25 September 2026.

\begin{proof}[\Cref{thm:frontier-polyhedral-separable-product-envelope}]
Set
\[
                         Q:=\sum_{i=1}^r p_i^*P_i.
\]
The projection of a polyhedral convex combination in the product is a
polyhedral convex combination in each factor.  Hence pullback by $p_i$
preserves polyhedral convexity, and finite sums of the pulled-back functions
are polyhedrally convex.  The hypotheses on $P_i$ therefore give
\[
 Q\in\mathrm{PAPC}_{\mathbf R}(\mathbf X,\gamma),
 \qquad
 Q\leq\sum_i p_i^*a_i=\gamma+u.
\]
Thus $Q$ is an admissible competitor and $Q\leq P_\gamma(u)$.

For the reverse inequality, let $\psi$ be any admissible competitor for
$P_\gamma(u)$.  We prove successively that
\[
 \psi(x_1,\ldots,x_r)
 \leq
 \sum_{i=1}^k P_i(x_i)+\sum_{i=k+1}^r a_i(x_i)
 \tag{**}\label{eq:frontier-polyhedral-product-induction}
\]
for $k=0,\ldots,r$.  The case $k=0$ is the obstacle inequality.  Assume the
claim for $k-1$ and first take all $x_i$ to be rational points of their
faces.  With the coordinates other than $x_k$ fixed, the slice of $\psi$ is
polyhedrally convex on $\mathbf X_k$: every polyhedral convex combination in
that factor lifts to one in the product by keeping the other rational
coordinates fixed.  Subtract the constant
\[
 C:=\sum_{i<k}P_i(x_i)+\sum_{i>k}a_i(x_i).
\]
The induction hypothesis gives
\[
                 \psi(x_1,\ldots,\mathord{\cdot},\ldots,x_r)-C
                 \leq a_k.
\]
The global growth bound $\psi\leq\gamma+O(1)$ gives the corresponding
$\gamma_k+O(1)$ upper growth bound on this slice.  The defining maximality
of $P_k=P_{\gamma_k}(u_k)$ therefore yields
\[
                 \psi(x_1,\ldots,x_r)-C\leq P_k(x_k),
\]
which is \eqref{eq:frontier-polyhedral-product-induction} for $k$.

Rational points are dense in every face of a rational polyhedral structure.
All functions in the displayed inequality are continuous, so the inequality
extends to arbitrary $(x_1,\ldots,x_r)$.  Induction up to $k=r$ gives
$\psi\leq Q$.  Taking the supremum over all admissible $\psi$ proves
$P_\gamma(u)\leq Q$, and hence
\eqref{eq:frontier-polyhedral-separable-product-envelope}.
\end{proof}

\begin{proof}[\Cref{cor:frontier-polyhedral-graph-affine-separable-regularity}]
The product of the factor balancing cycles is balanced.  On a product face,
the slope of $\gamma$ in a recession direction $(v_1,\ldots,v_r)$ is the
sum of the factor slopes.  If the direction is nonzero, at least one $v_i$
is nonzero, so factorwise positivity at infinity makes this sum strictly
positive.

If $\mathbf X_i$ is one-dimensional, then
\Cref{thm:frontier-polyhedral-one-dimensional-real-regularity} gives
$P_i\in\mathrm{PAPC}_{\mathbf R}(\mathbf X_i,\gamma_i)$.  If
$\mathbf X_i=N_{i,\mathbf R}$, the same conclusion follows from
\Cref{thm:frontier-polyhedral-affine-orthogonality}.  Apply
\Cref{thm:frontier-polyhedral-separable-product-envelope}.
\end{proof}
