\begin{proposition}
\label{prop:cld-psh-positive-current-verifier-extract}
On a pure-dimensional boundaryless tropical or analytic space, a locally
integrable CLD-psh function $u$ satisfies
\[
                         \dprimeddprime[u]\geq0.
\]
For a smooth $u$, the converse holds.
\end{proposition}

\begin{theorem}
\label{thm:cld-to-classical-psh-verifier-extract}
Every CLD-psh function is classically psh.
\end{theorem}

\begin{proposition}
\label{prop:classical-psh-calculus-verifier-extract}
Classically psh functions form a sheaf and a convex cone, are stable under
finite maxima and decreasing nets, scalar extension, and analytic pullback.
A real-valued $h$ is pluriharmonic when both $h$ and $-h$ are classically
psh.
\end{proposition}

\begin{theorem}
\label{thm:supplied-p061-singular-pluriharmonic-verifier-extract}
Let $K$ be a complete nontrivially valued non-Archimedean field, and let
$X$ be a good, separated, strict, boundaryless $K$-analytic space.  Every
classically pluriharmonic function $h\colon X\to\mathbb R$ is locally real
piecewise linear.
\end{theorem}

\begin{theorem}
\label{thm:p01-real-pl-positive-current-verifier-extract}
Let $K$ be a complete nontrivially valued non-Archimedean field, let $X$
be a good, separated, boundaryless, pure-dimensional, strictly
$K$-analytic space, and let $h\colon X\to\mathbf R$ be an $\mathbf R$-PL
function.  The following are equivalent:
\begin{enumerate}
\item $\dprimeddprime[h]$ is positive;
\item the residue class $L_h(x)$ is nef for every $x\in X$;
\item $h$ is semipositive in the sense of Gubler--Jell--Rabinoff.
\end{enumerate}
GJR harmonicity means that both $h$ and $-h$ are semipositive.
\end{theorem}

\begin{proposition}
\label{prop:gr-scalar-pluriharmonic-verifier-extract}
Let $L/K$ be an analytic extension field.  A function $h$ on a
$K$-analytic space is classically pluriharmonic if and only if its pullback
to the scalar extension is classically pluriharmonic.
\end{proposition}

The last proposition is Proposition~7.2(6) of Gubler--Rabinoff,
\emph{Classically psh and pluriharmonic functions on Berkovich spaces},
arXiv:2506.13548v2.
