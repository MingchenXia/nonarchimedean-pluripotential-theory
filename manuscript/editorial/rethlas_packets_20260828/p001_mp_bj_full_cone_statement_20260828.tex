\begin{theorem}
\label{thm:p001-mp-bj-full-cone-candidate}
Let \(X\) be a connected smooth projective variety over \(\mathbf C\), and
let

\[
                     \alpha\in\operatorname{Amp}(X)
                              \subset N^1(X)_{\mathbf R}.
\]

On the Mesquita--Piccione side, view \(\alpha\) as its real Bott--Chern
class.  Give \(\mathbf C\) the trivial absolute value, and let

\[
 \iota:X^{\beth}\xrightarrow{\ \sim\ }X^{\mathrm{an,triv}}
\]

be the canonical homeomorphism induced by GAGA.  On the left, let
\(\operatorname{PSH}_{\mathrm{MP}}(\alpha)\) denote the
Mesquita--Piccione cone: the decreasing completion of rational PL
functions whose cohomological model class is relatively nef.  On the
right, let
\(\operatorname{PSH}_{\mathrm{BJ}}(\alpha)\) denote the
Boucksom--Jonsson cone in numerical class \(\alpha\).  Then pullback by
\(\iota\) is an order isomorphism

\begin{equation}
\label{eq:p001-mp-bj-full-cone-candidate}
 \iota^*\operatorname{PSH}_{\mathrm{BJ}}(\alpha)
       =\operatorname{PSH}_{\mathrm{MP}}(\alpha).
\end{equation}

More precisely, it first identifies the rational PL model cones,

\begin{equation}
\label{eq:p001-mp-bj-model-cone-candidate}
 \iota^*\bigl(\operatorname{PL}(X^{\mathrm{an,triv}})
          \cap\operatorname{PSH}_{\mathrm{BJ}}(\alpha)\bigr)
 =\operatorname{PL}(X^{\beth})
          \cap\operatorname{PSH}_{\mathrm{MP}}(\alpha),
\end{equation}

and the two sides of
\eqref{eq:p001-mp-bj-full-cone-candidate} are the same decreasing
completion of this common rational PL cone.

If \(L\) is an ample \(\mathbf Q\)-line bundle with
\(c_1(L)=\alpha\), and \(\theta\) is a smooth representative of
\(c_1(L)\), then the unrestricted Darvas--Xia--Zhang--
Boucksom--Jonsson comparison, transported by \(\iota\), also gives an
order isomorphism

\[
 \operatorname{PSH}^{\mathrm{NA}}_{\mathrm{DXZ}}(X,\theta)
       \xrightarrow{\ \sim\ }
 \operatorname{PSH}_{\mathrm{MP}}(c_1(L))
\]

on all semivaluations, with no finite-energy hypothesis.
\end{theorem}
