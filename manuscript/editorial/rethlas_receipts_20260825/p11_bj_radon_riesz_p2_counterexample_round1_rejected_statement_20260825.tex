\begin{theorem}
Let \(k\) be an algebraically closed field with the trivial absolute value,
let
\[
             X=\mathbf P^2_k,
             \qquad L=\mathcal O_{\mathbf P^2}(1),
             \qquad \omega=c_1(L),
\]
and fix a line \(D\subset X\).  Choose pairwise distinct points
\(p_j\in D(k)\), and set
\[
             v_j=\operatorname{ord}_{p_j},
             \qquad v=\operatorname{ord}_D,
\]
where \(\operatorname{ord}_{p_j}\) is the divisorial valuation defined by
the exceptional divisor of the blowup of \(X\) at \(p_j\).  Then
\[
              v_j\longrightarrow v
              \quad\text{in the weak topology of }X^{\mathrm{lin}},
\]
and
\[
              T_\omega(v_j)=T_\omega(v)=1
              \qquad\text{for every }j,
\]
whereas
\[
              d_{\infty,\omega}(v_j,v)\geq1
              \qquad\text{for every }j.
\]
Consequently, weak convergence together with convergence of the
Boucksom--Jonsson exhaustion does not imply strong convergence on
\(X^{\mathrm{lin}}\), already for a smooth projective surface and in
arbitrary characteristic.
\end{theorem}
