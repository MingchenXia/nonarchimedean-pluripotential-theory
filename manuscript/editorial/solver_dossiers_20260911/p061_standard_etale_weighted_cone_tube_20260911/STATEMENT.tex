% Book-local P06.1 standard-etale weighted-cone tube theorem, 11 September 2026.
% It combines the existing contractive Newton lemma with relative cone descent;
% no novelty or priority claim is made.
\begin{theorem}[Standard-etale stability of a relative weighted-cone germ]
\label{thm:p061-standard-etale-weighted-cone-tube}
Let \(K\) be a complete non-Archimedean field with nontrivial absolute
value, let \(B\) be a strict \(K\)-affinoid algebra, and fix positive
integers \(w_1,\ldots,w_m\).  Suppose
\[
 A=B\langle Z_1,\ldots,Z_m\rangle/(F_1,\ldots,F_r),
\]
where every \(F_j\in B[Z_1,\ldots,Z_m]\) is homogeneous of strictly
positive weighted degree, with \(B\) of weight zero.  Let
\(\varepsilon:A\to B\) be evaluation at \(Z_1=\cdots=Z_m=0\), and let
\[
 C=A\langle T,U\rangle/\bigl(g(T),q(T)U-1\bigr),
 \tag{\(\heartsuit\)}\label{eq:p061-standard-etale-cone-presentation}
\]
where \(g,q\in A[T]\), the polynomial \(g\) is monic, and \(g'(T)\) is
invertible in \(C\).  Put
\[
                         D:=C\widehat\otimes_A B.
\]
View \(\mathcal M(D)\) as the closed central fibre in \(\mathcal M(C)\),
and fix \(x\in\mathcal M(D)\).

Assume that every classically pluriharmonic germ at \(x\) on
\(\mathcal M(D)\) is locally \(\mathbf R\)-PL.  Then every classically
pluriharmonic germ at \(x\) on \(\mathcal M(C)\) is locally
\(\mathbf R\)-PL and Gubler--Jell--Rabinoff harmonic.  In particular, the
conclusion holds whenever \(\mathcal M(B)\) is quasi-smooth.
\end{theorem}
