% The statement converts an algebraic etale germ into the strict analytic
% standard-etale presentation required by the preceding tube theorem.
\begin{theorem}[Algebraic-etale localization along a weighted-cone zero section]
\label{thm:p061-algebraic-etale-weighted-cone-zero-section}
Let \(K\) be a complete non-Archimedean field with nontrivial absolute
value.  Let \(\mathcal Y\) be a smooth finite-type \(K\)-scheme, let
\(\pi:\mathcal Z\to\mathcal Y\) be a finite-type \(K\)-scheme with a
section \(s:\mathcal Y\to\mathcal Z\), and fix
\(y\in\mathcal Y^{\mathrm{an}}\).
Assume that some affine open neighbourhood
\(\mathcal Y_0=\operatorname{Spec}R\) of the scheme-theoretic support of
\(y\) carries an isomorphism
\[
 \mathcal Z\times_{\mathcal Y}\mathcal Y_0
 \simeq
 \operatorname{Spec}
 \frac{R[Z_1,\ldots,Z_m]}{(F_1,\ldots,F_r)}                         \tag{\(*\)}
\]
under which \(s\) is the zero section.  Here \(m\geq1\),
\(w_1,\ldots,w_m\) are
positive integers and every \(F_j\) is homogeneous of strictly positive
weighted degree for \(\operatorname{wt}(Z_i)=w_i\) and
\(\operatorname{wt}(R)=0\).

Let \(q:\mathcal X\to\mathcal Z\) be a separated finite-type etale
morphism of \(K\)-schemes, and fix \(x\in\mathcal X^{\mathrm{an}}\) such that
\[
                  q^{\mathrm{an}}(x)=s^{\mathrm{an}}(y).
\]
Then every classically pluriharmonic germ at \(x\) on
\(\mathcal X^{\mathrm{an}}\) is locally \(\mathbf R\)-PL and
Gubler--Jell--Rabinoff harmonic.
\end{theorem}
