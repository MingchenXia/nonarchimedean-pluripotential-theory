\begin{corollary}[Endpoint mixed-tower Bedford--Taylor criterion]
\label{cor:p213-endpoint-mixed-tower-bt}
In the setting of
\Cref{thm:p213-endpoint-mixed-tower-classification}, assume that
\(B=B_\ell\) is boundaryless of pure dimension \(n\geq2\).  Let \(T\) be a
positive \(\mathrm d'\)- and \(\mathrm d''\)-closed ordinary CLD current of
bidegree \((p,p)\), \(0\leq p\leq n-1\), and let
\(u,v\in C^0(B,\mathbf R)\) satisfy
\[
                         \dprimeddprime[u]=\dprimeddprime[v].
\]
Then
\[
                         \dprimeddprime(uT)=\dprimeddprime(vT).
\]
In particular, this holds for every continuous psh pair in the
continuous-potential predicate of P21.3.
\end{corollary}

\begin{proof}
Put \(h=u-v\).  By
\Cref{thm:p213-endpoint-mixed-tower-classification}, the function \(h\) is
ordinary CLD-smooth with zero smooth Hessian.  Apply
\Cref{cor:ch16-smooth-h}.
\end{proof}

\begin{corollary}[Endpoint mixed-tower cover criterion]
\label{cor:p213-endpoint-mixed-tower-cover-bt}
Let \(K\) be complete and nontrivially valued, let \(X\) be a good,
topologically separated, pure, boundaryless strict \(K\)-analytic space of
dimension \(n\geq2\), and let \(U\subseteq X\) be ordinary open.  Suppose
that \(U\) has an ordinary open cover by sets \(B\) which, with their induced
analytic structures, satisfy the geometric and field hypotheses of
\Cref{cor:p213-endpoint-mixed-tower-bt}.  Then every positive closed
ordinary CLD current \(T\) of bidegree \((p,p)\) on \(U\),
\(0\leq p\leq n-1\), has Bedford--Taylor independence for every continuous
psh pair \(u,v\) on \(U\) with equal Hessian currents.
\end{corollary}

\begin{proof}
Put \(h=u-v\).  On every member \(B\) of the displayed cover,
\Cref{thm:p213-endpoint-mixed-tower-classification} makes \(h|_B\) ordinary
CLD-smooth with zero smooth Hessian.  These are local properties, so \(h\)
has them on \(U\).  Apply \Cref{cor:ch16-smooth-h} directly to \(T,u,v\) on
\(U\).
\end{proof}

\begin{corollary}[Endpoint mixed-tower basis criterion]
\label{cor:p213-endpoint-mixed-tower-basis-bt}
Let \(K\) and \(X\) be as in
\Cref{cor:p213-endpoint-mixed-tower-cover-bt}.  Suppose that the ordinary
topology of \(X\) has a basis of open sets \(B\) such that every \(B\), with
its induced analytic structure, satisfies the geometric and field hypotheses
of \Cref{cor:p213-endpoint-mixed-tower-bt}.  Then the continuous-pair
Bedford--Taylor predicate in P21.3 holds on every ordinary open subset of
\(X\) for every positive closed ordinary CLD current.
\end{corollary}

\begin{proof}
For an ordinary open \(U\subseteq X\), the basis members contained in \(U\)
form a cover to which
\Cref{cor:p213-endpoint-mixed-tower-cover-bt} applies.
\end{proof}
