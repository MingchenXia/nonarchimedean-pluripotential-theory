% Book-local finite-star product theorem, 10 September 2026.
% The proof is included in the live manuscript.  It makes no priority claim
% and concerns only the explicitly defined ambient-induced generator sheaf.
\begin{theorem}[Finite ambient stars with jointly convex Euclidean directions]
\label{thm:p071-finite-star-euclidean-collapse}
Let \(E\) be a finite-dimensional real vector space and let
\[
 S=\{0\}\cup\coprod_{i=1}^{m}(0,r_i)v_i\subset E,
 \qquad 1\leq m<\infty,
\]
where \(r_i>0\), \(v_i\ne0\), and the positive rays
\(\mathbf R_{>0}v_i\) are pairwise distinct.  Let
\(\Omega\subset\mathbf R^d\) be a nonempty open convex set and put
\(X:=S\times\Omega\).  For an ordinary open \(W\subset X\), let
\(\mathcal G_{X,0}(W)\) be the real-valued functions which are locally the
restrictions of smooth ambient functions on \(E\times\mathbf R^d\) and for
which, on every branch, the function
\[
 (t,y)\longmapsto g(tv_i,y)
\]
is jointly convex wherever it is defined.  Equivalently, its full Hessian in
the \((t,y)\) variables is positive semidefinite on every branch chamber.
Let \(\mathcal P_{X,1}\) be the sheaf of local pointwise decreasing limits
of sequences in \(\mathcal G_{X,0}\), and let \(\mathcal P_X\) be the
smallest subsheaf of \(\mathbf R\cup\{-\infty\}\)-valued functions which
contains \(\mathcal G_{X,0}\) and is closed under such limits.

Let \(\mathcal J_X(W)\) consist of the functions
\(u:W\to\mathbf R\cup\{-\infty\}\) with the following properties:
\begin{enumerate}
\item for every branch (i), the function obtained by setting
      \(u_i(0,y):=u(0,y)\) and \(u_i(t,y):=u(tv_i,y)\) for \(t>0\)
      is extended-real jointly convex on every convex subset of
      \[
       \{(t,y)\in[0,r_i)\times\Omega:(tv_i,y)\in W\};
      \]
      thus the branchwise condition includes every available segment from
      the common face into a branch; and
\item whenever \((0,y_0)\in W\), finitely many points
      \((t_a v_{i(a)},y_a)\in W\), and numbers \(\alpha_a\geq0\) satisfy
      \[
       \sum_a\alpha_a=1,
       \qquad
       \sum_a\alpha_a t_av_{i(a)}=0,
       \qquad
       \sum_a\alpha_a y_a=y_0,
      \]
      and every segment from \((0,y_0)\) to a point with
      \(\alpha_a>0\) is contained in \(W\), then
      \[
       u(0,y_0)\leq
       \sum_a\alpha_a u(t_av_{i(a)},y_a).
       \tag{\(\mathrm J_{S\times\Omega}\)}
      \]
      Terms with zero coefficient are omitted from the extended-real sum.
\end{enumerate}
Then \(\mathcal J_X\) is a sheaf and
\[
              \mathcal P_{X,1}=\mathcal P_X=\mathcal J_X.
\]
More precisely, every germ in these sheaves has a pointwise decreasing
approximation by members of \(\mathcal G_{X,0}\) on one fixed smaller
neighbourhood.  At a point of \(\{0\}\times\Omega\), that neighbourhood
may be chosen to be the product of a smaller finite substar and an ordinary
open convex ball in \(\Omega\).
\end{theorem}
