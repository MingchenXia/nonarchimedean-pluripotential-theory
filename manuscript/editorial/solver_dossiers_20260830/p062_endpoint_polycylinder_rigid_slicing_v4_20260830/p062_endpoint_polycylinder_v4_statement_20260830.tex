\begin{proposition}\label{prop:p062-strict-product-fibre-slicing-v4-research}
Let \(K\) be a complete non-Archimedean field with nontrivial absolute
value.  Let \(B\) be a good, topologically separated, pure
\(m\)-dimensional strict \(K\)-analytic space, and let \(Z\) be a good,
topologically separated, pure one-dimensional strict \(K\)-analytic space
without boundary.  Put \(X=B\times_K Z\).  If
\(h\in C^0(X,\mathbf R)\) satisfies
\begin{equation}\label{eq:p062-general-product-zero-current-v4-research}
                    \mathrm d'\mathrm d''[h]=0
\end{equation}
as a Chambert--Loir--Ducros current on \(X\), then, for every \(x\in B\),
the restriction
\[
       h_x:=h|_{X_x}\in C^0(X_x,\mathbf R),
       \qquad X_x\simeq Z_{\mathscr H(x)},
\]
satisfies
\begin{equation}\label{eq:p062-general-fibre-zero-current-v4-research}
                  \mathrm d'\mathrm d''[h_x]=0
\end{equation}
as a Chambert--Loir--Ducros current over the complete residue field
\(\mathscr H(x)\).
\end{proposition}

\begin{theorem}\label{thm:p062-endpoint-polycylinder-v4-research}
Let \(K\) be a complete non-Archimedean field with nontrivial absolute
value, and let \(p,q\) be nonnegative integers.  Products indexed by the
empty set are understood to be \(\mathcal M(K)\), and an empty sum is zero.
For \(1\leq i\leq p\), choose \(R_i\in|K^\times|\) and let
\[
 D_i=\{|T_i|<R_i\}
       \quad\text{or}\quad
 D_i=\{|T_i|\leq R_i\}
       \subset (\mathbf A^1_K)^{\an}.
\]
For \(1\leq j\leq q\), choose
\(
 0<r_j<R'_j
\)
in \(|K^\times|\), and let \(A_j\subset(\mathbf G_{\mathrm m,K})^{\an}\)
be any of
\[
\begin{gathered}
 \{r_j<|S_j|<R'_j\},\qquad
 \{r_j\leq|S_j|<R'_j\},\\
 \{r_j<|S_j|\leq R'_j\},\qquad
 \{r_j\leq|S_j|\leq R'_j\}.
\end{gathered}
\]
Put
\begin{equation}\label{eq:p062-endpoint-polycylinder-domain-v4-research}
             P=\prod_{i=1}^pD_i\times_K\prod_{j=1}^qA_j.
\end{equation}
If \(h\in C^0(P,\mathbf R)\) satisfies
\begin{equation}\label{eq:p062-endpoint-polycylinder-current-v4-research}
              \mathrm d'\mathrm d''[h]=0
              \qquad\text{on }\operatorname{Int}(P)
\end{equation}
as a Chambert--Loir--Ducros current on the absolute analytic interior,
then there are unique real constants
\(c,\lambda_1,\ldots,\lambda_q\) such that
\begin{equation}\label{eq:p062-endpoint-polycylinder-affine-v4-research}
                  h=c+\sum_{j=1}^q\lambda_j\log|S_j|
                  \qquad\text{on }P.
\end{equation}
When \(q=0\), this says that \(h\) is constant.  When \(p=q=0\), the
space \(P=\mathcal M(K)\) is zero-dimensional and the assertion is the
tautological equality \(h=h(*)\).

In particular, \(h\) is ordinary CLD-smooth with zero smooth Hessian,
weakly smooth, harmonic in the sense of Gubler--Jell--Rabinoff, and
classically pluriharmonic.

The theorem includes all finite products of standard strict open and closed
discs and standard strict open, closed, and half-closed annuli.  It makes no
assertion for a non-strict analytic polycylinder whose defining radii fall
outside \(|K^\times|\), an arbitrary analytic germ, or a current equation
imposed only on the coordinatewise strict radial core.
\end{theorem}
