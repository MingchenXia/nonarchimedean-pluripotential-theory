\begin{proposition}\label{prop:p062-finite-flat-continuous-trace}
Let \(K\) be a complete non-Archimedean field, let
\(p\colon Y\to X\) be a finite flat morphism of constant degree \(d\)
between good, topologically separated, pure \(n\)-dimensional
\(K\)-analytic spaces, and let \(h\in C^0(Y,\mathbf R)\).  For \(x\in X\),
set
\begin{equation}\label{eq:p062-finite-trace-definition}
 \operatorname{Tr}_p(h)(x)
   :=\int_{Y_x} h|_{Y_x},
\end{equation}
where the zero-dimensional fibre is integrated with its canonical
calibration, including analytic local degrees.  Then
\(\operatorname{Tr}_p(h)\) is continuous and
\begin{equation}\label{eq:p062-finite-trace-current}
                  p_*[h]=[\operatorname{Tr}_p(h)]
\end{equation}
as currents on \(X\).

If \(V\subset X\) is an ordinary open analytic subspace and
\(
 \mathrm d'\mathrm d''[h]=0
\)
on \(p^{-1}(V)\), then
\begin{equation}\label{eq:p062-finite-trace-hessian}
 \mathrm d'\mathrm d''[\operatorname{Tr}_p(h)]=0
 \qquad\text{on }V.
\end{equation}
\end{proposition}

\begin{proposition}\label{prop:p062-arbitrary-field-disc}
Let \(K\) be a complete non-Archimedean field with nontrivial absolute
value, let \(R\in |K^\times|\), and put
\[
 D_R^-:=\{x\in(\mathbf A_K^1)^{\an}:|T(x)|<R\},
 \qquad
 D_R^+:=\{x\in(\mathbf A_K^1)^{\an}:|T(x)|\leq R\}.
\]
If \(h\in C^0(D_R^-,\mathbf R)\) satisfies
\(
 \mathrm d'\mathrm d''[h]=0
\)
as a Chambert--Loir--Ducros current, then \(h\) is constant on \(D_R^-\).
If \(h\in C^0(D_R^+,\mathbf R)\) and the same equation holds on the
absolute analytic interior of \(D_R^+\), then \(h\) is constant on
\(D_R^+\).
\end{proposition}
