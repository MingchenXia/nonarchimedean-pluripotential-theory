\begin{lemma}
\label{lem:p061-product-open-tube}
Let \(K\) be a complete non-Archimedean field with nontrivial absolute
value.  Let \(A\) and \(B\) be strict \(K\)-affinoid algebras, let
\(a\in\mathcal M(A)(K)\), and write
\[
 P=A\widehat\otimes_KB,
 \qquad
 s_a:\mathcal M(B)\longrightarrow\mathcal M(P)
\]
for the section induced by evaluation at \(a\).  If
\(z\in\mathcal M(B)\) and \(\Omega\subset\mathcal M(P)\) is an ordinary
open neighbourhood of \(s_a(z)\), then there are strict affinoid
neighbourhoods \(U\subset\mathcal M(A)\) of \(a\) and
\(V\subset\mathcal M(B)\) of \(z\) such that
\[
                         U\times_KV\subset\Omega.
\]
\end{lemma}

\begin{proposition}
\label{prop:p061-parameterized-toric-fixed-fibre}
Let \(K\) be as above, let \(N\) be a lattice, put \(M=N^\vee\), and let
\(\rho\subset N_{\mathbf R}\) be a full-dimensional strongly convex
rational polyhedral cone.  Put
\[
 S=\rho^\vee\cap M,
 \qquad
 Z=\operatorname{Spec}K[S],
\]
and denote by \(a\in Z^{\mathrm{an}}(K)\) its torus-fixed point.  Let \(Y\)
be a good strict \(K\)-analytic space, let \(y\in Y\), and let
\(\Omega\) be an ordinary open neighbourhood of \((a,y)\) in
\(Z^{\mathrm{an}}\times_KY\).  For every classically pluriharmonic
function \(h:\Omega\to\mathbf R\), there are strict affinoid
neighbourhoods \(U\) of \(a\) and \(V\) of \(y\), with
\(U\times_KV\subset\Omega\), such that
\begin{equation}
\label{eq:p061-parameterized-toric-factorization}
 h|_{U\times_KV}
   =p_V^*g,
 \qquad
 g=s_a^*h|_V\in\operatorname{PH}(V).
\end{equation}
The neighbourhood \(U\) can be chosen from the cofinal weighted toric
affinoids
\begin{equation}
\label{eq:p061-parameterized-weighted-domain}
 U_n=\left\{x\in Z^{\mathrm{an}}:
  |\chi^u(x)|\leq|\varpi|^{n\langle u,v\rangle}
  \text{ for every }u\in S\right\},
\end{equation}
where \(0<|\varpi|<1\) and \(v\in N\cap\operatorname{int}(\rho)\).
In particular, if \(Y\) has the local property that every classically
pluriharmonic function is \(\mathbf R\)-PL, then the germ of \(h\) at
\((a,y)\) is \(\mathbf R\)-PL.  This applies when \(Y\) is quasi-smooth.
\end{proposition}

\begin{theorem}
\label{thm:p061-split-toric-all-orbits}
Let \(K\) be a complete non-Archimedean field with nontrivial absolute
value, let \(N\) be a lattice, and let
\(\sigma\subset N_{\mathbf R}\) be any strongly convex rational
polyhedral cone, with no simpliciality or full-dimensionality assumption.
Set
\[
 X_\sigma=\operatorname{Spec}
 K[\sigma^\vee\cap M],
 \qquad M=N^\vee.
\]
Every classically pluriharmonic function on an ordinary open analytic
subspace \(W\subset X_\sigma^{\mathrm{an}}\) is \(\mathbf R\)-PL and
harmonic in the sense of Gubler--Jell--Rabinoff.  Equivalently, this holds
at every analytic point of every torus orbit, including every
positive-dimensional orbit of a non-simplicial cone.
\end{theorem}

\begin{corollary}
\label{cor:p061-split-toric-varieties}
The conclusion of \Cref{thm:p061-split-toric-all-orbits} holds on every
ordinary open analytic subspace of the analytification of a split toric
variety associated to an arbitrary fan.  It also holds on every good
strict analytic space which is analytically locally isomorphic to an
ordinary open neighbourhood of a point
\[
 (a,y)\in Z^{\mathrm{an}}\times_KY,
\]
where \(Z\) and \(a\) are as in
\Cref{prop:p061-parameterized-toric-fixed-fibre} and \(Y\) is
quasi-smooth.
\end{corollary}

\begin{corollary}
\label{cor:p061-split-toric-quasismooth-products}
Let \(\mathcal X\) be a split toric \(K\)-variety and let \(Y\) be a
good strict quasi-smooth \(K\)-analytic space.  Then every classically
pluriharmonic function on an ordinary open analytic subspace
\[
 \Omega\subset \mathcal X^{\mathrm{an}}\times_KY
\]
is \(\mathbf R\)-PL and harmonic in the sense of
Gubler--Jell--Rabinoff.  Consequently, the same conclusion holds on every
good strict analytic space which is analytically locally isomorphic to an
ordinary open analytic subspace of such a product.
\end{corollary}

\begin{corollary}
\label{cor:p061-toric-forms}
Let \(X\) be a \(K\)-variety for which there is a finite analytic field
extension \(L/K\) such that \(X_L\) is a split toric \(L\)-variety.  Then
every classically pluriharmonic function on an ordinary open analytic
subspace of \(X^{\mathrm{an}}\) is \(\mathbf R\)-PL and
Gubler--Jell--Rabinoff harmonic.  In particular, the conclusion holds for
ordinary open subspaces of algebraic toric forms after any finite field
extension which splits the form.
\end{corollary}
