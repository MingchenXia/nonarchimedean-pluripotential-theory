\begin{proposition}\label{prop:p213-hessian-regular-basis-v2-candidate}
Let \(K\) be a complete non-Archimedean field, and let \(X\) be a good,
topologically separated, pure \(n\)-dimensional \(K\)-analytic space without
boundary. Assume that \(X\) has a basis \(\mathscr B\) for its ordinary
topology with the following property: if \(Q\in\mathscr B\) and
\(h\in C^0(Q,\mathbf R)\) satisfies
\begin{equation}\label{eq:p213-basis-zero-current-v2}
                    \mathrm d'\mathrm d''[h]=0,
\end{equation}
then \(h\) is ordinary Chambert--Loir--Ducros smooth on \(Q\) and

\[
                    \mathrm d'\mathrm d''h=0
\]
as a smooth form.

Let \(T\) be a positive Chambert--Loir--Ducros current on \(X\), of bidegree
\((\ell,\ell)\) with \(0\leq\ell\leq n-1\), such that
\(\mathrm d'T=\mathrm d''T=0\). Then, for every ordinary open set
\(W\subseteq X\) and every \(u,v\in C^0(W,\mathbf R)\) satisfying

\[
                 \mathrm d'\mathrm d''[u]
                 =\mathrm d'\mathrm d''[v],
\]
one has
\begin{equation}\label{eq:p213-basis-bt-v2}
 \mathrm d'\mathrm d''\bigl(u(T|_W)\bigr)
 =\mathrm d'\mathrm d''\bigl(v(T|_W)\bigr).
\end{equation}
In particular, \(T\) has continuous-pair Bedford--Taylor independence.
\end{proposition}

\begin{corollary}\label{cor:p213-polycylinder-basis-v2-candidate}
Let \(K\) be a complete non-Archimedean field with nontrivial absolute
value. Let \(X\) be a good, topologically separated, pure
\(n\)-dimensional strictly \(K\)-analytic space without boundary. Assume that
\(X\) has a basis for its ordinary topology whose members are analytically
isomorphic to standard strict open polycylinders
\begin{equation}\label{eq:p213-standard-polycylinder-v2}
 \prod_{i=1}^{a}\{|T_i|<R_i\}
 \times_K
 \prod_{j=1}^{b}\{r_j<|S_j|<R'_j\},
\end{equation}
where

\[
 a+b=n,\qquad R_i,r_j,R'_j\in|K^\times|,
 \qquad 0<r_j<R'_j.
\]
Then every positive Chambert--Loir--Ducros current \(T\) on \(X\), of bidegree
\((\ell,\ell)\) with \(0\leq\ell\leq n-1\), satisfying
\(\mathrm d'T=\mathrm d''T=0\), has the following property: for every
ordinary open \(W\subseteq X\) and every continuous equal-curvature pair
\(u,v\) on \(W\), the identity \eqref{eq:p213-basis-bt-v2} holds.
\end{corollary}

\begin{proposition}\label{prop:p213-fixed-polycylinder-v2-candidate}
Let \(K\) be a complete non-Archimedean field with nontrivial absolute
value, and let \(P\) be one standard strict open polycylinder of the form
\eqref{eq:p213-standard-polycylinder-v2}, of dimension \(n\). Let \(T\) be
a positive Chambert--Loir--Ducros current on \(P\), of bidegree
\((\ell,\ell)\) with
\(0\leq\ell\leq n-1\), such that \(\mathrm d'T=\mathrm d''T=0\). If
\(u,v\in C^0(P,\mathbf R)\) satisfy

\[
                  \mathrm d'\mathrm d''[u]
                  =\mathrm d'\mathrm d''[v],
\]
then
\begin{equation}\label{eq:p213-fixed-polycylinder-bt-v2}
             \mathrm d'\mathrm d''(uT)
             =\mathrm d'\mathrm d''(vT)
             \qquad\text{on }P.
\end{equation}
This is a fixed-domain assertion. It does not assert Bedford--Taylor
independence on every ordinary open subset of \(P\).
\end{proposition}
