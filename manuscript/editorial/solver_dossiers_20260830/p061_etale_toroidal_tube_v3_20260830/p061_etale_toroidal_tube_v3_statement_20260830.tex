\begin{proposition}
\label{prop:p061-v3-single-shilov-factorization}
Let \(K\) be a complete non-Archimedean field. Let
\[
 p:W\longrightarrow V,
 \qquad s:V\longrightarrow W,
 \qquad p\circ s=\operatorname{id}_V
\]
be morphisms of good \(K\)-analytic spaces, with \(W\) and \(V\)
affinoid. Assume that every analytic fibre
\[
                    W_b=W\widehat\otimes_V\mathscr H(b),
                    \qquad b\in V,
\]
is nonempty and has singleton Shilov boundary. Then every classically
pluriharmonic function \(h\) on an ordinary open neighbourhood of \(W\)
satisfies
\begin{equation}\label{eq:p061-v3-single-shilov-factorization}
                         h|_W=p^*(s^*h).
\end{equation}
This proposition is a factorization statement only; it makes no
piecewise-linearity assertion.
\end{proposition}

\begin{corollary}
\label{cor:p061-v3-single-shilov-rpl}
In addition to the hypotheses of
\cref{prop:p061-v3-single-shilov-factorization}, assume that \(K\) is
nontrivially valued and that \(W\) and \(V\) are good strictly
\(K\)-analytic spaces. If \(s^*h\) is locally \(\mathbf R\)-PL on \(V\),
then \(h|_W\) is locally \(\mathbf R\)-PL. In particular, this holds when
every classically pluriharmonic function on \(V\) is locally
\(\mathbf R\)-PL.
\end{corollary}

\begin{lemma}
\label{lem:p061-v3-contractive-standard-etale-tube}
Let \(K\) be complete and nontrivially valued. Let
\[
                         B\longrightarrow A\longrightarrow B
\]
be homomorphisms of strict \(K\)-affinoid algebras whose composite is the
identity, and put \(J=\ker(A\to B)\). Let \((A_n)_{n\geq1}\) be strict
affinoid \(B\)-algebras with compatible maps
\[
                         A\longrightarrow A_n\longrightarrow B
\]
extending \(A\to B\). Choose admissible Banach norms for which all four
structural maps \(B\to A\), \(A\to B\), \(A\to A_n\), and
\(A_n\to B\) are contractions.

Suppose that
\begin{equation}\label{eq:p061-v3-standard-etale-presentation}
 C=A\langle T,U\rangle/
       \bigl(g(T),h(T)U-1\bigr),
\end{equation}
where \(g\in A[T]\) is monic and \(g'(T)\) is invertible in \(C\).
Give \(C\) its quotient Gauss norm. Let \(g_0,h_0\in B[T]\) be the
coefficientwise images. For every coefficient \(a\) of \(g\) or \(h\),
write
\[
                         a^\circ=a-\iota\varepsilon(a)\in J,
\]
where \(\varepsilon:A\to B\) and \(\iota:B\to A\) are the displayed
retraction and section. Assume
\begin{equation}\label{eq:p061-v3-coefficient-contraction}
 \lim_{n\to\infty}
 \max_a\bigl\|a^\circ\bigr\|_{A_n}=0.
\end{equation}

Set
\[
 D=C\widehat\otimes_AB,
 \qquad C_n=C\widehat\otimes_AA_n,
 \qquad E_n=A_n\widehat\otimes_BD,
\]
identify the first two base changes with their canonical quotient Tate
presentations, give them the quotient Gauss norms, and give \(E_n\) the
projective tensor norm. Denote by \(t,u\) the images of
\(T,U\) in \(D\), and by \(T,U\) their images in \(C_n\). Then there are
one number \(0<r<1\) and one integer \(n_0\) such that, for every
\(n\geq n_0\), all four of the following assertions hold for that same
radius \(r\):
\begin{enumerate}
\item \(g\) has a unique zero \(t_n\) in the closed radius-\(r\) ball
 about \(t\) in \(E_n\);
\item \(g_0\) has \(t\) as its unique zero in that ball in \(E_n\);
\item \(g_0\) has a unique zero \(T_n^0\) in the closed radius-\(r\) ball
 about \(T\) in \(C_n\);
\item \(g\) has \(T\) as its unique zero in that ball in \(C_n\).
\end{enumerate}
Moreover, \(h(t_n)\) and \(h_0(T_n^0)\) are invertible, with inverses of
norm at most one, and the assignments
\[
 \begin{aligned}
 \alpha_n:C_n&\longrightarrow E_n,
 &T&\longmapsto t_n,
 &U&\longmapsto h(t_n)^{-1},\\
 \beta_n:E_n&\longrightarrow C_n,
 &t&\longmapsto T_n^0,
 &u&\longmapsto h_0(T_n^0)^{-1}
 \end{aligned}
\]
are inverse contractive \(A_n\)-homomorphisms. Thus
\begin{equation}\label{eq:p061-v3-etale-tube-isomorphism}
                         C_n\simeq E_n,
\end{equation}
and its reduction along \(A_n\to B\) is the identity of \(D\).
\end{lemma}

\begin{proposition}
\label{prop:p061-v3-etale-toric-product}
Let \(K\) be complete and nontrivially valued. Let \(N\) be a lattice,
let \(\rho\subset N_{\mathbf R}\) be a full-dimensional strongly convex
rational polyhedral cone, and set
\[
 S=\rho^\vee\cap N^\vee,
 \qquad Z=\operatorname{Spec}K[S].
\]
Let \(a\in Z(K)\) be the torus-fixed point, let \(\mathcal Y\) be a
smooth \(K\)-variety, and let
\[
 q:\mathcal X\longrightarrow Z\times_K\mathcal Y
\]
be a separated etale morphism of finite-type \(K\)-schemes. Fix
\[
 x\in\mathcal X^{\mathrm{an}},
 \qquad q^{\mathrm{an}}(x)=(a,y).
\]
Then there are a strict affinoid neighbourhood \(V\) of \(y\) in
\(\mathcal Y^{\mathrm{an}}\), a strict affinoid neighbourhood \(D\) of
\(x\) in the fixed fibre
\[
 \mathcal X_a^{\mathrm{an}}
 =\bigl(\mathcal X\times_{Z\times\mathcal Y}
        (\{a\}\times\mathcal Y)\bigr)^{\mathrm{an}},
\]
a weighted strict toric affinoid neighbourhood \(U\) of \(a\), and a
strict affinoid neighbourhood \(\Omega\) of \(x\) in
\(\mathcal X^{\mathrm{an}}\), together with an isomorphism
\begin{equation}\label{eq:p061-v3-actual-product}
                            \Omega\simeq U\times_KD
\end{equation}
over \(U\times_KV\). Under this isomorphism the fixed fibre is
\(\{a\}\times D\). The space \(D\) is quasi-smooth, and every analytic
fibre of the projection \(\Omega\to D\) has singleton Shilov boundary.
\end{proposition}

\begin{theorem}
\label{thm:p061-v3-etale-toroidal-ph-rpl}
In the setting of \cref{prop:p061-v3-etale-toric-product}, every
classically pluriharmonic function on an ordinary open neighbourhood of
\(x\) is \(\mathbf R\)-PL and Gubler--Jell--Rabinoff harmonic on a smaller
ordinary open neighbourhood of \(x\).
\end{theorem}

\begin{corollary}
\label{cor:p061-v3-zariski-etale-toric-charts}
Let \(K\) be complete and nontrivially valued. Let \(\mathcal X\) be a
separated finite-type \(K\)-scheme admitting, Zariski locally on the
source, ordinary etale charts to the product of a split toric
\(K\)-variety and a smooth \(K\)-variety. Then every classically
pluriharmonic function on every ordinary open analytic subspace of
\(\mathcal X^{\mathrm{an}}\) is \(\mathbf R\)-PL and
Gubler--Jell--Rabinoff harmonic.
\end{corollary}

\begin{proposition}
\label{prop:p061-analytic-standard-etale-toric-product}
Let \(K\) be complete and nontrivially valued, let \(N\) be a lattice, let
\(\rho\subset N_{\mathbf R}\) be a full-dimensional strongly convex rational
polyhedral cone, and put
\[
 S=\rho^\vee\cap N^\vee,
 \qquad Z=\operatorname{Spec}K[S].
\]
Let \(a\in Z(K)\) be the torus-fixed point.  Choose
\(0<|\varpi|<1\), choose \(v\in N\cap\operatorname{int}(\rho)\) when
\(\operatorname{rank}N>0\), and put \(v=0\) when
\(\operatorname{rank}N=0\).  For \(k\geq1\), write
\[
 U_k=\left\{z\in Z^{\mathrm{an}}:
  |\chi^s(z)|\leq|\varpi|^{k\langle s,v\rangle}
  \ \text{for every }s\in S\right\}.
\]
Let \(V=\mathcal M(B)\) be a good strict quasi-smooth affinoid
\(K\)-analytic space, fix \(m\geq1\), and set
\[
 R_k=\mathcal O(U_k),\qquad
 A=R_m\widehat\otimes_KB,
 \qquad A_n=R_{m+n}\widehat\otimes_KB.
\]
Let
\[
 C=A\langle T,U\rangle/\bigl(g(T),h(T)U-1\bigr),
\]
where \(g,h\in A[T]\), \(g\) is monic, and \(g'(T)\) is invertible in
\(C\); give \(C\) its quotient Gauss norm.  Let \(x\in\mathcal M(C)\)
map to \((a,y)\in U_m\times_KV\).  Then, for all sufficiently large
\(n\), the strict affinoid domain
\[
 \mathcal M(C_n),\qquad C_n=C\widehat\otimes_AA_n,
\]
is an ordinary neighbourhood of \(x\) and admits an isomorphism
\[
 \mathcal M(C_n)\simeq U_{m+n}\times_K\mathcal M(D),
 \qquad D=C\widehat\otimes_AB,
\]
over \(U_{m+n}\times_KV\).  This identifies the fixed fibre with
\(\{a\}\times_K\mathcal M(D)\); the space \(\mathcal M(D)\) is
quasi-smooth, and every analytic fibre of the projection
\(\mathcal M(C_n)\to\mathcal M(D)\) has singleton Shilov boundary.
\end{proposition}

\begin{theorem}
\label{thm:p061-analytic-standard-etale-toroidal-ph-rpl}
In the setting of
\Cref{prop:p061-analytic-standard-etale-toric-product}, every classically
pluriharmonic function on an ordinary open neighbourhood of \(x\) is
\(\mathbf R\)-PL and Gubler--Jell--Rabinoff harmonic on a smaller ordinary
open neighbourhood of \(x\).
\end{theorem}

\begin{corollary}
\label{cor:p061-analytic-standard-etale-toroidal-charts}
Let \(X\) be a good strict \(K\)-analytic space such that every point has
an ordinary open neighbourhood analytically isomorphic to an ordinary open
subspace of a strict affinoid space satisfying the hypotheses of
\Cref{prop:p061-analytic-standard-etale-toric-product}.  Then every
classically pluriharmonic function on every ordinary open analytic subspace
of \(X\) is \(\mathbf R\)-PL and Gubler--Jell--Rabinoff harmonic.
\end{corollary}

\begin{corollary}
\label{cor:p061-analytic-standard-etale-all-orbits}
Let \(X\) be a good strict \(K\)-analytic space.  Suppose that every
point of \(X\) has an ordinary open neighbourhood analytically isomorphic
to an ordinary open subspace of a strict affinoid space \(\mathcal M(C)\)
with the following presentation.  There are a split toric \(K\)-variety
\(\mathcal X\), a good strict quasi-smooth \(K\)-analytic space \(Y\), a
strict affinoid domain \(W=\mathcal M(A_0)\) in
\(\mathcal X^{\mathrm{an}}\times_KY\) which is an ordinary neighbourhood
of the image \(w\) of the chosen source point, and
\[
 C=A_0\langle T,U\rangle/
      \bigl(g(T),h(T)U-1\bigr),
\]
where \(g,h\in A_0[T]\), \(g\) is monic, and \(g'(T)\) is invertible in
\(C\).  Then every classically pluriharmonic function on every ordinary
open analytic subspace of \(X\) is \(\mathbf R\)-PL and
Gubler--Jell--Rabinoff harmonic.
\end{corollary}
