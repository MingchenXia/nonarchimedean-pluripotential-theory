\begin{proposition}\label{prop:p062-rational-centred-traces}
Let \(K\) be a complete non-Archimedean field with nontrivial absolute
value, let \(n\geq 1\), let
\(\mathbf R=(R_1,\ldots,R_n)\in |K^\times|^n\), and put
\[
 \begin{aligned}
 D^-_{\mathbf R}
  &=\{x\in(\mathbf A^n_K)^{\an}:|T_i(x)|<R_i
                       \text{ for every }i\},\\
 D^+_{\mathbf R}
  &=\{x\in(\mathbf A^n_K)^{\an}:|T_i(x)|\leq R_i
                       \text{ for every }i\}.
 \end{aligned}
\]
For \(a\in K^n\) and a positive polyradius
\(\mathbf r=(r_1,\ldots,r_n)\), write
\(\eta_{a,\mathbf r}\) for the corresponding Gauss point.

If \(h\in C^0(D^-_{\mathbf R},\mathbf R)\) satisfies
\begin{equation}\label{eq:p062-arbitrary-field-zero-current}
                       \mathrm d'\mathrm d''[h]=0
\end{equation}
as a Chambert--Loir--Ducros current, then
\begin{equation}\label{eq:p062-rational-centred-radial-identity}
                         h(\eta_{a,\mathbf r})=h(a)
\end{equation}
for every \(a\in D^-_{\mathbf R}(K)\) and every
\(0<r_i<R_i\).  Consequently \(h\) is constant on
\(D^-_{\mathbf R}(K)\), on the union of all \(K\)-centred Gauss
skeleta, and on the closure of that union.

If \(h\in C^0(D^+_{\mathbf R},\mathbf R)\) and
\eqref{eq:p062-arbitrary-field-zero-current} holds on the absolute
analytic interior of \(D^+_{\mathbf R}\), then the same conclusions hold
on \(D^+_{\mathbf R}\); in this case
\eqref{eq:p062-rational-centred-radial-identity} is valid for
\(a\in D^+_{\mathbf R}(K)\) and \(0<r_i\leq R_i\).
\end{proposition}

\begin{proposition}\label{prop:p062-explicit-base-change}
Keep the notation of \cref{prop:p062-rational-centred-traces}.

\begin{enumerate}
\item If there is a complete algebraically closed extension \(L/K\) such
that
\[
             \mathrm d'\mathrm d''[h_L]=0,
             \qquad h_L=h\circ p_L,
\]
on \(D^-_{\mathbf R,L}\), then \(h\) is constant on
\(D^-_{\mathbf R}\).

\item If \(h\in C^0(D^+_{\mathbf R},\mathbf R)\) and the analogous
equation holds on the absolute analytic interior of
\(D^+_{\mathbf R,L}\), then \(h\) is constant on
\(D^+_{\mathbf R}\).

\item The hypothesis of the first item is automatic if both \(h\) and
\(-h\) are plurisubharmonic in the approximation sense of
Chambert--Loir--Ducros on \(D^-_{\mathbf R}\).  Hence every such \(h\) is
constant.
\end{enumerate}
\end{proposition}

\begin{proposition}\label{prop:p062-finite-flat-continuous-trace}
Let \(K\) be a complete non-Archimedean field, let
\(p\colon Y\to X\) be a finite flat morphism of constant degree \(d\)
between good, topologically separated, pure \(n\)-dimensional
\(K\)-analytic spaces, and let \(h\in C^0(Y,\mathbf R)\).  For \(x\in X\),
set
\begin{equation}\label{eq:p062-finite-trace-definition}
 \operatorname{Tr}_p(h)(x)
   :=\int_{Y_x} h|_{Y_x},
\end{equation}
where the zero-dimensional fibre is integrated with its canonical
calibration, including analytic local degrees.  Then
\(\operatorname{Tr}_p(h)\) is continuous and
\begin{equation}\label{eq:p062-finite-trace-current}
                  p_*[h]=[\operatorname{Tr}_p(h)]
\end{equation}
as currents on \(X\).

If \(V\subset X\) is an ordinary open analytic subspace and
\(
 \mathrm d'\mathrm d''[h]=0
\)
on \(p^{-1}(V)\), then
\begin{equation}\label{eq:p062-finite-trace-hessian}
 \mathrm d'\mathrm d''[\operatorname{Tr}_p(h)]=0
 \qquad\text{on }V.
\end{equation}
\end{proposition}

\begin{proposition}\label{prop:p062-arbitrary-field-disc}
Let \(K\) be a complete non-Archimedean field with nontrivial absolute
value, let \(R\in |K^\times|\), and put
\[
 D_R^-:=\{x\in(\mathbf A_K^1)^{\an}:|T(x)|<R\},
 \qquad
 D_R^+:=\{x\in(\mathbf A_K^1)^{\an}:|T(x)|\leq R\}.
\]
If \(h\in C^0(D_R^-,\mathbf R)\) satisfies
\(
 \mathrm d'\mathrm d''[h]=0
\)
as a Chambert--Loir--Ducros current, then \(h\) is constant on \(D_R^-\).
If \(h\in C^0(D_R^+,\mathbf R)\) and the same equation holds on the
absolute analytic interior of \(D_R^+\), then \(h\) is constant on
\(D_R^+\).
\end{proposition}

\begin{proposition}\label{prop:p062-annulus-endpoint-arbitrary-field}
Let \(K\) be a complete non-Archimedean field with nontrivial absolute
value, and let
\(
             0<r<R
\)
belong to \(|K^\times|\).  Consider any one of the three standard strict
analytic domains
\[
 \begin{aligned}
 A_{r,R}^{\mathrm{oc}}
   &=\{x\in(\mathbf G_{\mathrm m,K})^{\mathrm{an}}:
                                      r<|T(x)|\leq R\},\\
 A_{r,R}^{\mathrm{co}}
   &=\{x\in(\mathbf G_{\mathrm m,K})^{\mathrm{an}}:
                                      r\leq|T(x)|<R\},\\
 A_{r,R}^{\mathrm{cc}}
   &=\{x\in(\mathbf G_{\mathrm m,K})^{\mathrm{an}}:
                                      r\leq|T(x)|\leq R\}.
 \end{aligned}
\]
Write \(A\) for the selected domain and
\(\operatorname{Int}(A)\) for its absolute analytic interior.  If
\(h\in C^0(A,\mathbf R)\) satisfies
\begin{equation}\label{eq:p062-annulus-endpoint-zero-current}
                 \mathrm d'\mathrm d''[h]=0
                 \qquad\text{on }\operatorname{Int}(A)
\end{equation}
as a Chambert--Loir--Ducros current, then there are unique
\(a,b\in\mathbf R\) such that
\begin{equation}\label{eq:p062-annulus-endpoint-affine}
                         h=a\log|T|+b
                         \qquad\text{on }A.
\end{equation}

This assertion is only one-dimensional.  It makes no endpoint claim for a
product, a polycylinder, or any higher-dimensional analytic domain.
\end{proposition}

\begin{proposition}\label{prop:p062-arbitrary-field-annulus-v4-source-correct}
Let \(K\) be as above, let \(0<r<R\) belong to \(|K^\times|\), and put
\[
 A^-_{r,R}:=\{x\in(\mathbf G_{\mathrm m,K})^{\mathrm{an}}:
                         r<|T(x)|<R\}.
\]
If \(h\in C^0(A^-_{r,R},\mathbf R)\) satisfies
\(
 \mathrm d'\mathrm d''[h]=0
\), then there are unique \(a,b\in\mathbf R\) such that
\begin{equation}\label{eq:p062-annulus-affine-v4-source-correct}
                         h=a\log|T|+b.
\end{equation}
\end{proposition}

\begin{theorem}\label{thm:p062-arbitrary-field-polycylinder-v4-source-correct}
Let \(K\) be as above, and let \(p,q\) be nonnegative integers. For
\(1\leq i\leq p\), let
\[
 D_i=\{x\in(\mathbf A^1_K)^{\mathrm{an}}:|T_i(x)|<R_i\},
 \qquad R_i\in|K^\times|,
\]
and, for \(1\leq j\leq q\), let
\[
 A_j=\{x\in(\mathbf G_{\mathrm m,K})^{\mathrm{an}}:
                  r_j<|S_j(x)|<R'_j\},
\]
where
\[
                  0<r_j<R'_j,
 \qquad r_j,R'_j\in|K^\times|.
\]
Put
\[
                         P=\prod_{i=1}^pD_i\times_K
                           \prod_{j=1}^qA_j.
\]
If \(h\in C^0(P,\mathbf R)\) satisfies
\(
 \mathrm d'\mathrm d''[h]=0
\), then there are unique real constants
\(c,\lambda_1,\ldots,\lambda_q\) such that
\begin{equation}\label{eq:p062-polycylinder-affine-v4-source-correct}
                   h=c+\sum_{j=1}^q\lambda_j\log|S_j|.
\end{equation}
In particular, \(h\) is ordinary CLD-smooth, weakly smooth, harmonic in
the sense of Gubler--Jell--Rabinoff, and classically pluriharmonic.
\end{theorem}

\paragraph{Scope firewall.}
Every source and target domain in the three assertions above is a
\emph{strict open} standard disc, annulus, or polycylinder.  No conclusion
is asserted for a closed or half-closed domain, for a non-strict endpoint,
or merely from a current equation imposed on the absolute analytic
interior of such a domain.

\begin{proposition}\label{prop:p062-strict-product-fibre-slicing-v4-research}
Let \(K\) be a complete non-Archimedean field with nontrivial absolute
value.  Let \(B\) be a good, topologically separated, pure
\(m\)-dimensional strict \(K\)-analytic space, and let \(Z\) be a good,
topologically separated, pure one-dimensional strict \(K\)-analytic space
without boundary.  Put \(X=B\times_K Z\).  If
\(h\in C^0(X,\mathbf R)\) satisfies
\begin{equation}\label{eq:p062-general-product-zero-current-v4-research}
                    \mathrm d'\mathrm d''[h]=0
\end{equation}
as a Chambert--Loir--Ducros current on \(X\), then, for every \(x\in B\),
the restriction
\[
       h_x:=h|_{X_x}\in C^0(X_x,\mathbf R),
       \qquad X_x\simeq Z_{\mathscr H(x)},
\]
satisfies
\begin{equation}\label{eq:p062-general-fibre-zero-current-v4-research}
                  \mathrm d'\mathrm d''[h_x]=0
\end{equation}
as a Chambert--Loir--Ducros current over the complete residue field
\(\mathscr H(x)\).
\end{proposition}

\begin{theorem}\label{thm:p062-endpoint-polycylinder-v4-research}
Let \(K\) be a complete non-Archimedean field with nontrivial absolute
value, and let \(p,q\) be nonnegative integers.  Products indexed by the
empty set are understood to be \(\mathcal M(K)\), and an empty sum is zero.
For \(1\leq i\leq p\), choose \(R_i\in|K^\times|\) and let
\[
 D_i=\{|T_i|<R_i\}
       \quad\text{or}\quad
 D_i=\{|T_i|\leq R_i\}
       \subset (\mathbf A^1_K)^{\an}.
\]
For \(1\leq j\leq q\), choose
\(
 0<r_j<R'_j
\)
in \(|K^\times|\), and let \(A_j\subset(\mathbf G_{\mathrm m,K})^{\an}\)
be any of
\[
\begin{gathered}
 \{r_j<|S_j|<R'_j\},\qquad
 \{r_j\leq|S_j|<R'_j\},\\
 \{r_j<|S_j|\leq R'_j\},\qquad
 \{r_j\leq|S_j|\leq R'_j\}.
\end{gathered}
\]
Put
\begin{equation}\label{eq:p062-endpoint-polycylinder-domain-v4-research}
             P=\prod_{i=1}^pD_i\times_K\prod_{j=1}^qA_j.
\end{equation}
If \(h\in C^0(P,\mathbf R)\) satisfies
\begin{equation}\label{eq:p062-endpoint-polycylinder-current-v4-research}
              \mathrm d'\mathrm d''[h]=0
              \qquad\text{on }\operatorname{Int}(P)
\end{equation}
as a Chambert--Loir--Ducros current on the absolute analytic interior,
then there are unique real constants
\(c,\lambda_1,\ldots,\lambda_q\) such that
\begin{equation}\label{eq:p062-endpoint-polycylinder-affine-v4-research}
                  h=c+\sum_{j=1}^q\lambda_j\log|S_j|
                  \qquad\text{on }P.
\end{equation}
When \(q=0\), this says that \(h\) is constant.  When \(p=q=0\), the
space \(P=\mathcal M(K)\) is zero-dimensional and the assertion is the
tautological equality \(h=h(*)\).

In particular, \(h\) is ordinary CLD-smooth with zero smooth Hessian,
weakly smooth, harmonic in the sense of Gubler--Jell--Rabinoff, and
classically pluriharmonic.

The theorem includes all finite products of standard strict open and closed
discs and standard strict open, closed, and half-closed annuli.  It makes no
assertion for a non-strict analytic polycylinder whose defining radii fall
outside \(|K^\times|\), an arbitrary analytic germ, or a current equation
imposed only on the coordinatewise strict radial core.
\end{theorem}
