\begin{proposition}\label{prop:p062-nonstrict-boundary-resonance-counterexample}
Let \(K\) be a complete non-Archimedean field with nontrivial absolute
value, and put
\[
 \sqrt{|K^\times|}
 :=\{a\in\mathbf R_{>0}:a^m\in|K^\times|
                         \text{ for some }m\geq1\}.
\]
Assume that \(R\in\mathbf R_{>0}\setminus\sqrt{|K^\times|}\), and let
\begin{equation}\label{eq:p062-resonant-bidisc-domain}
 P=\{|T|\leq R\}\times_K\{|S|\leq R^{-1}\}
       \subset(\mathbf A^2_K)^{\an}.
\end{equation}
Such pairs occur already over a discretely valued field: for example, one
may take \(K=\mathbf Q_\ell\) and \(R=|\ell|^{\sqrt 2}\).
Then
\begin{equation}\label{eq:p062-resonant-boundary-tube}
             W=\{x\in P:|T(x)S(x)-1|<1\}
\end{equation}
is a nonempty relatively open subset of
\(P\setminus\operatorname{Int}(P)\).

For every \(c\in(0,1)\), the function
\begin{equation}\label{eq:p062-resonant-cutoff}
 h_c(x)=\max\bigl\{0,1-c^{-1}|T(x)S(x)-1|\bigr\}
\end{equation}
is continuous and nonconstant on \(P\), is supported in \(W\), and
satisfies
\begin{equation}\label{eq:p062-resonant-zero-current}
                \mathrm d'\mathrm d''[h_c]=0
                \qquad\text{on }\operatorname{Int}(P).
\end{equation}
Consequently, the affine classification in the strict endpoint-polycylinder
theorem does not extend to arbitrary positive radii if the current equation
is imposed only on the absolute analytic interior.  This already fails for
a product of two closed discs, where the asserted conclusion would be that
the function is constant.
\end{proposition}

\begin{proposition}\label{prop:p062-nonstrict-scalar-stable-repair}
Let \(K\) be a complete non-Archimedean field with nontrivial absolute
value.  Form a finite product \(P\) with \(p\) standard open or closed
disc factors
\[
             \{|T_i|<R_i\}\quad\text{or}\quad\{|T_i|\leq R_i\}
\]
and \(q\) standard open, closed, or half-closed annulus factors, in
coordinates \(S_j\), chosen among
\[
\begin{gathered}
 \{r_j<|S_j|<R'_j\},\qquad
 \{r_j\leq|S_j|<R'_j\},\\
 \{r_j<|S_j|\leq R'_j\},\qquad
 \{r_j\leq|S_j|\leq R'_j\},
\end{gathered}
\]
where all \(R_i,r_j,R'_j\) are arbitrary positive real numbers and
\(r_j<R'_j\).
Let \(h\in C^0(P,\mathbf R)\).  Suppose that there is a complete extension
\(L/K\) such that every defining radius belongs to \(|L^\times|\) and,
for the scalar-extension projection \(p_L:P_L\to P\), the function
\(h_L=h\circ p_L\) satisfies
\begin{equation}\label{eq:p062-scalar-stable-current}
            \mathrm d'\mathrm d''[h_L]=0
            \qquad\text{on }\operatorname{Int}(P_L).
\end{equation}
Then there are unique real constants
\(c,\lambda_1,\ldots,\lambda_q\) such that
\begin{equation}\label{eq:p062-scalar-stable-affine}
              h=c+\sum_{j=1}^q\lambda_j\log|S_j|
              \qquad\text{on }P.
\end{equation}
In particular, the arbitrary-radius classification is valid for the
scalar-stable zero-current condition \eqref{eq:p062-scalar-stable-current}.
The counterexample in
\cref{prop:p062-nonstrict-boundary-resonance-counterexample} shows that
the corresponding equation over \(K\) on
\(\operatorname{Int}(P)\) does not, by itself, imply this scalar-stable
condition.
\end{proposition}
