% Direct local-isomorphism proof for the finite-split invariant corollary,
% 9 September 2026.

\begin{proof}
Fix \(w\in W\), and choose the displayed data.  The transported function
\((h|_U)_L\) on \(\operatorname{trop}^{-1}(O)\) satisfies every hypothesis of
\Cref{cor:p062-finite-split-invariant-scalar-stable-zero-current}.  Hence it
gives the stated three conclusions on \(U\).  Since \(w\) was arbitrary,
they hold locally on \(W\).
\end{proof}
