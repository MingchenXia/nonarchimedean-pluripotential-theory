\begin{corollary}
\label{cor:p062-open-polycylinder-products-over-curves}
Let \(K\), \(C\), and \(W\subset C^{\mathrm{an}}\) be as in
\Cref{thm:wanner-curve-comparison}.  Let \(s,r\geq1\), choose
\[
 \begin{gathered}
 A_i=\{z\in(\mathbf G_{\mathrm m,K})^{\mathrm{an}}:
                     r_i<|T_i(z)|<R_i\},
 \qquad 0<r_i<R_i,\quad r_i,R_i\in|K^\times|
       \quad(1\leq i\leq s),\\
 D_j^-=\{z\in(\mathbf A^1_K)^{\mathrm{an}}:|S_j(z)|<\rho_j\},
 \qquad \rho_j\in|K^\times|
       \quad(1\leq j\leq r),
 \end{gathered}
\]
and put
\[
 X=W\times_K\prod_{i=1}^sA_i\times_K\prod_{j=1}^rD_j^-,
 \qquad p:X\longrightarrow W,
 \qquad q_i:X\longrightarrow A_i.
\]
If \(h\in C^0(X,\mathbf R)\) satisfies
\(\mathrm d'\mathrm d''[h]=0\), then there are unique continuous
functions \(b,a_1,\ldots,a_s:W\to\mathbf R\) such that
\[
 h=p^*b+\sum_{i=1}^s(a_i\circ p)q_i^*\log|T_i|,
 \qquad \mathrm d'\mathrm d''[b]=0,
\]
and every \(a_i\) is locally constant.  In particular, \(h\) is classically
pluriharmonic, locally \(\mathbf R\)-PL, and Gubler--Jell--Rabinoff harmonic.
\end{corollary}
