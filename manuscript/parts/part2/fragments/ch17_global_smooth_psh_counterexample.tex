\subsection{A locally psh metric with no global smooth psh metric}
\label{sec:supplied-p081-global-smooth-counterexample}

The following argument is adapted from the supplied manuscript
\cite{SuppliedP08120260925}.  It answers the literal smooth-approximation
question of \cite[Remark~17.8.10]{CLD25}, rather than the distinct question
of approximation by rational model metrics.

Let $K=\mathbf C((t))$, with $0<|t|<1$, and let
\[
                         E\colon y^2=x^3-x
\]
be the elliptic curve over $\mathbf C$ with origin $O$ at infinity.  Choose
a transcendental number $\alpha\in\mathbf C$, a square root $\beta$ of
$\alpha^3-\alpha$, and put $P=(\alpha,\beta)$.  Write
\[
 X=(E_K)^{\An},\qquad L=\cO_{E_K}(P-O)^{\An}.
\]

\begin{theorem}
\label{thm:supplied-p081-global-smooth-counterexample}
The line bundle $L$ has a continuous metric $h_0$ whose local weights are
psh in the ordinary Chambert--Loir--Ducros sense.  Every local weight is, in
fact, locally a uniform limit of ordinary smooth psh functions.  The line
bundle $L$ admits no globally defined ordinary smooth psh metric.
Consequently $h_0$ is not globally psh.

The metric $h_0$ is the integral model metric induced by
$\cO_E(P-O)$ on the smooth constant model
$E\times_{\mathbf C}\Spec\mathbf C[[t]]$.  This model line bundle is nef on
the unique special-fibre curve.
\end{theorem}

We first isolate the obstruction, which applies to every constant smooth
projective curve.  Let $C/\mathbf C$ be such a curve, let $\xi$ be the
good-reduction point of $(C_K)^{\An}$, and let $B_R$ be the residual open
disc indexed by $R\in C(\mathbf C)$.  Choose a rational function
$z_R$ that is a local parameter at $R$ and put
$r_R=-\log|z_R|$ on $B_R\setminus\{R\}$, with $r_R\to0$ towards $\xi$.
For $g\in\mathbf C(C)^\times$ one has, on an initial radial segment,
\begin{equation}\label{eq:supplied-p081-residual-slope}
             \log|g(\zeta_{R,r})|=-\operatorname{ord}_R(g)r.
\end{equation}

\begin{lemma}\label{lem:supplied-p081-smooth-slope-divisor}
If $u$ is ordinary smooth on a neighbourhood of $\xi$, its outgoing slope
divisor
\[
 A_u=\sum_{R\in C(\mathbf C)}
       \left.\frac{\rd}{\rd r}u(\zeta_{R,r})\right|_{r=0+}[R]
\]
is a finite real linear combination of principal divisors.  In particular,
$\deg A_u=0$.
\end{lemma}

\begin{proof}
The local description of ordinary smooth functions
\cite[Proposition~15.1.1(b) and Corollary~15.1.2]{CLD25} gives, after
shrinking, an expression
\[
                  u=F(-\log|f_1|,\ldots,-\log|f_m|)
\]
with $F$ smooth and the $f_i$ invertible analytic functions.  Scale $f_i$
by a constant $b_i$ of its norm at $\xi$.  Its nonzero reduction is a
rational function $g_i\in\mathbf C(C)^\times$.  The ultrametric inequality
and continuity give $|f_i/b_i|=|g_i|$ on a smaller neighbourhood of $\xi$.
Thus \eqref{eq:supplied-p081-residual-slope} and the chain rule yield
\[
                   A_u=\sum_i(\partial_iF)(\xi)\operatorname{div}(g_i).
\]
Each divisor on the right has finite support and degree zero.
\end{proof}

\begin{proposition}\label{prop:supplied-p081-torsion-obstruction}
Let $D=\sum_Rd_R[R]$ be an integral divisor of degree zero on $C$, and put
$L_D=\cO_{C_K}(D)^{\An}$.  If $L_D$ admits a globally ordinary smooth psh
metric, then $[D]\in\Pic^0(C)$ is torsion.  Conversely, a torsion class
admits an ordinary smooth pluriharmonic metric on $L_D$.
\end{proposition}

\begin{proof}
Let $s$ be the canonical rational section of $L_D$, so that
$\operatorname{div}(s)=D$, and write $\lambda=-\log\|s\|$ near $\xi$.
For a hypothetical smooth psh metric, put
$A_\lambda=\sum_Ra_R[R]$.  By
\cref{lem:supplied-p081-smooth-slope-divisor}, this is real-principal and
has degree zero.

On $B_R$, the section $e_R=s/z_R^{d_R}$ is a frame and its weight is
\[
 w_R=-\log\|e_R\|=\lambda-d_Rr_R,
 \qquad
 w_R(\zeta_{R,r})=w_R(\xi)+(a_R-d_R)r+o(r).
\]
Choose a smooth nondecreasing function $\eta\colon\mathbf R\to[0,1]$ that
is zero on $(-\infty,1]$ and one on $[2,+\infty)$.  The function
$\chi_\varepsilon=\eta(r_R/\varepsilon)$, extended by one at the centre of
$B_R$, is an ordinary smooth nonnegative compactly supported test function.
Since $w_R$ is psh, its Hessian current is positive by
\cite[Proposition~8.3.1(a) and Theorem~15.7.1]{CLD25}.  One-dimensional
tropical integration, with multiplicity one, gives
\[
 \left\langle\dprimeddprime[w_R],\chi_\varepsilon\right\rangle
 =\int_\varepsilon^{2\varepsilon}
     w_R(\zeta_{R,r})\frac{\rd^2}{\rd r^2}
          \eta(r/\varepsilon)\,\rd r
 \longrightarrow -(a_R-d_R).
\]
Hence $a_R\leq d_R$ for every $R$.  The finite divisors $A_\lambda$ and
$D$ both have degree zero, so $A_\lambda=D$ coefficientwise.  Thus $D$ is
a finite real linear combination of principal divisors.  In the finite
free abelian group of divisors supported on the union of the relevant
supports, this is a linear system with rational coefficients and integral
right-hand side.  Row reduction over $\mathbf Q$ supplies a rational
solution; after clearing denominators, a positive multiple of $D$ is
principal.  Therefore $[D]$ is torsion.

Conversely, suppose $mD=\operatorname{div}(f)$ for
$f\in\mathbf C(C)^\times$.  Define
$-\log\|s\|=-(1/m)\log|f|$ away from $\operatorname{supp}D$.  At
$R\in\operatorname{supp}D$, write
$f=z_R^{md_R}u_R$ with $u_R$ a constant-field unit.  Since $|u_R|=1$ on
$B_R$, the weight of $e_R=s/z_R^{d_R}$ is zero there.  These weights glue
to a globally ordinary smooth metric with zero Hessian.
\end{proof}

\begin{proof}[Proof of \Cref{thm:supplied-p081-global-smooth-counterexample}]
The point $P$ has infinite order.  Indeed, for each $m\geq1$ the finite
multiplication morphism $[m]\colon E\to E$ is defined over $\mathbf Q$, so
the coordinates of every torsion point are algebraic over $\mathbf Q$,
whereas the $x$-coordinate of $P$ is transcendental.  The standard
identification $E(\mathbf C)\simeq\Pic^0(E)$ sends $P$ to $[P-O]$; it also
follows directly from Riemann--Roch and the chord-and-tangent law.  Hence
\cref{prop:supplied-p081-torsion-obstruction} excludes every globally
ordinary smooth psh metric on $L$.

It remains to construct the locally psh metric.  In the canonical rational
frame $s$ on $X\setminus\{P,O\}$, set
\[
 \lambda_0=-\log\|s\|_{h_0}=
 \begin{cases}
 r_P&\text{on }B_P\setminus\{P\},\\
 -r_O&\text{on }B_O\setminus\{O\},\\
 0&\text{on every other residual disc and at }\xi.
 \end{cases}
\]
The weight of $s/z_P$ is zero on $B_P$, and that of $sz_O$ is zero on
$B_O$.  Thus the formula defines a continuous metric at the two centres;
it is continuous at $\xi$ because only two residual directions carry
nonzero values.  This is precisely the integral metric induced by
$\cO_E(P-O)$ on the constant smooth model.  Its special-fibre degree is
zero, hence the model line bundle is nef.

Away from $\xi$, the displayed frame weights are constant or logarithms of
invertible analytic functions and are ordinary smooth pluriharmonic.  To
verify pshness at $\xi$, fix $a>0$ and remove the two closed subdiscs
$\{r_P\geq a\}$ and $\{r_O\geq a\}$.  The result is an ordinary
neighbourhood $V$ of $\xi$ on which $s$ is invertible and
$|\lambda_0|\leq a$.

For $n\geq1$, put $R_n=(n+1)P$.  Non-torsion of $P$ makes the points
$R_n$ distinct from $P$ and $O$.  The elliptic group law supplies a
rational function $f_n$ with
\[
             \operatorname{div}(f_n)=(n+1)[P]-n[O]-[R_n].
\]
Choose a smooth convex function $\rho\colon\mathbf R\to\mathbf R$ equal to
$u$ for $u\geq-1$ and to $-2$ for $u\leq-3$.  On
$V\setminus B_{R_n}$ put $\mu_n=-(1/n)\log|f_n|$, and on $B_{R_n}$ put
\[
                         \mu_n=\frac1n\rho(-r_{R_n}).
\]
The two definitions agree near the boundary of $B_{R_n}$.  The function
$\mu_n$ is ordinary smooth on $V$; its Hessian vanishes outside the
transition annulus and there equals
\[
 \dprimeddprime\mu_n=
   \frac{\rho''(-r_{R_n})}{n}
       \dprime r_{R_n}\wedge\ddprime r_{R_n}\geq0.
\]
Thus $\mu_n$ is ordinary smooth psh.  The divisor formula gives
\[
 \mu_n=(1+1/n)r_P\quad\text{on }B_P\cap V,
 \qquad
 \mu_n=-r_O\quad\text{on }B_O\cap V,
\]
and $\mu_n=0$ on every other residual disc except $B_{R_n}$, where
$|\mu_n|\leq2/n$.  Consequently
\[
                    \sup_V|\mu_n-\lambda_0|
                    \leq\frac{\max\{a,2\}}n\longrightarrow0.
\]
The uniform-closure definition of continuous psh functions
\cite[Definition~8.1.7 and Propositions~8.1.9, 8.1.11]{CLD25} now makes
$\lambda_0$ psh on $V$.  Lemma~8.1.10 of the same source also permits a
decreasing subsequence after adding constants tending to zero.  Hence all
local frame weights of $h_0$ are psh by
\cite[Definition~17.8.1]{CLD25}.  Since no globally smooth psh metric exists
on $L$, the metric $h_0$ cannot be globally psh in the sense of
\cite[\S17.8.5]{CLD25}.
\end{proof}

\begin{remark}\label{rem:supplied-p081-scope}
The obstruction occurs in degree zero and does not contradict the
positive-degree comparison in
\cref{thm:p081-corrected-curve-comparison}.  It is stronger than the Tate
example \cref{thm:p081-irrational-tate-period-obstruction}: that example
separates smooth global psh approximation from rational-model
approximation, whereas the present non-torsion constant bundle has no
globally smooth psh metric at all.  No assertion about ample-class envelope
continuity is involved.
\end{remark}
