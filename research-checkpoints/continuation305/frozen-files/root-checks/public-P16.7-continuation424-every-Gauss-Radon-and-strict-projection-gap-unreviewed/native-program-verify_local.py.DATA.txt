from pathlib import Path

def dot(x,y):
    return sum((a*b for a,b in zip(x,y)), Q(0))

def clip(poly, a, b):
    out = []
    for p,q in zip(poly, poly[1:]+poly[:1]):
        fp,fq = dot(a,p)-b, dot(a,q)-b
        if fp <= 0:
            out.append(p)
        if (fp < 0 and fq > 0) or (fp > 0 and fq < 0):
            t = fp/(fp-fq)
            out.append(tuple(p[k]+t*(q[k]-p[k]) for k in range(2)))
    return out

def area(poly):
    return abs(sum((p[0]*q[1]-p[1]*q[0] for p,q in zip(poly,poly[1:]+poly[:1])),Q(0)))/2

def full_cells(sites, weights):
    results = []
    for i,x in enumerate(sites):
        half = Q(1,2)
        p = [(x[0]-half,x[1]-half),(x[0]+half,x[1]-half),
             (x[0]+half,x[1]+half),(x[0]-half,x[1]+half)]
        for j,z in enumerate(sites):
            for k in product(range(-2,3),repeat=2):
                if i==j and k==(0,0):
                    continue
                y = tuple(z[l]+k[l] for l in range(2))
                a = tuple(y[l]-x[l] for l in range(2))
                b = (dot(y,y)-dot(x,x))/2+weights[i]-weights[j]
                p = clip(p,a,b)
        results.append((p,area(p)))
    return results

cases = 0
for den in range(2,21):
    for num in range(1,den):
        target = Q(num,den)
        d = (target-Q(1,2))/4
        cells = full_cells([(Q(0),Q(0)),(Q(1,2),Q(0))],[d,Q(0)])
        assert cells[0][1] == target and cells[1][1] == 1-target
        cases += 1
sites = [(Q(0),Q(0)),(Q(1,2),Q(0)),(Q(0),Q(1,2)),(Q(1,2),Q(1,2))]
cells = full_cells(sites,[Q(0)]*4)
assert [a for p,a in cells] == [Q(1,4)]*4
three = full_cells([(Q(0),Q(0)),(Q(1,2),Q(0)),(Q(0),Q(1,2))], [Q(-1,20),Q(0),Q(1,40)])
assert sum((a for p,a in three),Q(0)) == 1
assert all(a>0 for p,a in three)

# Exact separable one-atom moments, with independent integration of polynomials.
average_one = Q(1,2)*(Q(1,2)-Q(1,3))-Q(1,8)
assert average_one == Q(-1,24)
reference = 2*(2*average_one)
mixed = 2*(-Q(1,8)+average_one)
top = 2*(-Q(1,4))
energy = (reference+mixed+top)/3
dual = energy-top
assert (reference,mixed,top,energy,dual) == (Q(-1,6),Q(-1,3),Q(-1,2),Q(-1,3),Q(1,6))

# Entire cubic midpoint identity on a basis, not a numerical inference.
for degree in range(4):
    p0 = Q(int(degree==0))
    p1 = Q(1)
    pm = Q(1,2)**degree
    dp0 = Q(int(degree==1))
    dp1 = Q(degree)
    assert pm-(p0+p1)/2 == (dp0-dp1)/8

    assert depth>=0
