"""Own bounded, inert UTF-8 packet reader. It never imports packet code.

Usage: python packet_read.py math 134 1 180
       python packet_read.py private NAME_FRAGMENT 1 180
Byte verification is separate from semantic reading.
"""
from pathlib import Path
import sys, json, hashlib

BASE = Path('/workspace/research/campaigns/round-272/P16.7-whole383-intrinsic-E1-and-actual-branch-Dirac')
kind, key = sys.argv[1:3]
manifest = BASE / ('mathematical-packet-bindings.json' if kind == 'math' else 'private-primary-packet-bindings.json')
entries = json.loads(manifest.read_text())
if kind == 'math':
    selected = [e for e in entries if Path(e['path']).name.startswith(key.zfill(3) + '-')]
else:
    selected = [e for e in entries if key in e['path']]
if len(selected) != 1:
    print(json.dumps([e['path'] for e in selected], indent=2)); sys.exit(2)
e = selected[0]
p = Path(e['path'])
if any(s in p.name for s in ('P16.1-', 'P16.2-', 'P16.4-', 'P16.5-', 'P16.6-', 'P21.3-')):
    raise SystemExit('Other active mathematics excluded')
b = p.read_bytes()
digest = hashlib.sha256(b).hexdigest()
assert len(b) == e['bytes'] and digest == e['sha256'], 'binding mismatch'
lines = b.decode('utf-8').split('\n')
lo = int(sys.argv[3]) if len(sys.argv) > 3 else 1
hi = int(sys.argv[4]) if len(sys.argv) > 4 else min(lo + 179, len(lines))
assert hi - lo < 500 and 1 <= lo <= hi <= len(lines)
record = dict(path=str(p), sha256=digest, bytes=len(b), total_lines=len(lines), emitted_lines=[lo,hi], semantic_review='not inferred from emission')
with open('reader-emissions.jsonl','a') as f:
    f.write(json.dumps(record)+'\n')
print(json.dumps(record))
for i in range(lo-1, hi): print(f'{i+1}: {lines[i]}')
