"""Own bounded exact diagnostics; never executes supplied/native material."""
from fractions import Fraction as F
from itertools import product, combinations
from pathlib import Path
import json
import platform
import sys

ROOT = Path(__file__).resolve().parents[1]
OUT = ROOT / "evidence"
OUT.mkdir(exist_ok=True)

def rank(rows):
    pivots = {}
    for row in rows:
        work = list(map(F, row))
        for pos in sorted(pivots):
            scale = work[pos]
            if scale:
                work = [x - scale*y for x, y in zip(work, pivots[pos])]
        pos = next((i for i, x in enumerate(work) if x), None)
        if pos is not None:
            scale = work[pos]
            pivots[pos] = [x/scale for x in work]
    return len(pivots)

periods = []
for A, B, C, D in product(range(-4, 5), repeat=4):
    # Formal valuation and the independent rational prime exponents.
    if B == C and F(2)**D == F(3)**A:
        periods.append([A, B, C, D])
assert periods == [[0, k, k, 0] for k in range(-4, 5)]

vectors = list(product((-1, 0, 1), repeat=4))
index = list(combinations(range(4), 2))
quad_index = list(combinations(range(6), 2))
rows = []
failures = 0
for a in vectors:
    for b in vectors:
        w = [a[i]*b[j] - a[j]*b[i] for i, j in index]
        Q = w[0]*w[5] - w[1]*w[4] + w[2]*w[3]
        failures += Q != 0
        rows.append([x*x for x in w] + [w[i]*w[j] for i, j in quad_index])
assert failures == 0
assert rank(rows) == 20
assert rank(rows + [[1,0,0,0,0,1] + [int((i,j)==(0,5)) for i,j in quad_index]]) == 21

energy = []
for k in range(1, 9):
    N, a = F(k**3), F(1, k)
    raw = -F(1,6) - a**3*N/3
    value = -F(1,6) - a*N
    normalized_energy = raw - F(1,6)
    normalized_value = value - F(1,12)
    derivative = -a*a*N
    assert raw == -F(1,2)
    assert normalized_energy == -F(2,3)
    assert normalized_value == -F(1,4) - k*k
    assert derivative == -k
    energy.append({"k":k,"N":str(N),"a":str(a),"E":str(raw),"value_at_y":str(value),
                   "normalized_E":str(normalized_energy),"normalized_y":str(normalized_value),
                   "pencil_derivative":str(derivative)})

states = []
def gluing(types):
    if types in states:
        return
    states.append(types)
    if "A" in types:
        i = types.index("A")
        gluing(types[:i]+"C"+types[i+1:])
        gluing(types[:i]+"D"+types[i+1:])
gluing("AA")
assert all("A" not in x for x in states if x in ("CC","CD","DC","DD"))

data = {"scope":"Exhaustive finite rational diagnostics only, not analytic correctness",
        "period_box":{"domain":"[-4,4]^4","tested":9**4,"solutions":periods},
        "plucker":{"vector_domain":"{-1,0,1}^4 squared","tested":len(rows),"failures":failures,
                    "exterior_dimension":6,"symmetric_coefficient_dimension":21,
                    "finite_strong_square_span_rank":20,"rank_after_nondecomposable_witness":21},
        "energy_pencil":{"domain":"N=k^3,a=1/k,1<=k<=8","rows":energy},
        "gluing_states":states,
        "not_computationally_established":["Gruson theorem","coherent analytic descent","Picard vanishing",
           "quotient properness","meromorphic-field constancy","nonalgebraization","current products",
           "all-real-N branch identities","measure finite energy","unbounded E1 existence or uniqueness"]}
(OUT/"diagnostics.json").write_text(json.dumps(data,indent=2)+"\n")
(OUT/"environment.json").write_text(json.dumps({"python":sys.version,"platform":platform.platform(),
    "machine":platform.machine(),"cwd":str(ROOT),"libraries":"Python standard library only",
    "heavy_compute":"not required; no supervised facility supplied or used"},indent=2)+"\n")
print(json.dumps(data,indent=2))
