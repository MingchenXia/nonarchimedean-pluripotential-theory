import hashlib
import json
import platform
import shutil
import sys
from pathlib import Path

BASE = Path('/workspace/research/campaigns/round-273/P16.7-whole385387-full-bounded-dual-and-intrinsic-E1')
HERE = Path(__file__).resolve().parent

def digest(data):
    return hashlib.sha256(data).hexdigest()

def tex_escape(text):
    replacements = {'\\':r'\textbackslash{}', '{':r'\{', '}':r'\}',
                    '_':r'\_', '^':r'\textasciicircum{}', '&':r'\&',
                    '%':r'\%', '#':r'\#', '$':r'\$', '~':r'\textasciitilde{}'}
    return ''.join(replacements.get(char,char) for char in text)

def main():
    evidence = HERE/'evidence'; evidence.mkdir(exist_ok=True)
    packets = {}
    records = []
    for short,filename in [('math','mathematical-packet-bindings.json'),
                           ('primary','private-primary-packet-bindings.json'),
                           ('book','book-context-bindings.json')]:
        raw = (BASE/filename).read_bytes()
        target = HERE/'inputs-LOCALONLY'/filename
        target.parent.mkdir(exist_ok=True)
        target.write_bytes(raw)
        items = json.loads(raw); packets[short] = items
        for i,item in enumerate(items):
            content = Path(item['path']).read_bytes()
            records.append({'packet':short,'index':i,'path':item['path'],
                            'bytes':len(content),'sha256':digest(content),
                            'binding_matches':len(content)==item['bytes'] and digest(content)==item['sha256']})
    (evidence/'packet-identities.json').write_text(json.dumps(records,indent=2)+'\n')
    def read(index): return Path(packets['math'][index]['path']).read_text()
    comp = read(171); tex = read(143)
    actual = json.loads(read(146)); expanded = json.loads(read(148))
    versions = []
    for a,e in zip(actual['notes'],expanded['notes']):
        versions.append({'id':a['id'],'actual_bytes':len(a['text'].encode()),
                         'expanded_bytes':len(e['text'].encode()),
                         'actual_sha256':digest(a['text'].encode()),
                         'expanded_sha256':digest(e['text'].encode()),
                         'different':a['text']!=e['text'],
                         'entire_expanded_body_in_companion':e['text'] in comp})
    missing = []; order_end = 0; unique_segments = []
    for number,line in enumerate(read(142).split('\n'),1):
        if not line: continue
        if line.startswith('## '): encoded = r'\subsection*{'+tex_escape(line[3:])+'}'
        elif line.startswith('# '): encoded = r'\section*{'+tex_escape(line[2:])+'}'
        else: encoded = tex_escape(line)
        position = tex.find(encoded, order_end)
        if position < 0:
            missing.append({'line':number,'expected':encoded[:240]})
        else:
            if tex[order_end:position].strip():
                unique_segments.append(tex[order_end:position])
            order_end = position+len(encoded)
    if tex[order_end:].strip():
        unique_segments.append(tex[order_end:])
    (evidence/'TeX-unique-segments.json').write_text(json.dumps(unique_segments,indent=2)+'\n')
    report = {'versions':versions,
              'md_is_exact_companion_suffix':comp.endswith(read(142)),
              'ledger_is_entire_companion_substring':read(181).strip() in comp,
              'TeX_all_MD_nonempty_lines_encoded_in_order':not missing,
              'missing_encoded_lines':missing,
              'TeX_unique_segment_count':len(unique_segments),
              'identity_is_not_physical_reading_or_correctness':True}
    (evidence/'version-identities.json').write_text(json.dumps(report,indent=2)+'\n')
    selected_math = [134,136,137,138,139,140,141,142,143,144,146,148,156,171,173,174,175,176,177,179,180,181,182]
    selected_primary = [4,5,6,7,22,23,24,120,121,243,244,247,248,313,314,315,316,317,318,320,321,322,323,324,327,328,329]
    for short,indices in [('math',selected_math),('primary',selected_primary),('book',range(17))]:
        folder = HERE/({'math':'inputs-LOCALONLY','primary':'sources-LOCALONLY','book':'book-context-LOCALONLY'}[short])
        folder.mkdir(exist_ok=True)
        for index in indices:
            source = Path(packets[short][index]['path'])
            shutil.copyfile(source,folder/(str(index)+'-'+source.name))
    environment = {'python':sys.version,'platform':platform.platform(),
                   'working_directory':str(HERE),'implementation':'own standard library only',
                   'native_model_configuration':'not independently inspectable here; no extra model call',
                   'supplied_program_execution_or_import':False,
                   'supervised_compute_facility':'none supplied; no heavy computation needed',
                   'compiler_or_external_mutation':False}
    (evidence/'environment.json').write_text(json.dumps(environment,indent=2)+'\n')
    print(json.dumps({'binding_rows':len(records),'all_binding_rows_match':all(x['binding_matches'] for x in records),
                      'version_rows':versions,'TeX_missing_line_count':len(missing),
                      'missing_examples':missing[:3],'mechanical_evidence_only':True},indent=2))

if __name__=='__main__': main()
