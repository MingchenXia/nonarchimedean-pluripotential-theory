def sha(data):
    return hashlib.sha256(data).hexdigest()

def escape(s):
    table = {'\\':r'\textbackslash{}', '{':r'\{', '}':r'\}',
             '_':r'\_', '^':r'\textasciicircum{}', '&':r'\&',
             '%':r'\%', '#':r'\#', '$':r'\$', '~':r'\textasciitilde{}'}
    return ''.join(table.get(c,c) for c in s)

def encode(body):
    result=[]
    for line in body.splitlines():
        if line.startswith('## '): result.append(r'\subsection*{'+escape(line[3:])+'}')
        elif line.startswith('# '): result.append(r'\section*{'+escape(line[2:])+'}')
        elif line.strip(): result.append(escape(line))
        else: result.append('')
    return '\n'.join(result)+'\n'
