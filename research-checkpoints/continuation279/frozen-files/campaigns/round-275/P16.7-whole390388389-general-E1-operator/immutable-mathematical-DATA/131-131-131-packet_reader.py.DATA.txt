#!/usr/bin/env python3
"""Own inert-data reader. A display record is not a mathematical audit."""
from pathlib import Path
import argparse
import hashlib
import json

ROOT = Path(__file__).resolve().parents[1]
parser = argparse.ArgumentParser()
parser.add_argument("path")
parser.add_argument("first", type=int)
parser.add_argument("last", type=int)
args = parser.parse_args()
path = Path(args.path)
raw = path.read_bytes()
parts = raw.decode("utf-8").split("\n")
lines = [part + "\n" for part in parts[:-1]]
if parts[-1]:
    lines.append(parts[-1])
assert 1 <= args.first <= args.last <= len(lines)
record = {"path": str(path), "sha256": hashlib.sha256(raw).hexdigest(),
          "first_line": args.first, "last_line": args.last,
          "total_lines": len(lines), "status": "requested_display_only"}
(ROOT / "evidence").mkdir(exist_ok=True)
with (ROOT / "evidence" / "display_requests.jsonl").open("a") as out:
    out.write(json.dumps(record) + "\n")
print("".join(lines[args.first-1:args.last]), end="")
