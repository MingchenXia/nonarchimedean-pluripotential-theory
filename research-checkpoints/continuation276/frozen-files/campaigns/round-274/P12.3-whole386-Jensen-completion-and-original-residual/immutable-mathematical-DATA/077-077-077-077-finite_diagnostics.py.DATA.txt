def det(a):
    if not a:return F(1)
    return sum((-1)**j*a[0][j]*det([row[:j]+row[j+1:] for row in a[1:]]) for j in range(len(a)))
def inverse(a):
    n=len(a);b=[[F(x) for x in row]+[F(i==j) for j in range(n)] for i,row in enumerate(a)]
    for j in range(n):
        q=next(i for i in range(j,n) if b[i][j]);b[j],b[q]=b[q],b[j]
        d=b[j][j];b[j]=[x/d for x in b[j]]
        for i in range(n):
            if i!=j:
                d=b[i][j];b[i]=[x-d*y for x,y in zip(b[i],b[j])]
    return [row[n:] for row in b]
