def esc(s):
    substitutions={'\\':r'\textbackslash{}','{':r'\{','}':r'\}','$':r'\$','&':r'\&','#':r'\#','%':r'\%','_':r'\_','~':r'\textasciitilde{}','^':r'\textasciicircum{}'}
    return ''.join(substitutions.get(c,c) for c in s)

def prose(s):
    # Preserve the author's literal TeX math, escape all other text.
    bits=re.split(r'(\\\(.*?\\\)|\\\[.*?\\\])',s,flags=re.S)
    return ''.join(b if j%2 else esc(b) for j,b in enumerate(bits))
