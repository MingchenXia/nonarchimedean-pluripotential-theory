def hull(points):
    p=sorted(set(points))
    if len(p)<3:return p
    def cross(o,a,b):return (a[0]-o[0])*(b[1]-o[1])-(a[1]-o[1])*(b[0]-o[0])
    lo=[];hi=[]
    for a in p:
        while len(lo)>1 and cross(lo[-2],lo[-1],a)<=0:lo.pop()
        lo.append(a)
    for a in reversed(p):
        while len(hi)>1 and cross(hi[-2],hi[-1],a)<=0:hi.pop()
        hi.append(a)
    return lo[:-1]+hi[:-1]
def rawarea(points):
    h=hull(points)
    if len(h)<3:return F(0)
    return abs(sum(h[i][0]*h[(i+1)%len(h)][1]-h[i][1]*h[(i+1)%len(h)][0] for i in range(len(h))))
def minkowski(a,b):return [(x[0]+y[0],x[1]+y[1]) for x in a for y in b]
def scaled(a,k):return [(k*x,k*y) for x,y in a]
z=F(0);o=F(1)
square=[(z,z),(o,z),(z,o),(o,o)]
upper=[(o,z),(z,o),(o,o)]
lower=[(z,z),(o,z),(z,o)]
edge=[(o,z),(z,o)]
grid=[F(i,8) for i in range(9)]
depths=[(F(1,8),F(1,4),F(1)),(F(1,16),F(1,8),F(1,4)),
        (F(1),F(2),F(3)),(F(1,3),F(5,7),F(11,5))]
cases=0;areas=0;scalar=0
for values in product(grid,repeat=3):
    if not values[0]>=values[1]>=values[2]:continue
    a=list(values)+[z]
    beta=[a[k]-a[k+1] for k in range(3)]
    mass=[a[k]**2-a[k+1]**2 for k in range(3)]
    assert all(m>=0 for m in mass) and sum(mass)==a[0]**2<=1
    gauss=minkowski(scaled(square,1-a[0]),scaled(upper,a[0]))
    assert rawarea(gauss)==2-a[0]**2
    areas+=1
    for k in range(3):
        poly=minkowski(scaled(lower,beta[k]),scaled(edge,a[k+1]))
        assert rawarea(poly)==mass[k]
        # Exact fixed-fibre mixed-product expansion.
        expansion=beta[k]**2+2*beta[k]*sum(beta[k+1:])
        assert expansion==mass[k]
        areas+=1
    for c in depths:
        d=lambda t:sum(beta[k]*min(t,c[k]) for k in range(3))
        moments=sum(mass[k]*d(c[k]) for k in range(3))
        tail=sum(a[k]**3*(c[k]-(c[k-1] if k else 0)) for k in range(3))
        assert moments==tail
        E=-F(1,3)-tail/3
        P=-F(1,2)-tail
        assert E-P==F(1,6)+2*tail/3
        assert 2-a[0]**2+sum(mass)==2
        for t in [F(j,16) for j in range(65)]:
            by_intervals=sum(a[k]*max(z,min(t,c[k])-(c[k-1] if k else 0)) for k in range(3))
            assert d(t)==by_intervals
            scalar+=1
        cases+=1
