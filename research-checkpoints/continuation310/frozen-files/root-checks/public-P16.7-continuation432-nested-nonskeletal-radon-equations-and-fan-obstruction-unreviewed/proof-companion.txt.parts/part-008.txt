def dot(x,y):
    return sum((a*b for a,b in zip(x,y)), Q(0))

def clip(poly, a, b):
    out = []
    for p,q in zip(poly, poly[1:]+poly[:1]):
        fp,fq = dot(a,p)-b, dot(a,q)-b
        if fp <= 0:
            out.append(p)
        if (fp < 0 and fq > 0) or (fp > 0 and fq < 0):
            t = fp/(fp-fq)
            out.append(tuple(p[k]+t*(q[k]-p[k]) for k in range(2)))
    return out

def area(poly):
    return abs(sum((p[0]*q[1]-p[1]*q[0] for p,q in zip(poly,poly[1:]+poly[:1])),Q(0)))/2

def full_cells(sites, weights):
    results = []
    for i,x in enumerate(sites):
        half = Q(1,2)
        p = [(x[0]-half,x[1]-half),(x[0]+half,x[1]-half),
             (x[0]+half,x[1]+half),(x[0]-half,x[1]+half)]
        for j,z in enumerate(sites):
            for k in product(range(-2,3),repeat=2):
                if i==j and k==(0,0):
                    continue
                y = tuple(z[l]+k[l] for l in range(2))
                a = tuple(y[l]-x[l] for l in range(2))
                b = (dot(y,y)-dot(x,x))/2+weights[i]-weights[j]
                p = clip(p,a,b)
        results.append((p,area(p)))
    return results
