notes=[]
for i,(summary,detail,support) in enumerate(specs,1):
    assert set(support)<={n['id'] for n in notes}
    note=dict(id=f'n{i}',summary=summary,detailedSummary=detail,text=Path(f'note-n{i}.txt').read_text().strip(),support=support)
    notes.append(note)
    Path(f'actual-n{i}-all-five-fields.json').write_text(json.dumps(note,indent=2)+'\n')

     'actual-n1-all-five-fields.json','actual-n2-all-five-fields.json','actual-n3-all-five-fields.json','actual-n4-all-five-fields.json',

for name in own:

    parts.extend([f'\n===== WHOLE OWN BODY {name}; bytes={len(raw)}; SHA256={sha(raw)} =====\n'.encode(),raw])
    manifest.append(dict(kind='own',name=name,bytes=len(raw),sha256=sha(raw)))
companion=b'\n'.join(parts)
records=json.loads(Path('retained-mathematical-bindings.json').read_text())
# Start with the large safe supplied mathematical companion, then preserve
# all current unique bodies, including separate original actual/formal fields.
records.sort(key=lambda e:e['bytes'],reverse=True)
for entry in records:
    raw=Path(entry['local']).read_bytes()
    assert len(raw)==entry['bytes'] and sha(raw)==entry['sha256']
    already=raw in companion
    if not already:
        header=f'\n===== WHOLE SUPPLIED MATHEMATICAL INERT DATA index={entry["index"]}; {entry["local"]}; bytes={len(raw)}; SHA256={sha(raw)} =====\n'.encode()
