def escape(t):
    substitutions = {'\\':r'\textbackslash{}', '{':r'\{', '}':r'\}',
                     '$':r'\$', '&':r'\&', '%':r'\%', '#':r'\#', '_':r'\_',
                     '^':r'\textasciicircum{}', '~':r'\textasciitilde{}',
                     '<':r'\textless{}', '>':r'\textgreater{}'}
    return ''.join(substitutions.get(c,c) for c in t)
