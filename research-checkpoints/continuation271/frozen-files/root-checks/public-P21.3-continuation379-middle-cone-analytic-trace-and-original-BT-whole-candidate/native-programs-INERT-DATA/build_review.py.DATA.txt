def escape_text(s):
    mapping = {'\\': r'\textbackslash{}', '&': r'\&', '%': r'\%', '$': r'\$',
               '#': r'\#', '_': r'\_', '{': r'\{', '}': r'\}',
               '~': r'\textasciitilde{}', '^': r'\textasciicircum{}'}
    return ''.join(mapping.get(c, c) for c in s)

def tex_prose(s):
    chunks = re.split(r'(\\\(.*?\\\)|\\\[.*?\\\])', s, flags=re.S)
    return ''.join(c if c.startswith((r'\(', r'\[')) else escape_text(c)
                   for c in chunks)

def tex_section(s):
    lines = s.splitlines()
    return r'\section{' + escape_text(lines[0].removeprefix('# ')) + '}\n\n' + tex_prose('\n'.join(lines[1:]))

def validate_math(text):
    assert text.count(r'\(') == text.count(r'\)')
    assert text.count(r'\[') == text.count(r'\]')
    assert not any(ord(c) < 32 and c not in '\n\t' for c in text)
    for m in re.finditer(r'\\\(.*?\\\)|\\\[.*?\\\]', text, re.S):
        body = m.group()
        assert not re.search(r'(?<![A-Za-z\\])(langle|rangle|dim|sup|rho|theta|beta)(?![A-Za-z])', body), body
        depth = 0
        for k, c in enumerate(body):
            if k and body[k-1] == '\\':
                continue
            if c == '{':
                depth += 1
            if c == '}':
                depth -= 1
            assert depth >= 0, body
        assert depth == 0, body
