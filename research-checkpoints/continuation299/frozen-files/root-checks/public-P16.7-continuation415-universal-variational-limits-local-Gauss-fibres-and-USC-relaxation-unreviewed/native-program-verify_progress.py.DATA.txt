"""Own finite algebra and document checks; never executes supplied code."""
from fractions import Fraction as F
import json, re
from pathlib import Path
checks=[]
for k in range(4):
    # p(t)=t**k: p(1/2)-(p(0)+p(1))/2=(p'(0)-p'(1))/8.
    p0=F(1 if k==0 else 0);p1=F(1);pm=F(1,2)**k
    dp0=F(1 if k==1 else 0);dp1=F(k)
    lhs=pm-(p0+p1)/2;rhs=(dp0-dp1)/8
    assert lhs==rhs;checks.append({'monomial_degree':k,'midpoint_gap':str(lhs)})
# Columns A,q0,q1 of the exact -E(ta) polynomial.
cols=[[F(0),F(1),F(0),F(0)],[F(0),F(0),F(1),F(-1,3)],[F(0),F(0),F(0),F(1,3)]]
assert [sum(c) for c in cols]==[F(1),F(2,3),F(1,3)]
assert [[2*c[2],6*c[3]] for c in cols]==[[F(0),F(0)],[F(2),F(-2)],[F(0),F(2)]]
s=Path('scoped-paper.tex').read_text()
stack=[]
for m in re.finditer(r'\\(begin|end)\{([^}]+)\}',s):
    if m.group(1)=='begin':stack.append(m.group(2))
    else:
        assert stack and stack.pop()==m.group(2),(m.group(0),stack)
assert not stack
labels=re.findall(r'\\label\{([^}]+)\}',s);refs=re.findall(r'\\(?:eqref|ref)\{([^}]+)\}',s)
assert len(labels)==len(set(labels));assert set(refs)<=set(labels)
report={'finite_cubic_basis_identity':checks,'scaling_coefficient_columns':[[str(x) for x in c] for c in cols],'tex_environments_balanced':True,'unique_labels':len(labels),'undefined_references':[],'compiler_invoked':False,'mathematical_correctness_certificate':False}
Path('verify-progress-output.json').write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps(report,indent=2))
