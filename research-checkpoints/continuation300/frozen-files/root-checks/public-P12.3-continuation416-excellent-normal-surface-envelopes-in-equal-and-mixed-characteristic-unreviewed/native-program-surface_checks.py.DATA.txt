"""Tiny exact rational polyhedron diagnostics; no ring-filtration or proof certification."""
from fractions import Fraction as F
from itertools import combinations, product
from pathlib import Path
import json

Q=((-2,1),(1,-1))

def vertices(c):
    # inequalities row*z <= rhs, including z <= c and Q*z >= 0
    constraints=[((F(1),F(0)),c[0]),((F(0),F(1)),c[1])]
    constraints += [(tuple(F(-x) for x in row),F(0)) for row in Q]
    out=set()
    for (a,b),(d,e) in combinations(constraints,2):
        det=a[0]*d[1]-a[1]*d[0]
        if not det: continue
        z=((b*d[1]-a[1]*e)/det,(a[0]*e-b*d[0])/det)
        if all(sum(x*y for x,y in zip(row,z))<=rhs for row,rhs in constraints):out.add(z)
    return sorted(out)

def feasible(z,c):
    return all(z[i]<=c[i] for i in range(2)) and all(sum(F(a)*b for a,b in zip(row,z))>=0 for row in Q)

def main():
    cases=0; maxima=0
    for d in product((F(i,2) for i in range(-10,5)),repeat=2):
        c=tuple(min(x,F(0)) for x in d)
        vs=vertices(c)
        assert vs
        greatest=tuple(max(z[i] for z in vs) for i in range(2))
        assert feasible(greatest,c)
        for z,w in product(vs,repeat=2):
            assert feasible(tuple(max(a,b) for a,b in zip(z,w)),c)
            maxima+=1
        cases+=1
    c=(F(-1),F(-3));vs=vertices(c)
    greatest=tuple(max(z[i] for z in vs) for i in range(2))
    assert greatest==(F(-3,2),F(-3))
    result={'kind':'exhaustive finite rational linear-algebra diagnostic',
            'domain':'225 obstacles d_i=i/2, -10<=i<=4; c_i=min(d_i,0); all vertices from pairs of four inequalities',
            'Q':Q,'obstacles':cases,'vertex_pair_maxima':maxima,
            'example':{'d':['-1','-3'],'z_star':[str(z) for z in greatest],
                       'Q_z_star':[str(sum(F(a)*b for a,b in zip(row,greatest))) for row in Q]},
            'status':'PASS','limitation':'No actual ideals or semivaluations computed; finite checks do not certify the theorem.'}
    Path('reports').mkdir(exist_ok=True)
    Path('reports/surface-checks.json').write_text(json.dumps(result,indent=2)+'\n')
    print(json.dumps(result,indent=2))

if __name__=='__main__': main()
