"""Fresh bounded exact diagnostics. No supplied code is imported or run."""
import itertools
import json
from fractions import Fraction
from pathlib import Path

def add(a, b, p):
    c = dict(a)
    for j, v in b.items():
        c[j] = (c.get(j, 0) + v) % p
        if not c[j]:
            del c[j]
    return c

def mul(a, b, p):
    c = {}
    for i, u in a.items():
        for j, v in b.items():
            c[i+j] = (c.get(i+j, 0) + u*v) % p
    return {j: v for j, v in c.items() if v}

def determinant(a, p):
    n = len(a)
    out = {}
    for perm in itertools.permutations(range(n)):
        inversions = sum(perm[i] > perm[j] for i in range(n) for j in range(i+1,n))
        term = {0: (-1 if inversions % 2 else 1) % p}
        for i, j in enumerate(perm):
            term = mul(term, a[i][j], p)
        out = add(out, term, p)
    return out

def main():
    root_inputs = [
        {"p": 2, "q": 2, "monomials": [[i,j] for i in range(4) for j in range(2)]},
        {"p": 2, "q": 4, "monomials": [[i,j] for i in range(4) for j in range(2)]},
        {"p": 3, "q": 3, "monomials": [[i,j] for i in range(2) for j in range(2)]},
    ]
    root_count = 0
    for case in root_inputs:
        p, q, monomials = case['p'], case['q'], case['monomials']
        for cs in itertools.product(range(p), repeat=len(monomials)):
            grouped = {}
            expected = {}
            for exponent, c in zip(monomials, cs):
                if not c:
                    continue
                alpha = tuple(i % q for i in exponent)
                beta = tuple(i // q for i in exponent)
                grouped.setdefault(alpha, {})[beta] = c
                expected[tuple(exponent)] = c
            recovered = {}
            for alpha, series in grouped.items():
                for beta, c in series.items():
                    exponent = tuple(q*b+a for a,b in zip(alpha,beta))
                    recovered[exponent] = pow(c,q,p)
            assert recovered == expected
            root_count += 1

    trace_rows = []
    for p in (2,3,5):
        for n in range(2,6):
            if n % p == 0:
                continue
            # R=F_p[t], A=F_p[x], x=t^n. Trace(t^j)=n*x^(j/n) if n|j.
            matrix = [[{(i+j)//n: n % p} if (i+j) % n == 0 else {}
                       for j in range(n)] for i in range(n)]
            actual = determinant(matrix,p)
            # One cyclic permutation contributes to the positive-degree block.
            expected = {n-1: ((-1)**((n-1)*(n-2)//2)*pow(n,n,p)) % p}
            assert actual == expected
            trace_rows.append({'p':p,'n':n,'discriminant_coefficients':actual})

    obstacles = [Fraction(-2),Fraction(-1),Fraction(0),Fraction(1),Fraction(3)]
    masses = [Fraction(i,4) for i in range(5)]
    jensen_rows=[]
    for f in obstacles:
        m = min(t*f for t in masses)
        assert m == min(f,0)
        jensen_rows.append({'obstacle':str(f),'finite_grid_minimum':str(m)})

    # Root-basis coefficient projection phi selects exponent residue 0 mod q.
    # For q=3, Delta=x: source multiplication Delta^q yields phi(x)=x,
    # while multiplication Delta yields phi(x^(1/3))=0.
    frobenius_example={'p':3,'q':3,'Delta':'x','input_root':'1',
                       'correct_source_multiplier':'x^3','correct_output':'x',
                       'incorrect_source_multiplier':'x','incorrect_output':'0'}
    assert 3//3 == 1 and 3 % 3 == 0 and 1 % 3 != 0
    field_hom_rows=[{'possible_generator_image':a,'a^2+a+1_mod2':(a*a+a+1)%2} for a in (0,1)]
    assert all(row['a^2+a+1_mod2'] == 1 for row in field_hom_rows)

    report={'status':'PASS','root_recomposition_cases':root_count,
            'trace_discriminant_cases':len(trace_rows),'trace_rows':trace_rows,
            'jensen_grid_cases':len(jensen_rows),'jensen_rows':jensen_rows,
            'frobenius_precomposition_example':frobenius_example,
            'F4_to_F2_unital_hom_candidates':field_hom_rows,
            'F4_to_F2_unital_homs':0,
            'limits':'Exhaustive only on these finite polynomial/coefficient, matrix and candidate domains. Not proofs of formal-series, geometric, trace-tail or net claims.'}
    inputs={'root_inputs':root_inputs,'trace_primes':[2,3,5],'trace_n':[2,3,4,5],
            'trace_filter':'p does not divide n','jensen_obstacles':[str(x) for x in obstacles],
            'jensen_mass_grid':[str(x) for x in masses]}
    Path('reports').mkdir(exist_ok=True)
    Path('reports/finite-inputs.json').write_text(json.dumps(inputs,indent=2)+'\n')
    Path('reports/finite-checks.json').write_text(json.dumps(report,indent=2)+'\n')
    print(json.dumps(report,indent=2))

if __name__=='__main__':
    main()
