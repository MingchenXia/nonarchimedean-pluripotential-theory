def escaped(s):
    substitutions = {'\\':r'\textbackslash{}','{':r'\{','}':r'\}',
                     '#':r'\#','$':r'\$','%':r'\%','&':r'\&',
                     '_':r'\_','^':r'\textasciicircum{}','~':r'\textasciitilde{}'}
    return ''.join(substitutions.get(c,c) for c in s)

MATH = re.compile(r'(\\\[.*?\\\]|\\\(.*?\\\)|\$\$.*?\$\$|\$[^$]*?\$)',re.S)

def prose(s):
    out = []
    for line in s.splitlines(keepends=True):
        heading = re.match(r'^(#{1,6}) (.*?)(\n?)$',line)
        if heading:
            depth = len(heading[1])
            command = 'section' if depth <= 2 else 'subsection'
            out.append('\\'+command+'{'+escaped(heading[2])+'}\n')
        else:
            out.append(escaped(line))
    return ''.join(out)

def encode(s):
    # Math tokens are literal, only surrounding prose is escaped.
    return ''.join(token if MATH.fullmatch(token) else prose(token)
                   for token in MATH.split(s))

def read(name):
    return (ROOT/name).read_text()

def section(text, start, end=None):
    a=text.index(start)
    b=text.index(end,a+len(start)) if end else len(text)
    return text[a:b].strip()+'\n'
