def bound(i):

def digest(b): return hashlib.sha256(b).hexdigest()
def prose(s):
    table={'\\':r'\textbackslash{}','{':r'\{','}':r'\}','_':r'\_',
           '&':r'\&','#':r'\#','$':r'\$','%':r'\%',
           '~':r'\textasciitilde{}','^':r'\textasciicircum{}'}
    return ''.join(table.get(c,c) for c in s)
def native_encode(s):
    parts=[]
    for p in s.strip().split('\n\n'):
        if p.startswith('# '):
            continue
        if p.startswith('## '):
            parts.append(r'\section{'+prose(p[3:])+'}')
        else:
            atoms=re.split(r'(\\\[.*?\\\]|\\\(.*?\\\))',p,flags=re.S)
            parts.append(''.join(a if (a.startswith(r'\[') or a.startswith(r'\(')) else prose(a) for a in atoms))
    return '\n\n'.join(parts)
