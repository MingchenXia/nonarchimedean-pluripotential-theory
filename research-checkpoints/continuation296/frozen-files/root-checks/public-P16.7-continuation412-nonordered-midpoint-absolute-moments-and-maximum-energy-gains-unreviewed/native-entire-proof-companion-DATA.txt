\[
 \int d'f_j\wedge J\beta\wedge T

 \int gL_Tf=-Q_T(f,g),\qquad

\]

for continuous dpsh functions on $X$.

 E(b)-E(a)=\tfrac13\int(b-a)[\MA(a)+B(a,b)+\MA(b)].

\end{equation}

For the gradient bound use

\begin{lemma}\label{alltest}
For nonpositive $a,b,w\in\C$ of energies at least $-C$, put
$f=a-b$ and $I=\int f(\MA(b)-\MA(a))$. Then
\[
 I=Q_{\theta_a}(f)+Q_{\theta_b}(f)\ge0,\qquad
 Q_{\theta_w}(f)\le\Psi_C(I),\quad
 \Psi_C(t)=t/2+\sqrt{42Ct}.
\]
\end{lemma}
\begin{proof}
Set $g=(a+b)/2$, $h=w-g$. Product transgression gives
$Q_{\theta_w}(f)-I/2=Q_{\theta_a}(f,h)-Q_{\theta_b}(f,h)$;
both sides have the signed expression $-\int f\DD f\wedge\DD h$.
Since $|h|\le-w+(-a-b)/2$, \eqref{moments} bounds the MA moments
of $|h|$ by $6C$ and its mixed moments by $12C$.
Expanding
\[
 Q_{\theta_a}(h)=-\int h[B(a,w)-(\MA(a)+B(a,b))/2]
\]
gives $Q_{\theta_a}(h)\le12C+3C+6C=21C$, and the same holds
with $b$. Direct-sum Cauchy--Schwarz proves the result.
\end{proof}

\section{New nonordered absolute moments and finite maxima}
Put $\C_C=\{a\in\C:\sup_Xa=0,\ E(a)\ge-C\}$, $C\ge0$.

\[

       \{4/N+4A_N\sqrt{\Psi_{C_N}(t)}\}\right),\qquad
 H_C(t)=\Omega_C(t)+\sqrt{6C}

weights. For any fixed $N$ the second term tends to zero as $t\to0$;

\begin{theorem}\label{nonordered}
For arbitrary, not necessarily ordered, $a,b\in\C_C$, let
$f=a-b$, $I=\int f(\MA(b)-\MA(a))$, and $m=\max(a,b)$.
Then
\begin{align*}
 E((a+b)/2)-(E(a)+E(b))/2&=I/8,\\
 \int|f|\theta^2&\le\Omega_C(I),\\
 \int|f|\MA(a),\ \int|f|\MA(b)&\le H_C(I),\\
 0\le E(m)-E(a),\ E(m)-E(b)&\le H_C(I).
\end{align*}
For a fixed nonpositive bounded $w\in\C$, with $C'=\max(C,-E(w),0)$,
\[
 \int|f|\MA(w)\le\Omega_C(I)+
 \bigl(\sqrt{Q_{\theta_w}(w)}+\sqrt{Q_\theta(w)}\bigr)
              \sqrt{\Psi_{C'}(I)}.
\]
All these statements concern actual continuous weights and actual
analytic measures, including nonskeletal bounded test measures.
\end{theorem}
\begin{proof}
In \eqref{variations}, $Q_{\theta_{a_t}}(f)$ is affine in $t$.
Twice integrating the second derivative at $t=1/2$ gives exactly
$(Q_{\theta_a}(f)+Q_{\theta_b}(f))/8=I/8$.

The restrictions $g_a,g_b$ are each coordinate $1/2$-Lipschitz,
so $f|_S$ is coordinate $1$-Lipschitz. Every point of $S$ has
circle-$\ell^1$ distance at most $1/N$ from a grid vertex.

\[
 \operatorname{osc}_S f\le2/N+2A_N\sqrt{\Psi_{C_N}(I)}.
\]

$\min_Sf\le0\le\max_Sf$, so $\|f\|_{C(S)}$ is at most this
oscillation. Multiplication by the mass two gives the bound
$4/N+4A_N\sqrt{\Psi_{C_N}(I)}$ for $\int|f|\theta^2$.
Also $|f|\le1/2$ on $S$, giving the bound one. Infimize over $N$.

Here is the absolute-value Green step with its precise justification.
Set $F_\epsilon(q)=\sqrt{q^2+\epsilon^2}$. Its derivative is continuous
and bounded by one; $F_\epsilon(f)$ lies in each strong-gradient
domain and $Q_R(F_\epsilon(f))\le Q_R(f)$. We need no assertion
that $\DD(F_\epsilon(f)R)$ is Radon. Green pairs it with the fixed

$\MA(a)-\theta^2=\DD(a(\theta_a+\theta))$,
\[
 \int F_\epsilon(f)\MA(a)

Monotonicity and the tangent inequality at $a$ give
\[
 0\le E(m)-E(a)\le\int(m-a)\MA(a)

\]
The other endpoint is identical. No minimum or infinite maximum
is inserted in the psh class.
\end{proof}

\section{Prescribed-measure variational consequences without assumed attainment}

\begin{proof}
