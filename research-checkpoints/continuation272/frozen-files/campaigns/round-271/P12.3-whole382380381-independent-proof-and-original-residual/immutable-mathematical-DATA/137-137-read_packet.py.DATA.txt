"""Print exact bounded line windows of retained data and log those windows."""
from pathlib import Path
import json,sys,hashlib
records=json.loads(Path('evidence/packet-retention.json').read_text())
rid=sys.argv[1]; rec=next(r for r in records if r['id']==rid)
path=Path(rec['local']); lines=path.read_text().splitlines(keepends=True)
lo=int(sys.argv[2]) if len(sys.argv)>2 else 1
hi=int(sys.argv[3]) if len(sys.argv)>3 else len(lines)
print(f'{rid} {path} lines {lo}..{min(hi,len(lines))} of {len(lines)}')
for i in range(lo-1,min(hi,len(lines))):
    print(f'{i+1}: {lines[i]}',end='' if lines[i].endswith('\n') else '\n')
with Path('evidence/displayed-windows.jsonl').open('a') as f:
    f.write(json.dumps({'id':rid,'lo':lo,'hi':min(hi,len(lines)),'sha256':rec['actual_sha256'],'status':'display requested; reading/truncation judgment recorded separately'})+'\n')
