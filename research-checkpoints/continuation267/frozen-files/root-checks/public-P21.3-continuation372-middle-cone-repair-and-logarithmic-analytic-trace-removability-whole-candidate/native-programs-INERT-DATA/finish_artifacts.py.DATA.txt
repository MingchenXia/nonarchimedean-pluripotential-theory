def esc_text(s):
    # Text-mode escaping only; formulas and displays are preserved verbatim.
    result=[]; start=0
    for match in re.finditer(r'\\\(.*?\\\)|\\\[.*?\\\]',s,re.S):
        result.append(esc_plain(s[start:match.start()])); result.append(match.group()); start=match.end()
    result.append(esc_plain(s[start:])); return ''.join(result)
def esc_plain(s):
    return ''.join({'&':r'\&','%':r'\%','_':r'\_','#':r'\#','$':r'\$','{':r'\{','}':r'\}'}.get(c,c) for c in s)
