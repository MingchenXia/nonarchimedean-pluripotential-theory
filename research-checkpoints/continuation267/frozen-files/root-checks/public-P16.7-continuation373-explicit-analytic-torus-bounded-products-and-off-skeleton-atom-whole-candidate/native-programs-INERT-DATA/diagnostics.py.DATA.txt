#!/usr/bin/env python3
"""Own bounded exact calculations. These do not certify analytic theorems."""
from fractions import Fraction as F
from itertools import product
import json
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
checks = []

def check(name, lhs, rhs):
    assert lhs == rhs, (name, lhs, rhs)
    checks.append({'name': name, 'observed': str(lhs), 'expected': str(rhs), 'equal': True})

def poly_value(coeffs, x):
    return sum((a*x**k for k, a in enumerate(coeffs)), F(0))

def integral_01(coeffs):
    return sum((a/F(k+1) for k, a in enumerate(coeffs)), F(0))

def multiply(a, b):
    out = [F(0)]*(len(a)+len(b)-1)
    for i, x in enumerate(a):
        for j, y in enumerate(b):
            out[i+j] += x*y
    return out

G = [F(1,12), F(-1,2), F(1,2)]
dG = [F(-1,2), F(1)]
check('Green endpoint equality', poly_value(G,F(0)), poly_value(G,F(1)))
check('Green mean zero', integral_01(G), F(0))
check('Green minimum', poly_value(G,F(1,2)), F(-1,24))
check('Green endpoint maximum', poly_value(G,F(0)), F(1,12))
check('Green derivative jump at 0', poly_value(dG,F(0))-poly_value(dG,F(1)), F(-1))
check('Green derivative square integral', integral_01(multiply(dG,dG)), F(1,12))
check('Green oscillation', F(1,12)-F(-1,24), F(1,8))
check('h0 energy', -F(1,2)*integral_01(multiply(dG,dG)), -F(1,24))
check('u0 separable raw energy', F(2)*(-F(1,24)-F(1,24)), -F(1,6))
check('u0 value on tropical-zero fiber', -F(2)*poly_value(G,F(0)), -F(1,6))
check('rigid-point restricted dual objective', -F(1,6)-F(2)*(-F(1,6)), F(1,6))
check('theta square on unit square', F(2)*F(1), F(2))
check('simplex area', F(1,2)*F(1)*F(1), F(1,2))
check('off-skeleton max atom factorial times area', F(2)*F(1,2), F(1))

# A rational point in the softmax gradient simplex checks the determinant
# formula diag(p_1,p_2) - (p_1,p_2)^t(p_1,p_2), at epsilon=1.
p0,p1,p2 = F(1,2),F(1,3),F(1,6)
h11,h22,h12 = p1*(1-p1),p2*(1-p2),-p1*p2
check('softmax Hessian determinant at rational simplex point', h11*h22-h12*h12, p0*p1*p2)

# Exact finite test of (5.6) for h=aG, k=bG. Here a,b in [-1,0]
# make dy+h''=(1+a)dy-a delta_0 positive. The universal proof uses
# distributional integration by parts, not this test.
coefficients = [F(-1), F(-3,4), F(-1,2), F(-1,4), F(0)]
for a,b in product(coefficients, repeat=2):
    lhs = -b*b/F(24) + a*a/F(24)
    rhs = -(b-a)*a/F(12) - (b-a)**2/F(24)
    check(f'energy difference identity a={a}, b={b}', lhs, rhs)

# Exhaustive finite box only. Fractions keep negative character exponents
# exact. The elementary prime-factor proof in the paper establishes the
# result for all integer exponents, independently of this enumeration.
box = range(-4,5)
solutions = []
for a,b,c,d in product(box, repeat=4):
    if b == c and F(3)**a == F(2)**d:
        solutions.append((a,b,c,d))
expected = [(0,k,k,0) for k in box]
check('symmetric-character finite-box solution list', sorted(solutions), expected)
determinants = [{'k': k, 'determinant': -k*k,
                 'positive_definite': False} for k in box]
assert all(x['determinant'] <= 0 for x in determinants)

report = {
    'type': 'exact bounded rational diagnostics and exhaustive finite enumeration',
    'universal_analytic_certification': False,
    'input': {'integer_character_box': [-4,4], 'energy_coefficients': list(map(str,coefficients))},
    'checks_passed': len(checks),
    'checks': checks,
    'finite_period_enumeration': {
        'tuples_examined': len(box)**4,
        'symmetry_solutions': [list(x) for x in solutions],
        'matrices': determinants,
        'limitation': 'This finite list alone gives no conclusion outside the box.'
    },
    'commands': ['python scripts/diagnostics.py'],
    'dependencies': 'Python standard library only; no supplied code imported or executed'
}
(ROOT/'evidence'/'diagnostics.json').write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps({'checks_passed':len(checks),'tuples_examined':len(box)**4,
                  'symmetry_solutions':len(solutions),'universal_analytic_certification':False}))
