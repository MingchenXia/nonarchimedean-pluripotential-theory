def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def write_json(name, value):

def escape_tex(s):
    replacements = {
        '\\': r'\textbackslash{}', '&': r'\&', '%': r'\%', '$': r'\$',
        '#': r'\#', '_': r'\_', '{': r'\{', '}': r'\}',
        '~': r'\textasciitilde{}', '^': r'\textasciicircum{}',
        '–': '--', '—': '---',
    }
    return ''.join(replacements.get(ch, ch) for ch in s)


def prose_to_tex(s):
    # This converter's complete authored input grammar is headings, paragraphs,
    # inline code/emphasis, and literal TeX math delimited by \( \) or \[ \].
    spans = re.split(r'(\\\([\s\S]*?\\\)|\\\[[\s\S]*?\\\])', s)
    result = []
    for span in spans:
        if span.startswith(('\\(', '\\[')):
            result.append(span)
        else:
            bits = re.split(r'(`[^`]+`|\*[^*\n]+\*)', span)
            for bit in bits:
                if bit.startswith('`') and bit.endswith('`'):
                    result.append(r'\texttt{' + escape_tex(bit[1:-1]) + '}')
                elif bit.startswith('*') and bit.endswith('*'):
                    result.append(r'\emph{' + escape_tex(bit[1:-1]) + '}')
                else:
                    result.append(escape_tex(bit))
    return ''.join(result)


def section_to_tex(body):
    paragraphs = re.split(r'\n\s*\n', body.strip())
    out = []
    for para in paragraphs:
        if para.startswith('## '):
            out.append(r'\subsection{' + prose_to_tex(para[3:]) + '}')
        elif para.startswith('# '):
            out.append(r'\section{' + prose_to_tex(para[2:]) + '}')
        else:
            out.append(prose_to_tex(para))
    return '\n\n'.join(out) + '\n'
