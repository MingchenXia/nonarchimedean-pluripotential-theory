"""Fresh verification of data retention and finite evidence, not mathematics."""
from pathlib import Path
import hashlib
import json

HERE = Path(__file__).resolve().parent
ROOT = HERE.parent.parent
math = json.loads((ROOT/'mathematical-packet-bindings.json').read_text())
primary = json.loads((ROOT/'private-primary-packet-bindings.json').read_text())
result = json.loads((HERE/'notes.json').read_text())
assert result['candidate'] is False
tex = (HERE/'P12.3-marked-centres.tex').read_bytes()
companion = (HERE/'proof-companion.md').read_bytes()
seen = {'G331'}
for note in result['notes']:
    assert set(note) == {'id','summary','detailedSummary','text','support'}
    assert set(note['support']) <= seen
    assert note['text'] == (HERE/(note['id']+'.md')).read_text().strip()
    assert (HERE/(note['id']+'.tex')).read_bytes() in tex
    assert json.dumps(note, ensure_ascii=False, indent=2).encode() in companion
    seen.add(note['id'])
for rec in math:
    body = Path(rec['path']).read_bytes()
    assert body == (HERE/'retained'/Path(rec['path']).name).read_bytes()
    assert body in companion
for index in [4,5,6,7,8,15,26,2,3,27,28,32,36,37,0]:
    assert Path(math[index]['path']).read_bytes() in tex
for index in [10,11,12,13,14]:
    assert Path(math[index]['path']).read_bytes().rstrip() in Path(math[0]['path']).read_bytes()
for index, name in [(1,'original-task.json'), (27,'original-completion-criteria.txt'), (28,'original-residual.md')]:
    assert Path(math[index]['path']).read_bytes() == (HERE/name).read_bytes()
actual = json.loads(Path(math[26]['path']).read_text())
assert all(json.loads(Path(math[index]['path']).read_text()) == actual['notes'][index-4]
           for index in range(4,9))
assembled = json.loads(Path(math[15]['path']).read_text())
assert actual != assembled
grant = json.loads((HERE/'qualified-G331.json').read_text())
input_grant = next(n for n in json.loads((HERE/'input.json').read_text())['notes'] if n['id']=='G331')
assert grant == input_grant
excluded = 0
for index in [1244,1245,1246,1247,1248,1249,1250,1251,1254,1255,1256,1257,1258,1259]:
    body = Path(primary[index]['path']).read_bytes()
    assert body not in tex and body not in companion
    excluded += 1
assert all(r['binding_match'] for r in json.loads((HERE/'reports/input-hashes.json').read_text()))
assert json.loads((HERE/'reports/diagnostics.json').read_text())['status'] == 'PASS'
output = {'status': 'PASS', 'candidate': False, 'notes': len(result['notes']),
          'whole_incoming_source_bytes': len(Path(math[0]['path']).read_bytes()),
          'whole_incoming_companion_bytes': len(Path(math[30]['path']).read_bytes()),
          'all_43_math_inputs_retained': True, 'all_actual_expanded_fields_retained': True,
          'actual_and_assembled_are_distinct': True, 'G331_identical_to_input': True,
          'original_residual_byte_unchanged': True, 'ordered_supports_valid': True,
          'LOCALONLY_primary_carrier_exclusions': excluded,
          'compiler_invoked': False, 'mathematical_certification': False}
(HERE/'reports/verification.json').write_text(json.dumps(output, indent=2)+'\n')
hashes = {str(p.relative_to(HERE)): hashlib.sha256(p.read_bytes()).hexdigest()
          for p in HERE.rglob('*') if p.is_file() and p.name != 'own-hashes.json'
          and p.relative_to(HERE).parts[0] not in ['.git','.agents','.codex','.aws']}
(HERE/'reports/own-hashes.json').write_text(json.dumps(hashes, indent=2)+'\n')
print(json.dumps(output, indent=2))
