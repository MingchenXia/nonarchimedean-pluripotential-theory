"""Fresh inert reader. Never execute/import supplied programs.

Usage: python read_data.py math INDEX START STOP
       python read_data.py primary INDEX START STOP
Bounds are Unicode character offsets; -1 means end. Logs displayed data hashes.
"""
import hashlib
import json
import pathlib
import sys

ROOT = pathlib.Path('/workspace/research/campaigns/round-294/P12.3-whole422-irreducible-conductor-centres-and-normal-residual')
KIND = {'math': 'mathematical-packet-bindings.json',
        'primary': 'private-primary-packet-bindings.json',
        'book': 'book-context-bindings.json'}
kind, index, start, stop = sys.argv[1:]
binding = json.loads((ROOT / KIND[kind]).read_text())[int(index)]
data = pathlib.Path(binding['path']).read_text()
start = int(start)
stop = len(data) if int(stop) == -1 else int(stop)
assert 0 <= start <= stop <= len(data)
assert stop - start <= 26000, 'Use bounded outputs.'
body = data[start:stop]
print(json.dumps({'kind': kind, 'index': int(index), 'path': binding['path'],
                  'characters': len(data), 'start': start, 'stop': stop}))
print(body)
pathlib.Path('reports').mkdir(exist_ok=True)
with pathlib.Path('reports/displayed-data.jsonl').open('a') as f:
    f.write(json.dumps({'kind': kind, 'index': int(index), 'path': binding['path'],
                        'start': start, 'stop': stop, 'characters': len(data),
                        'display_sha256': hashlib.sha256(body.encode()).hexdigest()}) + '\n')
