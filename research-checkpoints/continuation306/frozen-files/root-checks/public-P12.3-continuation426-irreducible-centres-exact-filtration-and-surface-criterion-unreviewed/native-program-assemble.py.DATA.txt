def sha(body):
    return hashlib.sha256(body).hexdigest()

def escape(body):
    mapping = {'\\': r'\textbackslash{}', '{': r'\{', '}': r'\}',
               '&': r'\&', '%': r'\%', '$': r'\$', '#': r'\#', '_': r'\_',
               '^': r'\textasciicircum{}', '~': r'\textasciitilde{}',
               '<': r'\textless{}', '>': r'\textgreater{}'}
    return ''.join(mapping.get(ch, ch) for ch in body)
