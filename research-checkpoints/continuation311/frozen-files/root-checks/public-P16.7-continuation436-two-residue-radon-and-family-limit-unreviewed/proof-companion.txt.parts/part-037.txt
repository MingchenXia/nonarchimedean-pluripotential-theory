\begin{lemma}\label{alltest}

$f=a-b$ and $I=\int f(\MA(b)-\MA(a))$. Then
\[

\]
\end{lemma}

\section{New nonordered absolute moments and finite maxima}

\begin{theorem}\label{nonordered}

\begin{align*}
 E((a+b)/2)-(E(a)+E(b))/2&=I/8,\\
 \int|f|\theta^2&\le\Omega_C(I),\\
 \int|f|\MA(a),\ \int|f|\MA(b)&\le H_C(I),\\
 0\le E(m)-E(a),\ E(m)-E(b)&\le H_C(I).
\end{align*}
For a fixed nonpositive bounded $w\in\C$, with $C'=\max(C,-E(w),0)$,
\[
 \int|f|\MA(w)\le\Omega_C(I)+
 \bigl(\sqrt{Q_{\theta_w}(w)}+\sqrt{Q_\theta(w)}\bigr)
              \sqrt{\Psi_{C'}(I)}.
\]
All these statements concern actual continuous weights and actual
analytic measures, including nonskeletal bounded test measures.
\end{theorem}
\begin{proof}

$(Q_{\theta_a}(f)+Q_{\theta_b}(f))/8=I/8$.

 0\le E(m)-E(a)\le\int(m-a)\MA(a)

The other endpoint is identical. No minimum or infinite maximum

\section{Prescribed-measure variational consequences without assumed attainment}

\begin{abstract}
On the actual analytic quotient used in the current P16.7 packet, we
prove that a small cubic-energy midpoint deficit forces small absolute
moving moments against both endpoint Monge--Ampere measures, and small
energy gains under finite maximum. No order between the endpoints is
required. The proof fixes constants using actual coordinate Gauss fibres
and finite Gauss grids. It also gives uniform energy bounds for
variational maximizing nets for arbitrary prescribed mass-two measures
dominated by finite sums of bounded Monge--Ampere tests. Such nets are
Cauchy against each fixed bounded test and at every full-dimensional
point. These are actual bounded-weight estimates, not an analytic
representability theorem for their limits. Nonordered compactness,
obstacle orthogonality, and the full prescribed-measure equation remain
unresolved. No supplied verdict is a premise. The full historical
source audit requested by the operator is not completed here.
\end{abstract}

\section{All-stage bounded Green and energy, reconstructed}
For a positive closed $(1,1)$ current $T$, complete compact smooth
$(1,0)$ forms in the positive semidefinite pairing
$\int\alpha\wedge J\beta\wedge T$, locally using nonnegative
smooth cutoffs. Let $H_T$ consist of continuous functions with locally
uniform smooth approximants whose gradients converge strongly in all
these local seminorms. A uniformly zero sequence with a strong
gradient limit has zero gradient: pairing with a compact smooth form
$\beta$ gives, by closedness,
\[
 \int d'f_j\wedge J\beta\wedge T
 =-\int f_j d'(J\beta)\wedge T\longrightarrow0.
\]
The differentiated smooth coefficient has a finite measure on its
compact support; smooth Hilbert tests are dense. Partitions and
Leibniz patch local approximants. Bounded continuous multipliers act
by scalar approximation with norm bound $\|h\alpha\|\le
\|h\|_\infty\|\alpha\|$.

For $f\in H_T$ with $L_Tf=\DD(fT)$ signed Radon, smooth
approximation in both slots gives
\[
 \int gL_Tf=-Q_T(f,g),\qquad
 Q_{T,\chi}(f)=\tfrac12\int f^2\DD\chi\wedge T-\int\chi fL_Tf.
\]
The sheaf of $f\in H_T$ with $L_Tf\ge0$ contains smooth psh
functions and is locally uniformly closed. Indeed, for a uniformly
convergent sequence $f_j$ in this sheaf, choose cutoffs $\chi,\eta$,
$\eta=1$ near $\operatorname{supp}\chi$, and $|f_j|\le B$ on the
\emph{whole} support of $\eta$. With $\epsilon=\|f_j-f_k\|$ there,
positivity and the identity above give
\[
 Q_{T,\chi}(f_j-f_k)\le
 \tfrac12\epsilon^2\|\DD\chi\wedge T\|_{\rm TV}
 +2B\epsilon\|\chi\|_\infty\|\DD\eta\wedge T\|_{\rm TV}.
\]
Thus gradients are strongly Cauchy. Diagonal smooth approximants
and distributional positivity prove closure. Minimality includes
every continuous psh stage. Differences and finite patching give
\begin{equation}\label{green}
 Q_T(f,g)=-\int f\DD(gT)=-\int g\DD(fT),\qquad Q_T(f)\ge0
\end{equation}
for continuous dpsh functions on $X$.
Smooth $C^1$ one-variable composition with bounded derivative has
the strong-gradient chain rule, because uniform convergence of
the approximants gives uniform convergence of their derivative
multipliers, while their gradients converge strongly. This rule
uses continuous multipliers only.

Exact product terms have zero integral against $1$ by boundaryless
Stokes, so every $B(a,b)$ has the reference mass $M=2$, established
geometrically below. Define
\[
 E(a)=\tfrac13\int a[\theta^2+B(a,0)+\MA(a)].
\]
Cubic expansion and \eqref{green} give, for $a_t=(1-t)a+tb$,
\begin{equation}\label{variations}
 E'(a_t)=\int(b-a)\MA(a_t),\qquad E''(a_t)=-2Q_{\theta_{a_t}}(b-a).
\end{equation}
Hence $E$ is concave and monotone, $E(0)=0$, $E(a+c)=E(a)+2c$,
and
\begin{equation}\label{cocycle}
 E(b)-E(a)=\tfrac13\int(b-a)[\MA(a)+B(a,b)+\MA(b)].
\end{equation}
For $a\le b$, each nonnegative integral on the right is at most
$3(E(b)-E(a))$. The tangent inequality yields
\[
 D(\MA(q))=E(q)-\int q\MA(q),\quad
 D(\mu)=\sup_{p\in\C}\left(E(p)-\int p\,d\mu\right).
\]
Constants force mass two for a positive measure with finite $D$.
If $p,q,r\le0$ have $E\ge-C$, the nonnegative terms of $-3E(q)$
give $D(\MA(q))\le2C$, whence
\begin{equation}\label{moments}
 \int(-p)\MA(q)\le3C,\quad \int(-p)B(q,r)\le6C,
 \quad Q_{\theta_q}(p)\le6C,\quad Q_\theta(p)\le6C.
\end{equation}
For the mixed bound use $B(q,r)\le2\MA((q+r)/2)$ and concavity.
For the gradient bound use
$Q_{\theta_q}(p)=\int(-p)B(p,q)-\int(-p)B(0,q)$.

\begin{lemma}\label{alltest}
For nonpositive $a,b,w\in\C$ of energies at least $-C$, put
$f=a-b$ and $I=\int f(\MA(b)-\MA(a))$. Then
\[
 I=Q_{\theta_a}(f)+Q_{\theta_b}(f)\ge0,\qquad
 Q_{\theta_w}(f)\le\Psi_C(I),\quad
 \Psi_C(t)=t/2+\sqrt{42Ct}.
\]
\end{lemma}
\begin{proof}
Set $g=(a+b)/2$, $h=w-g$. Product transgression gives
$Q_{\theta_w}(f)-I/2=Q_{\theta_a}(f,h)-Q_{\theta_b}(f,h)$;
both sides have the signed expression $-\int f\DD f\wedge\DD h$.
Since $|h|\le-w+(-a-b)/2$, \eqref{moments} bounds the MA moments
of $|h|$ by $6C$ and its mixed moments by $12C$.
Expanding
\[
 Q_{\theta_a}(h)=-\int h[B(a,w)-(\MA(a)+B(a,b))/2]
\]
gives $Q_{\theta_a}(h)\le12C+3C+6C=21C$, and the same holds
with $b$. Direct-sum Cauchy--Schwarz proves the result.
\end{proof}

\section{Gauss normalization and actual fibre comparisons}
Let $S$ be the joint coordinate Gauss section modulo the deck group,
$S\simeq\R^2/\mathbb Z^2$. It is used to fix constants, not to
restrict prescribed measure support.

\begin{lemma}\label{gauss}
For $a\in\C$ the function $g_a(r)=a(\sigma(r))$ is separately
$1/2$-Lipschitz for circle distances, $a(x)\le g_a(\tau(x))$, and
$\sup_Xa=\max_Sa$. Consequently, if $\sup_Xa=0$, then
$-1/2\le a|_S\le0$. The actual measure $\theta^2$ is twice
Lebesgue measure on $S$, so its raw mass is two.
\end{lemma}
\begin{proof}
Fix $r_2$ and pass to the actual $z_2$ Gauss fibre
$\mathbb G_{m,\mathscr H(\eta_{e^{-r_2}})}^{\rm an}(z_1)$.
The analytic map sets $z_2$ to its canonical nonzero residue-field
element. Classical functoriality applies. The pulled-back local
weight glues to $\Phi(q)=q^2/2+r_2^2/2+g_a(q,r_2)$.
On an annulus $b\le q\le c$, subtract the affine interpolation
of its endpoint Gauss values. GR 6.4 bounds the difference by zero,
so $q^2/2+g_a(q,r_2)$ is finite convex. Period one gives subgradient
bounds $q-1/2\le p\le q+1/2$ using the secants to $q\pm1$.
Local absolute continuity gives $|\partial_qg_a|\le1/2$ almost
everywhere. Swap the coordinates for the other bound. Their combined
oscillation on the two circles is at most $1/2$.
Small rectangular polyannuli around an arbitrary $x$ have four
joint Gauss Shilov vertices. GR 6.4 bounds $(\rho+a)(x)$ by their
largest value. Shrinking the rectangle proves $a(x)\le g_a(\tau(x))$.

If $x$ is not coordinate Gauss, a Laurent polynomial $P=\sum c_Iz^I$
has strict cancellation $|P(x)|<\max|c_Iz^I(x)|$. This persists
on an ordinary neighborhood, where at least two terms realize
the maximum. Coordinate tropicalization there lies in a finite
tropical hypersurface of dimension at most one. The coordinate
$(2,2)$ form $\theta^2$ vanishes there by the CLD moment-form
dimension rule. Its measure is therefore supported on $S$.
On $S$, identity moments have saturated tangent lattice, degree one,
and calibration one by CLD 14.5.2--3. CLD 5.10.1--2 identifies the
Radon measure with the calibrated tropical integral. The Hessian of
$|r|^2/2$ is the identity; its raw wedge factor is $2!$, giving
density $2\,dr_1dr_2$. Quotient gluing gives the asserted measure.
\end{proof}

\begin{lemma}\label{paths}
For each finite grid $\mathscr D_N=(N^{-1}\mathbb Z/\mathbb Z)^2$
and its base vertex $o$, there are finitely many fixed nonpositive
$w_l\in\C$ and a finite $A_N$ such that, for every continuous dpsh $f$,
\[
 |f(\xi)-f(o)|\le A_N\max_l\sqrt{Q_{\theta_{w_l}}(f)}
 \qquad(\xi\in\mathscr D_N).
\]
\end{lemma}
\begin{proof}
Connect the vertices to $o$ by coordinate-radius paths. Each path
uses one coordinate at a time and partitions into finitely many
injective tropical boxes. In a horizontal piece with fixed $r_2$,
CLD 18.6.2 with $d=1$, $f=z_2$, $t=1$ states
\[
 T_{r_2}=\DD\max(0,\log(|z_2|/e^{-r_2}))
 =\delta_{z_2^{-1}(\eta_{e^{-r_2}})}.
\]
This is the integration current of the actual curve over its Gauss
residue field; that field is unchanged throughout this one piece.
The source space is separated, boundaryless, equidimensional; the
coordinate morphism is smooth of relative dimension one. These are
precisely the corollary's hypotheses. Its immediate proof uses the
relative curvature/intersection formula of 18.6.1 and the intersection
coefficient one on the Gauss fibre. No variable section is asserted
analytic, and no finite-extension degree is inserted.

Let $\rho_0$ be a centered quadratic reference on the box.
The finite continuous psh $\psi=\max(0,\log(|z_2|/e^{-r_2}))$
is bounded on its compact closure. Choose an inner ball containing
the piece, an outer collar, and $\delta,\epsilon>0$ so that
$w=\max(0,\delta+\epsilon\psi-\rho_0)$ is positive on the inner
ball and zero on the outer collar. This is possible by making the
inner ball smaller than the outer one and $\epsilon$ sufficiently
small. Its reference weight is a maximum of continuous psh slots;
it extends by zero to $\C$. Inside, $\theta_w=\epsilon T_{r_2}$.
A cutoff $0\le\chi\le1$, one on the path, gives
\[
 Q_{\theta_w}(f)\ge\epsilon\int_{\mathbb G_{m,\mathscr H(\eta)}}
          \chi\,d'f\wedge d''f.
\]
For smooth ambient $f$, its radial restriction on this actual curve
is piecewise smooth. Unit log norms on a compact annulus have finite
piecewise affine radial presentations. Canonical degree-one calibration
makes radial gradient energy $|\partial_q f|^2dq$, and other curve
branches contribute nonnegative energy. Cauchy--Schwarz therefore
bounds the squared endpoint difference by $\ell/\epsilon$ times
$Q_{\theta_w}(f)$, where $\ell$ is the finite radial length.
The same fibre-current inequality for differences of local smooth
approximants proves strong convergence of their curve gradients;
uniform convergence gives endpoints. Thus the bound holds for all
continuous dpsh $f$ in the reconstructed strong domain. Sum the
finitely many $\sqrt{\ell/\epsilon}$ over each path, take a maximum,
and subtract finite constants from the caps. Vertical pieces use $z_1$.
\end{proof}

For every grid include $w_0=0$ among the fixed tests; for $N=1$
take $A_1=0$. Thus all maxima over the fixed test list are defined.

\section{New nonordered absolute moments and finite maxima}
Put $\C_C=\{a\in\C:\sup_Xa=0,\ E(a)\ge-C\}$, $C\ge0$.
For the fixed grids above let
$C_N=\max(C,\max_l[-E(w_l)],0)$ and define
\[
 \Omega_C(t)=\min\left(1,\inf_{N\ge1}
       \{4/N+4A_N\sqrt{\Psi_{C_N}(t)}\}\right),\qquad
 H_C(t)=\Omega_C(t)+\sqrt{6C}
       \left(\sqrt t+\sqrt{\Psi_C(t)}\right).
\]
These finite geometric constants are independent of the pair of
weights. For any fixed $N$ the second term tends to zero as $t\to0$;
then $N\to\infty$ proves $\Omega_C(t),H_C(t)\to0$.

\begin{theorem}\label{nonordered}
For arbitrary, not necessarily ordered, $a,b\in\C_C$, let
$f=a-b$, $I=\int f(\MA(b)-\MA(a))$, and $m=\max(a,b)$.
Then
\begin{align*}
 E((a+b)/2)-(E(a)+E(b))/2&=I/8,\\
 \int|f|\theta^2&\le\Omega_C(I),\\
 \int|f|\MA(a),\ \int|f|\MA(b)&\le H_C(I),\\
 0\le E(m)-E(a),\ E(m)-E(b)&\le H_C(I).
\end{align*}
For a fixed nonpositive bounded $w\in\C$, with $C'=\max(C,-E(w),0)$,
\[
 \int|f|\MA(w)\le\Omega_C(I)+
 \bigl(\sqrt{Q_{\theta_w}(w)}+\sqrt{Q_\theta(w)}\bigr)
              \sqrt{\Psi_{C'}(I)}.
\]
All these statements concern actual continuous weights and actual
analytic measures, including nonskeletal bounded test measures.
\end{theorem}
\begin{proof}
In \eqref{variations}, $Q_{\theta_{a_t}}(f)$ is affine in $t$.
Twice integrating the second derivative at $t=1/2$ gives exactly
$(Q_{\theta_a}(f)+Q_{\theta_b}(f))/8=I/8$.

The restrictions $g_a,g_b$ are each coordinate $1/2$-Lipschitz,
so $f|_S$ is coordinate $1$-Lipschitz. Every point of $S$ has
circle-$\ell^1$ distance at most $1/N$ from a grid vertex.
Lemmas \ref{alltest} and \ref{paths} give
\[
 \operatorname{osc}_S f\le2/N+2A_N\sqrt{\Psi_{C_N}(I)}.
\]
Normalization $\max_Sg_a=\max_Sg_b=0$ implies
$\min_Sf\le0\le\max_Sf$, so $\|f\|_{C(S)}$ is at most this
oscillation. Multiplication by the mass two gives the bound
$4/N+4A_N\sqrt{\Psi_{C_N}(I)}$ for $\int|f|\theta^2$.
Also $|f|\le1/2$ on $S$, giving the bound one. Infimize over $N$.

Here is the absolute-value Green step with its precise justification.
Set $F_\epsilon(q)=\sqrt{q^2+\epsilon^2}$. Its derivative is continuous
and bounded by one; $F_\epsilon(f)$ lies in each strong-gradient
domain and $Q_R(F_\epsilon(f))\le Q_R(f)$. We need no assertion
that $\DD(F_\epsilon(f)R)$ is Radon. Green pairs it with the fixed
signed Radon $\DD(aR)$. Since
$\MA(a)-\theta^2=\DD(a(\theta_a+\theta))$,
\[
 \int F_\epsilon(f)\MA(a)
 \le\int F_\epsilon(f)\theta^2+
 \sqrt{Q_{\theta_a}(f)Q_{\theta_a}(a)}+
 \sqrt{Q_\theta(f)Q_\theta(a)}.
\]
Uniform convergence of $F_\epsilon(f)$ to $|f|$ passes each
integral against its finite signed measure. There is no indicator
multiplier or assumed strong absolute-value limit.
Now $Q_{\theta_a}(f)\le I$, $Q_\theta(f)\le\Psi_C(I)$, and
both $a$-slot energies are at most $6C$ by \eqref{moments}.
This proves the $a$ endpoint bound; use $b$ for the other.
Use the same argument with $w$ instead of $a$ for the fixed-test bound.

Monotonicity and the tangent inequality at $a$ give
\[
 0\le E(m)-E(a)\le\int(m-a)\MA(a)
 =\int(b-a)_+\MA(a)\le H_C(I).
\]
The other endpoint is identical. No minimum or infinite maximum
is inserted in the psh class.
\end{proof}

\section{Prescribed-measure variational consequences without assumed attainment}
For a positive Radon measure $\mu$ of raw mass two define
$\mathcal F_\mu(a)=E(a)-\int a\,d\mu$. This is invariant under
constants. If $D(\mu)<\infty$, put
$D_C(\mu)=\sup_{a\in\C_C}\mathcal F_\mu(a)$, so
$0\le D_C(\mu)\le D(\mu)<\infty$.

\begin{corollary}\label{maximizing}
If $a,b\in\C_C$ have deficits
$\varepsilon_a=D_C(\mu)-\mathcal F_\mu(a)$ and
$\varepsilon_b=D_C(\mu)-\mathcal F_\mu(b)$, then
$I(a,b)\le4(\varepsilon_a+\varepsilon_b)$.
Consequently every maximizing net on $\C_C$ is Cauchy in
$L^1(\MA(w))$ for every fixed $w\in\C$, and Cauchy at every
full-dimensional point. Its maximum energy gains tend to zero.
\end{corollary}
\begin{proof}
The midpoint $h=(a+b)/2$ has $E(h)\ge-C$, $\sup h\le0$.
Normalizing $h-\sup h$ preserves its functional and increases its
energy, so it is in $\C_C$. Linearity of the measure term and the
exact midpoint formula imply
\[
 I/8=\mathcal F_\mu(h)-
       (\mathcal F_\mu(a)+\mathcal F_\mu(b))/2
 \le(\varepsilon_a+\varepsilon_b)/2.
\]
Use Theorem \ref{nonordered} and \eqref{point}. A maximizing net
exists as an approximation to the real supremum; no maximizer is
assumed. The argument uses no countable dense set or full-support
Radon measure on $X$.
\end{proof}

\begin{theorem}\label{coercive}
Suppose $\mu\ge0$, $\mu(X)=2$, and
$\mu\le\sum_{l=1}^k\beta_l\MA(w_l)$ for fixed bounded
$w_l\in\C$, $\beta_l>0$. This is domination of a prescribed
measure, not the assertion $\mu=\MA(u)$. Then $D(\mu)<\infty$,
and every normalized net maximizing the \emph{full} bounded dual
eventually lies in one fixed $\C_C$. Hence all the nonordered
Cauchy consequences of Corollary \ref{maximizing} hold for that net,
including $L^1(\mu)$ convergence and convergence of its real energies.
\end{theorem}
\begin{proof}
Normalize each $w_l$ and put $c_l=-E(w_l)\ge0$,
\[
 A_l=\sqrt6\bigl(\sqrt{Q_{\theta_{w_l}}(w_l)}+
                         \sqrt{Q_\theta(w_l)}\bigr).
\]
For normalized $a\in\C$, set $e=-E(a)\ge0$.
Gauss normalization gives $\int(-a)\theta^2\le1$.
Transgression against $w_l$ and Green, followed by \eqref{moments}
with energy bound $\max(e,c_l)$, yield
\[
 \int(-a)\MA(w_l)\le1+A_l\sqrt{\max(e,c_l)}
 \le1+A_l\sqrt{c_l}+A_l\sqrt e.
\]
Thus, with $B_0=\sum\beta_l(1+A_l\sqrt{c_l})$,
$B_1=\sum\beta_l A_l$,
\[
 \int(-a)\mu\le B_0+B_1\sqrt e,\qquad
 \mathcal F_\mu(a)\le-e+B_0+B_1\sqrt e
 \le B_0+B_1^2/4.
\]
