\begin{lemma}[Strict compact dominance]
Let $X$ be compact, let USC $g_k$ decrease pointwise to $g$, and let $a$ be
finite continuous with $g<a$ everywhere. Then $g_k<a$ everywhere for all
sufficiently large $k$.
\end{lemma}
\begin{proof}
The ordinary open sets $\{g_k<a\}$ increase and cover $X$.
A finite subcover has a largest index. The proof includes points where
$g=-\infty$ and does not require metrizability.
\end{proof}

\begin{theorem}[Actual countable monotone closure]
If $u_k\in\G$ decrease pointwise and $L=\inf_kE(u_k)>-\infty$, their actual
pointwise infimum $u$ belongs to $\G$, and $E(u_k)\downarrow E(u)$.
\end{theorem}
\begin{proof}
Choose $f_{k,j}\in\C$ decreasing in $j$ to $u_k$ and enumerate every pair
$(k,j)$ as continuous functions $h_1,h_2,\ldots$. Then
$\inf_mh_m=\inf_ku_k=u$ at every point. Put
\[
 \epsilon_n=2^{-n},\qquad r_n=\sum_{\ell>n}\epsilon_\ell=2^{-n}.
\]
Each continuous term is bounded on compact $X$; no uniform lower bound on
the entire presentation is assumed. Initialize $k_0=0$.
Inductively choose $k_n\geq\max(n,k_{n-1})$ with
$u_{k_n}<h_m+\epsilon_n$ on all of $X$ for $m\leq n$.
The lemma applies to the finite continuous minimum of these bounds.
That minimum is used only as a bound, never as a psh potential.
For $n>1$, also $u_{k_n}\leq u_{k_{n-1}}\leq v_{n-1}$, where the preceding
stage has selected $v_{n-1}$. Apply the lemma to the presentation of $u_{k_n}$
to choose $j_n$ such that $v_n=f_{k_n,j_n}$ obeys
\[
 v_n<h_m+\epsilon_n\ (m\leq n),\qquad

\]
All bounds are strict and finite continuous, so a single index works.

\[
 a_n-a_{n-1}<\epsilon_n+r_n-r_{n-1}=0\quad(n>1).
\]

$m$, since $a_n<h_m+\epsilon_n+r_n$ for $n\geq m$.

$E(a_n)\geq E(u_{k_n})\geq L$. This proves $u\in\G$ without presupposing

$\eta\to0$ to obtain $L\leq E(u)$.
\end{proof}

\begin{theorem}[Summable nonpositive convex series]
